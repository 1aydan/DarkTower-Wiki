150int32 ADTGameState::AwardExperienceToAllPlayers(const int64 ExperienceAmount)
151{
152 if (!HasAuthority() || ExperienceAmount <= 0)
153 {
154 return 0;
155 }
156
157 int32 AwardedPlayerCount = 0;
158 for (APlayerState* PlayerState : PlayerArray)
159 {
160 ADTPlayerState* DTPlayerState = Cast<ADTPlayerState>(PlayerState);
161 if (!DTPlayerState)
162 {
163 continue;
164 }
165
166 if (UDTPlayerProgressionComponent* ProgressionComponent = DTPlayerState->GetPlayerProgressionComponent())
167 {
168 ProgressionComponent->AddExperience(ExperienceAmount);
