56EDataValidationResult UDTSkillTreeNodeDefinition::IsDataValid(FDataValidationContext& Context) const
57{
58 EDataValidationResult Result = Super::IsDataValid(Context);
59
60 if (!NodeTag.IsValid())
61 {
62 Context.AddError(FText::FromString(TEXT("Skill tree node has no valid NodeTag")));
63 Result = EDataValidationResult::Invalid;
64 }
65
66 if (MaxRank < 1)
67 {
68 Context.AddError(FText::FromString(TEXT("Skill tree node MaxRank must be >= 1")));
69 Result = EDataValidationResult::Invalid;
70 }
71
72 if (PointsPerRank < 1)
73 {
74 Context.AddError(FText::FromString(TEXT("Skill tree node PointsPerRank must be >= 1")));
75 Result = EDataValidationResult::Invalid;
76 }
77
78 const bool bGrantsPassive = PassiveEffectClass != nullptr;
79 const bool bGrantsAbility = GrantedAbilityClass != nullptr;
80 const bool bGrantsSet = !GrantedAbilitySet.IsNull();
81 if (!bGrantsPassive && !bGrantsAbility && !bGrantsSet)
82 {
83 Context.AddWarning(FText::FromString(TEXT("Skill tree node grants no rewards (no passive effect, granted ability, or ability set)")));
84 }
85
86 // Ability sets are granted once at rank 1 and don't scale per rank, so a multi-rank set node wastes ranks 2..N.
87 if (bGrantsSet && MaxRank > 1)
88 {
89 Context.AddWarning(FText::FromString(TEXT("Node grants an ability set but has MaxRank > 1; ability sets are granted once at rank 1 and do not scale per rank")));
90 }
91
92 return Result;
93}
