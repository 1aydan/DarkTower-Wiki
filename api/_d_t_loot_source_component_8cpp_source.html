39bool UDTLootSourceComponent::Internal_DropLoot(const FTransform& SpawnTransform, TArray<ADTItemActor*>& OutSpawnedLoot, APlayerController* OwningPlayer) const
40{
41 bool bDidDropAnything = false;
42
43 AActor* Owner = GetOwner();
44 if (!Owner)
45 {
46 return false;
47 }
48
49 if (!Owner->HasAuthority())
50 {
51 UE_LOG(LogDTLoot, Warning, TEXT("UDTLootSourceComponent::Internal_DropLoot - called on client. Loot must be dropped on server."));
52 return false;
53 }
54
55 if (!HasValidLootConfiguration())
56 {
57 UE_LOG(LogDTLoot, Verbose, TEXT("UDTLootSourceComponent::Internal_DropLoot - Owner '%s' has no loot entries configured"), *Owner->GetName());
58 return false;
59 }
60
61 UWorld* World = GetWorld();
62 UDTLootSubsystem* LootSubsystem = World ? World->GetSubsystem<UDTLootSubsystem>() : nullptr;
63 UDTCurrencySubsystem* CurrencyDropSub = World ? World->GetSubsystem<UDTCurrencySubsystem>() : nullptr;
64
65 // Seed the per-entry drop gate from the run's dungeon seed + this source's (coarse) location, so
66 // "which entries fire" is reproducible from the seed instead of frame-dependent global RNG.
67 // Reproducibility is therefore as deterministic as the dungeon's enemy/chest placement.
68 int32 DungeonSeed = 0;
69 if (const ADTGameState* GS = World ? World->GetGameState<ADTGameState>() : nullptr)
70 {
71 DungeonSeed = GS->GetCurrentDungeonSeed();
72 }
73 const FVector SourceLoc = Owner->GetActorLocation();
74 const int32 SourceHash = GetTypeHash(FIntVector(
75 FMath::RoundToInt(SourceLoc.X / 100.0f),
76 FMath::RoundToInt(SourceLoc.Y / 100.0f),
77 FMath::RoundToInt(SourceLoc.Z / 100.0f)));
78
79 FRandomStream DropStream;
80 const int32 DropSeed = UDTMathLibrary::GenerateDerivedSeed(DungeonSeed, SourceHash);
81 if (DropSeed != 0)
82 {
83 DropStream.Initialize(DropSeed);
84 }
85 else
86 {
87 DropStream.GenerateNewSeed();
88 }
89
90 for (const FLootSourceEntry& Entry : LootEntries)
91 {
92 // Roll for this specific entry (seeded so it's reproducible from the dungeon seed)
93 if (DropStream.FRand() > Entry.DropChance)
94 {
95 continue;
96 }
97
98 switch (Entry.EntryType)
99 {
100 case ELootSourceEntryType::LootTable:
101 {
102 if (Entry.LootTables.Num() > 0 && LootSubsystem)
103 {
104 FLootContext Context = UDTBlueprintLibrary::CreateLootContext(Entry.LootTables, Owner, SpawnTransform, Entry.RollCount.Min, Entry.RollCount.Max, bSpawnForAllPlayers, OwningPlayer);
105 Context.RequestorTags = RequestorTags;
106
107 TArray<ADTItemActor*> DroppedItems = LootSubsystem->SpawnLootFromContext(Context);
108 OutSpawnedLoot.Append(DroppedItems);
109 if (DroppedItems.Num() > 0)
110 {
111 bDidDropAnything = true;
112 }
113 }
114
115 break;
116 }
117 case ELootSourceEntryType::Item:
118 {
119 if (Entry.ItemDefinitions.Num() > 0 && LootSubsystem)
120 {
121 if (Entry.bDropAllItems)
122 {
123 FLootContext Context = UDTBlueprintLibrary::CreateItemLootContext(Entry.ItemDefinitions, Owner, SpawnTransform, 1, 1, bSpawnForAllPlayers, OwningPlayer);
124 Context.MinItemStackCount = FMath::Max(1, Entry.StackCount.Min);
125 Context.MaxItemStackCount = FMath::Max(Context.MinItemStackCount, Entry.StackCount.Max);
126 Context.RequestorTags = RequestorTags;
127 Context.bGuaranteeAllItems = true;
128
129 TArray<ADTItemActor*> DroppedItems = LootSubsystem->SpawnLootFromContext(Context);
130 OutSpawnedLoot.Append(DroppedItems);
131 if (DroppedItems.Num() > 0)
132 {
133 bDidDropAnything = true;
134 }
135 }
136 else
137 {
138 FLootContext Context = UDTBlueprintLibrary::CreateItemLootContext(Entry.ItemDefinitions, Owner, SpawnTransform, Entry.RollCount.Min, Entry.RollCount.Max, bSpawnForAllPlayers, OwningPlayer);
139 Context.MinItemStackCount = FMath::Max(1, Entry.StackCount.Min);
140 Context.MaxItemStackCount = FMath::Max(Context.MinItemStackCount, Entry.StackCount.Max);
141 Context.RequestorTags = RequestorTags;
142
143 TArray<ADTItemActor*> DroppedItems = LootSubsystem->SpawnLootFromContext(Context);
144 OutSpawnedLoot.Append(DroppedItems);
145 if (DroppedItems.Num() > 0)
146 {
147 bDidDropAnything = true;
148 }
149 }
150 }
151
152 break;
153 }
154 case ELootSourceEntryType::Currency:
155 {
156 if (CurrencyDropSub && Entry.CurrencyRewards.Num() > 0)
157 {
158 CurrencyDropSub->SpawnCurrencyDrops(Entry.CurrencyRewards, SpawnTransform.GetLocation(), OwningPlayer, GetCurrentDungeonFloor(), RequestorTags);
159 bDidDropAnything = true;
160 }
161
162 break;
163 }
164 }
165 }
166
167 if (OnLootDropped.IsBound() && OutSpawnedLoot.Num() > 0)
168 {
169 OnLootDropped.Broadcast(OutSpawnedLoot);
170 }
171
172 UE_LOG(LogDTLoot, Display, TEXT("UDTLootSourceComponent::Internal_DropLoot - Owner '%s' spawned %d loot item(s)"),
173 *Owner->GetName(), OutSpawnedLoot.Num());
174
175 return bDidDropAnything;
176}
