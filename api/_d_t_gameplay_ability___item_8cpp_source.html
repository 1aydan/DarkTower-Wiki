39void UDTGameplayAbility_Item::UseItem()
40{
41 UDTEquipmentInstance* Item = GetItemInstance();
42 if (!Item)
43 {
44 UE_LOG(LogDTAbility, Warning, TEXT("%s: UseItem called but no item instance found via SourceObject"), *GetName());
45 return;
46 }
47
48 AActor* Avatar = GetAvatarActorFromActorInfo();
49 if (!Avatar || !Avatar->Implements<UInventoryInterface>())
50 {
51 return;
52 }
53
54 UDTEquipmentComponent* EquipComp = Cast<UDTEquipmentComponent>(IInventoryInterface::Execute_GetEquipmentComponent(Avatar));
55 if (!EquipComp)
56 {
57 return;
58 }
59
60 EquipComp->UseItem(Item->GetEquippedSlotTag());
61
62 OnItemUsed();
63}
