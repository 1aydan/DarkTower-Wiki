88{

93 SkillTreeComponent->OnSkillTreeRespec.RemoveDynamic(this, &UDTSkillTreePanelWidget::HandleSkillTreeRespec);

113 // Build initial graph nodes

117 SkillTreeComponent->GetTreeInstance(TreeTag, TreeInst);

191int32 UDTSkillTreePanelWidget::NativePaint(const FPaintArgs& Args, const FGeometry& AllottedGeometry, const FSlateRect& MyCullingRect,
192 FSlateWindowElementList& OutDrawElements, int32 LayerId, const FWidgetStyle& InWidgetStyle, bool bParentEnabled) const
193{
194 if (TreeDefinition)
195 {
196 for (const FSkillTreeNodeLink& Link : TreeDefinition->NodeLinks)
197 {
198 const UDTSkillTreeNodeWidget* FromWidget = FindNodeWidget(Link.FromNodeTag);
199 const UDTSkillTreeNodeWidget* ToWidget = FindNodeWidget(Link.ToNodeTag);
200 if (!FromWidget || !ToWidget)
201 {
202 continue;
203 }
204
205 const FGeometry& FromGeometry = FromWidget->GetCachedGeometry();
206 const FGeometry& ToGeometry = ToWidget->GetCachedGeometry();
207 const FVector2D FromTopLeft = AllottedGeometry.AbsoluteToLocal(FromGeometry.LocalToAbsolute(FVector2D::ZeroVector));
208 const FVector2D ToTopLeft = AllottedGeometry.AbsoluteToLocal(ToGeometry.LocalToAbsolute(FVector2D::ZeroVector));
209 const FVector2D FromSize = FromGeometry.GetLocalSize();
210 const FVector2D ToSize = ToGeometry.GetLocalSize();
211 const FVector2D Start = FromTopLeft + FVector2D(FromSize.X * 0.5f, FromSize.Y);
212 const FVector2D End = ToTopLeft + FVector2D(ToSize.X * 0.5f, 0.0f);
213
214 TArray<FVector2D> Points;
215 if (TreeDefinition->bUseTreeStyleLinkRouting)
216 {
217 const float MidY = Start.Y + ((End.Y - Start.Y) * 0.5f);
218 Points.Add(Start);
219 Points.Add(FVector2D(Start.X, MidY));
220 Points.Add(FVector2D(End.X, MidY));
221 Points.Add(End);
222 }
223 else
224 {
225 Points.Add(FromTopLeft + (FromSize * 0.5f));
226 Points.Add(ToTopLeft + (ToSize * 0.5f));
227 }
228
229 FSlateDrawElement::MakeLines(
230 OutDrawElements,
231 LayerId,
232 AllottedGeometry.ToPaintGeometry(),
233 Points,
234 ESlateDrawEffect::None,
235 GetLinkDisplayColor(Link),
236 true,
237 2.0f);
238 }
239 }
240
241 return Super::NativePaint(Args, AllottedGeometry, MyCullingRect, OutDrawElements, LayerId + 1, InWidgetStyle, bParentEnabled);

289void UDTSkillTreePanelWidget::BuildGraphNodes(const TArray<FSkillTreeNode>& Nodes, const TArray<FSkillTreeNodeRankEntry>& NodeRanks)
290{
291 auto FindRank = [&NodeRanks](const FGameplayTag& Tag) -> int32
292 {
293 for (const FSkillTreeNodeRankEntry& E : NodeRanks) { if (E.NodeTag.MatchesTagExact(Tag)) { return E.Rank; } }
294 return 0;
295 };
296
297 UCanvasPanel* GraphContainer = GetGraphContainer();
298 if (!GraphContainer || !NodeWidgetClass)
299 {
300 return;
301 }
302
303 for (int32 NodeIndex = 0; NodeIndex < Nodes.Num(); ++NodeIndex)
304 {
305 const FSkillTreeNode& NodeDef = Nodes[NodeIndex];
306
307 UDTSkillTreeNodeWidget* NodeWidget = nullptr;
308 if (IsDesignTime() || !GetOwningPlayer())
309 {
310 NodeWidget = CreateWidget<UDTSkillTreeNodeWidget>(this, NodeWidgetClass);
311 }
312 else
313 {
314 NodeWidget = CreateWidget<UDTSkillTreeNodeWidget>(GetOwningPlayer(), NodeWidgetClass);
315 }
316
317 if (!NodeWidget)
318 {
319 continue;
320 }
321
322 const int32 CurrentRank = FindRank(NodeDef.GetNodeTag());
323 NodeWidget->SetNodeData(TreeTag, NodeDef, CurrentRank, IsNodeUnlockedForDisplay(NodeDef, NodeRanks));
324
325 UCanvasPanelSlot* CanvasSlot = GraphContainer->AddChildToCanvas(NodeWidget);
326 if (CanvasSlot)
327 {
328 CanvasSlot->SetAutoSize(true);
329 CanvasSlot->SetPosition(GetNodeCanvasPosition(NodeDef, NodeIndex));
330 }
331
332 NodeWidgets.Add(NodeWidget);
333 }

336bool UDTSkillTreePanelWidget::IsNodeUnlockedForDisplay(const FSkillTreeNode& NodeDef, const TArray<FSkillTreeNodeRankEntry>& NodeRanks) const
337{
338 auto FindRank = [&NodeRanks](const FGameplayTag& Tag) -> int32
339 {
340 for (const FSkillTreeNodeRankEntry& E : NodeRanks) { if (E.NodeTag.MatchesTagExact(Tag)) { return E.Rank; } }
341 return 0;
342 };
343
344 if (PointsInvested < NodeDef.GetUnlockPointsRequired())
345 {
346 return false;
347 }
348
349 for (const FGameplayTag& PrerequisiteNodeTag : NodeDef.GetPrerequisiteNodeTags())
350 {
351 if (FindRank(PrerequisiteNodeTag) <= 0)
352 {
353 return false;
354 }
355 }
356
357 if (TreeDefinition)
358 {
359 TArray<FGameplayTag> UnlockLinkSourceTags;
360 TreeDefinition->GetUnlockLinkSourceNodeTags(NodeDef.GetNodeTag(), UnlockLinkSourceTags);
361 if (UnlockLinkSourceTags.Num() > 0)
362 {
363 for (const FGameplayTag& SourceNodeTag : UnlockLinkSourceTags)
364 {
365 if (FindRank(SourceNodeTag) > 0)
366 {
367 return true;
368 }
369 }
370
371 return false;
372 }
373 }
374
375 return true;
376}

402FLinearColor UDTSkillTreePanelWidget::GetLinkDisplayColor(const FSkillTreeNodeLink& Link) const
403{
404 auto FindCachedRank = [this](const FGameplayTag& Tag) -> int32
405 {
406 for (const FSkillTreeNodeRankEntry& E : CachedNodeRanks) { if (E.NodeTag.MatchesTagExact(Tag)) { return E.Rank; } }
407 return 0;
408 };
409
410 const FSkillTreeNode* ToNode = TreeDefinition ? TreeDefinition->FindNode(Link.ToNodeTag) : nullptr;
411 const bool bTargetUnlocked = ToNode && IsNodeUnlockedForDisplay(*ToNode, CachedNodeRanks);
412 const bool bSourceInvested = FindCachedRank(Link.FromNodeTag) > 0;
413 const bool bTargetInvested = FindCachedRank(Link.ToNodeTag) > 0;
414
415 if (bSourceInvested && bTargetInvested)
416 {
417 return Link.bIsUnlockLink ? FLinearColor(0.95f, 0.78f, 0.28f, 1.0f) : FLinearColor(0.84f, 0.72f, 0.34f, 1.0f);
418 }
419
420 if (bTargetUnlocked)
421 {
422 return Link.bIsUnlockLink ? FLinearColor(0.32f, 0.78f, 0.48f, 0.95f) : FLinearColor(0.42f, 0.60f, 0.82f, 0.95f);
423 }
424
425 return Link.bIsUnlockLink ? FLinearColor(0.18f, 0.28f, 0.22f, 0.75f) : FLinearColor(0.24f, 0.28f, 0.36f, 0.75f);
426}
