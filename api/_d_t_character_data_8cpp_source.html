30EDTHitReactTier UDTCharacterData::EvaluateHitReactTier(const float ReactMagnitude) const
31{
32 EDTHitReactTier Tier = EDTHitReactTier::None;
33 if (ReactMagnitude >= HitReactConfig.HeavyMagnitudeThreshold)
34 {
35 Tier = EDTHitReactTier::Heavy;
36 }
37 else if (ReactMagnitude >= HitReactConfig.MediumMagnitudeThreshold)
38 {
39 Tier = EDTHitReactTier::Medium;
40 }
41 else if (ReactMagnitude >= HitReactConfig.LightMagnitudeThreshold)
42 {
43 Tier = EDTHitReactTier::Light;
44 }
45
46 Tier = static_cast<EDTHitReactTier>(FMath::Max(static_cast<uint8>(Tier), static_cast<uint8>(HitReactConfig.MinReactTier)));
47 Tier = static_cast<EDTHitReactTier>(FMath::Min(static_cast<uint8>(Tier), static_cast<uint8>(HitReactConfig.MaxReactTier)));
48 return Tier;
49}
