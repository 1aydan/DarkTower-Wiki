114float ADTAICharacter::PreSignificanceUpdate(USignificanceManager::FManagedObjectInfo* ObjectInfo, const FTransform& Viewpoint) const
115{
116 TRACE_CPUPROFILER_EVENT_SCOPE(USignificanceManager::PreSignificanceUpdate)
117
118 AActor* MyActor = Cast<AActor>(ObjectInfo->GetObject());
119
120 if (!IsValid(MyActor))
121 {
122 return -UE_MAX_FLT;
123 }
124
125 float DistanceSquared = (MyActor->GetActorLocation() - Viewpoint.GetLocation()).SizeSquared();
126
127 // Boost priority for recently rendered actors (visible on screen)
128 if (MyActor->WasRecentlyRendered())
129 {
130 DistanceSquared *= 0.5f;
131 }
132
133 return -DistanceSquared;
134}

136void ADTAICharacter::PostSignificanceUpdate(USignificanceManager::FManagedObjectInfo* ObjectInfo, float OldSignificance, float Significance, bool bFinal)
137{
138 TRACE_CPUPROFILER_EVENT_SCOPE(ADTAICharacter::PostSignificanceUpdate)
139
140 const UDTGameSettings* GameSettings = GetDefault<UDTGameSettings>();
141 ESignificanceLevel NewSignificanceLevel = GameSettings->GetSignificanceThreshold(Significance);
142
143 if (MinimumSignificanceOverride != ESignificanceLevel::Invalid)
144 {
145 NewSignificanceLevel = (NewSignificanceLevel < MinimumSignificanceOverride) ? MinimumSignificanceOverride : NewSignificanceLevel;
146 }
147
148 if (NewSignificanceLevel != CurrentSignificanceLevel)
149 {
150 CurrentSignificanceLevel = NewSignificanceLevel;
151 }
152
153 const float BudgetSignificance = static_cast<float>(CurrentSignificanceLevel) / static_cast<float>(ESignificanceLevel::High);
154
155 const bool bIsSignificant = (CurrentSignificanceLevel != ESignificanceLevel::Off);
156
157 USkeletalMeshComponentBudgeted* BudgetMesh = Cast<USkeletalMeshComponentBudgeted>(GetMesh());
158 if (BudgetMesh)
159 {
160 BudgetMesh->SetComponentSignificance(BudgetSignificance);
161 BudgetMesh->bPauseAnims = !bIsSignificant;
162 }
163
164 SetActorHiddenInGame(CurrentSignificanceLevel == ESignificanceLevel::Off);
165
166 if (!bTickEnabledOverride.IsSet())
167 {
168 Super::SetActorTickEnabled(PrimaryActorTick.bStartWithTickEnabled && CurrentSignificanceLevel != ESignificanceLevel::Off);
169 }
170
171#if !UE_BUILD_SHIPPING
172 if (bDebugDrawSignificance)
173 {
174 FColor DebugColor;
175 FString LevelName;
176 switch (CurrentSignificanceLevel)
177 {
178 case ESignificanceLevel::Off:
179 DebugColor = FColor::Red;
180 LevelName = TEXT("OFF");
181 break;
182 case ESignificanceLevel::Low:
183 DebugColor = FColor::Orange;
184 LevelName = TEXT("LOW");
185 break;
186 case ESignificanceLevel::Medium:
187 DebugColor = FColor::Yellow;
188 LevelName = TEXT("MED");
189 break;
190 case ESignificanceLevel::High:
191 DebugColor = FColor::Green;
192 LevelName = TEXT("HIGH");
193 break;
194 default:
195 DebugColor = FColor::White;
196 LevelName = TEXT("???");
197 break;
198 }
199
200 DrawDebugString(GetWorld(),GetActorLocation() + FVector(0, 0, 100),FString::Printf(TEXT("Sig: %s (%.0f)"), *LevelName, Significance),nullptr,DebugColor,0.0f,true);
201 }
202#endif
203}
