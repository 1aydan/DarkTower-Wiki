10void FDTAbilitySystemUtils::TryGrantAbility(UAbilitySystemComponent* InASC, const FDTGameFeatureAbilityMapping& InAbilityMapping, FGameplayAbilitySpecHandle& OutAbilityHandle, FGameplayAbilitySpec& OutAbilitySpec)
11{
12 check(InASC);
13
14 if (InAbilityMapping.AbilityType.IsNull())
15 {
16 UE_LOG(LogDTAbility, Error, TEXT("FDTAbilitySystemUtils::TryGrantAbility - Failed to grant ability \"%s\" because SoftClassPtr is null"), *InAbilityMapping.AbilityType.ToString());
17 return;
18 }
19
20 const TSubclassOf<UGameplayAbility> AbilityType = InAbilityMapping.AbilityType.LoadSynchronous();
21 check(AbilityType);
22
23 OutAbilitySpec = InASC->BuildAbilitySpecFromClass(AbilityType, InAbilityMapping.Level);
24 if (InAbilityMapping.InputTag.IsValid())
25 {
26 OutAbilitySpec.GetDynamicSpecSourceTags().AddTag(InAbilityMapping.InputTag);
27 }
28
29 if (InASC->IsOwnerActorAuthoritative())
30 {
31 // Only grant abilities on authority, and only if not granted yet
32 if (!IsAbilityGranted(InASC, AbilityType, InAbilityMapping.Level))
33 {
34 OutAbilityHandle = InASC->GiveAbility(OutAbilitySpec);
35 }
36 else
37 {
38 // In case granting is prevented because of ability already existing, return the existing handle
39 if (const FGameplayAbilitySpec* ExistingAbilitySpec = InASC->FindAbilitySpecFromClass(AbilityType))
40 {
41 OutAbilityHandle = ExistingAbilitySpec->Handle;
42 }
43 }
44 }
45 else
46 {
47 // For clients, try to resolve the replicated spec handle
48 if (const FGameplayAbilitySpec* ExistingAbilitySpec = InASC->FindAbilitySpecFromClass(AbilityType))
49 {
50 OutAbilityHandle = ExistingAbilitySpec->Handle;
51 }
52 }
53}

55void FDTAbilitySystemUtils::TryGrantAttributes(UAbilitySystemComponent* InASC, const FDTGameFeatureAttributeSetMapping& InAttributeSetMapping, UAttributeSet*& OutAttributeSet)
56{
57 check(InASC);
58
59 AActor* OwnerActor = InASC->GetOwnerActor();
60 if (!IsValid(OwnerActor))
61 {
62 UE_LOG(LogDTAbility, Error, TEXT("FDTAbilitySystemUtils::TryGrantAttributes - Ability System Component owner actor is not valid"));
63 return;
64 }
65
66 const TSubclassOf<UAttributeSet> AttributeSetType = InAttributeSetMapping.AttributeSet.LoadSynchronous();
67 if (!AttributeSetType)
68 {
69 UE_LOG(LogDTAbility, Error, TEXT("FDTAbilitySystemUtils::TryGrantAttributes - AttributeSet class is invalid"));
70 return;
71 }
72
73 // Prevent adding the same attribute set multiple times
74 if (UAttributeSet* AttributeSet = GetAttributeSet(InASC, AttributeSetType))
75 {
76 OutAttributeSet = AttributeSet;
77 return;
78 }
79
80 OutAttributeSet = NewObject<UAttributeSet>(OwnerActor, AttributeSetType);
81 if (!InAttributeSetMapping.InitializationData.IsNull())
82 {
83 if (const UDataTable* InitData = InAttributeSetMapping.InitializationData.LoadSynchronous())
84 {
85 OutAttributeSet->InitFromMetaDataTable(InitData);
86 }
87 }
88
89 InASC->AddAttributeSetSubobject(OutAttributeSet);
90}

92void FDTAbilitySystemUtils::TryGrantGameplayEffect(UAbilitySystemComponent* InASC, const TSubclassOf<UGameplayEffect> InEffectType, const float InLevel, TArray<FActiveGameplayEffectHandle>& OutEffectHandles)
93{
94 check(InASC);
95
96 if (!InASC->IsOwnerActorAuthoritative())
97 {
98 return;
99 }
100
101 if (!InEffectType)
102 {
103 UE_LOG(LogDTAbility, Warning, TEXT("FDTAbilitySystemUtils::TryGrantGameplayEffect - Trying to apply an effect from an invalid class"));
104 return;
105 }
106
107 // Don't apply if ASC already has it
108 TArray<FActiveGameplayEffectHandle> ActiveEffects;
109 if (HasGameplayEffectApplied(InASC, InEffectType, ActiveEffects))
110 {
111 OutEffectHandles = MoveTemp(ActiveEffects);
112 return;
113 }
114
115 const FGameplayEffectContextHandle EffectContext = InASC->MakeEffectContext();
116 const FGameplayEffectSpecHandle NewHandle = InASC->MakeOutgoingSpec(InEffectType, InLevel, EffectContext);
117 if (NewHandle.IsValid())
118 {
119 FActiveGameplayEffectHandle EffectHandle = InASC->ApplyGameplayEffectSpecToSelf(*NewHandle.Data.Get());
120 OutEffectHandles.Add(MoveTemp(EffectHandle));
121 }
122}

124bool FDTAbilitySystemUtils::TryGrantAbilitySet(UAbilitySystemComponent* InASC, const UDTAbilitySet* InAbilitySet, FDTAbilitySetHandle& OutAbilitySetHandle)
125{
126 check(InASC);
127
128 if (!InAbilitySet)
129 {
130 return false;
131 }
132
133 // Add Abilities
134 int32 AbilitiesIndex = 0;
135 for (const FDTGameFeatureAbilityMapping& AbilityMapping : InAbilitySet->GrantedAbilities)
136 {
137 AbilitiesIndex++;
138
139 if (AbilityMapping.AbilityType.IsNull())
140 {
141 UE_LOG(LogDTAbility, Error, TEXT("FDTAbilitySystemUtils::TryGrantAbilitySet - GrantedAbilities AbilityType on ability set %s is not valid at Index %d"), *GetNameSafe(InAbilitySet), AbilitiesIndex - 1);
142 continue;
143 }
144
145 FGameplayAbilitySpec AbilitySpec;
146 FGameplayAbilitySpecHandle AbilityHandle;
147 TryGrantAbility(InASC, AbilityMapping, AbilityHandle, AbilitySpec);
148 OutAbilitySetHandle.Abilities.Add(AbilityHandle);
149 }
150
151 // Add Attributes
152 int32 AttributesIndex = 0;
153 for (const FDTGameFeatureAttributeSetMapping& Attributes : InAbilitySet->GrantedAttributes)
154 {
155 AttributesIndex++;
156
157 if (Attributes.AttributeSet.IsNull())
158 {
159 UE_LOG(LogDTAbility, Error, TEXT("FDTAbilitySystemUtils::TryGrantAbilitySet - GrantedAttributes AttributeSet on ability set %s is not valid at Index %d"), *GetNameSafe(InAbilitySet), AttributesIndex - 1);
160 continue;
161 }
162
163 UAttributeSet* AddedAttributeSet = nullptr;
164 TryGrantAttributes(InASC, Attributes, AddedAttributeSet);
165
166 if (AddedAttributeSet)
167 {
168 OutAbilitySetHandle.Attributes.Add(AddedAttributeSet);
169 }
170 }
171
172 // Add Effects
173 int32 EffectsIndex = 0;
174 for (const FDTGameFeatureGameplayEffectMapping& Effect : InAbilitySet->GrantedEffects)
175 {
176 EffectsIndex++;
177
178 if (Effect.EffectType.IsNull())
179 {
180 UE_LOG(LogDTAbility, Error, TEXT("FDTAbilitySystemUtils::TryGrantAbilitySet - GrantedEffects EffectType on ability set %s is not valid at Index %d"), *GetNameSafe(InAbilitySet), EffectsIndex - 1);
181 continue;
182 }
183
184 TryGrantGameplayEffect(InASC, Effect.EffectType.LoadSynchronous(), Effect.Level, OutAbilitySetHandle.EffectHandles);
185 }
186
187 // Add Owned Gameplay Tags
188 if (InAbilitySet->OwnedTags.IsValid())
189 {
190 AddLooseGameplayTagsUnique(InASC, InAbilitySet->OwnedTags);
191
192 // Store a copy of the tags, so that they can be removed later on from handle
193 OutAbilitySetHandle.OwnedTags = InAbilitySet->OwnedTags;
194 }
195
196 OutAbilitySetHandle.AbilitySetPathName = InAbilitySet->GetPathName();
197 return true;
198}

225bool FDTAbilitySystemUtils::IsAbilityGranted(const UAbilitySystemComponent* InASC, TSubclassOf<UGameplayAbility> InAbility, const int32 InLevel)
226{
227 check(InASC);
228
229 TArray<FGameplayAbilitySpec> AbilitySpecs = InASC->GetActivatableAbilities();
230 for (const FGameplayAbilitySpec& ActivatableAbility : AbilitySpecs)
231 {
232 if (!ActivatableAbility.Ability)
233 {
234 continue;
235 }
236
237 if (ActivatableAbility.Ability->GetClass() == InAbility && ActivatableAbility.Level == InLevel)
238 {
239 return true;
240 }
241 }
242
243 return false;

246void FDTAbilitySystemUtils::AddLooseGameplayTagsUnique(UAbilitySystemComponent* InASC, const FGameplayTagContainer& InTags)
247{
248 check(InASC);
249
250 FGameplayTagContainer ExistingTags;
251 InASC->GetOwnedGameplayTags(ExistingTags);
252
253 TArray<FGameplayTag> TagsToCheck;
254 InTags.GetGameplayTagArray(TagsToCheck);
255
256 FGameplayTagContainer TagsToAdd;
257 for (const FGameplayTag& Tag : TagsToCheck)
258 {
259 if (!ExistingTags.HasTagExact(Tag))

261 TagsToAdd.AddTag(Tag);
