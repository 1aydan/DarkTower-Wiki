105void UDTAbilityTask_UpdateChain::TickTask(float DeltaTime)
106{
107 Super::TickTask(DeltaTime);
108
109 if (bFinished)
110 {
111 return;
112 }
113
114 for (int32 ChainIndex = 0; ChainIndex < ChainStates.Num(); ++ChainIndex)
115 {
116 if (!ChainStates.IsValidIndex(ChainIndex))
117 {
118 return;
119 }
120
121 FChainState& Runtime = ChainStates[ChainIndex];
122
123 switch (Runtime.Phase)
124 {
125 case EDTChainPhase::Extending:
126 Runtime.PhaseElapsed += DeltaTime;
127 UpdateChainVFXExtend(ChainIndex);
128 FireHits(ChainIndex);
129 if (Runtime.PhaseElapsed >= Runtime.ExtendDuration)
130 {
131 EnterHoldPhase(ChainIndex);
132 }
133 break;
134
135 case EDTChainPhase::Holding:
136 UpdateChainVFXFollow(ChainIndex);
137 if (bWaitForRetractNotify)
138 {
139 if (bRetractNotifyReceived)
140 {
141 EnterRetractPhase(ChainIndex);
142 }
143 }
144 else
145 {
146 Runtime.PhaseElapsed += DeltaTime;
147 if (Runtime.PhaseElapsed >= ChainVFXHoldDuration)
148 {
149 EnterRetractPhase(ChainIndex);
150 }
151 }
152 break;
153
154 case EDTChainPhase::Retracting:
155 Runtime.PhaseElapsed += DeltaTime;
156 UpdateChainVFXRetract(ChainIndex);
157 FirePulls(ChainIndex);
158 if (Runtime.PhaseElapsed >= ChainRetractDuration)
159 {
160 EnterDonePhase(ChainIndex);
161 }
162 break;
163
164 case EDTChainPhase::Done:
165 break;
166 }
167
168 // A hit/pull broadcast can end the owning ability (and this task) mid-loop.
169 if (bFinished)
170 {
171 return;
172 }
173 }
174
175 TryFinishTask();
176}

374void UDTAbilityTask_UpdateChain::FireHits(int32 ChainIndex)
375{
376 if (!ChainTraces.IsValidIndex(ChainIndex) || !ChainStates.IsValidIndex(ChainIndex))
377 {
378 return;
379 }
380
381 FChainState& Runtime = ChainStates[ChainIndex];
382 const FChainTrace& Chain = ChainTraces[ChainIndex];
383
384 for (int32 TargetIndex = 0; TargetIndex < Chain.Targets.Num(); ++TargetIndex)
385 {
386 if (Runtime.TargetHitStates[TargetIndex] != 0)
387 {
388 continue;
389 }
390
391 const float HitDelay = (ChainTravelSpeed > UE_SMALL_NUMBER) ? Chain.Targets[TargetIndex].DistanceAlongChain / ChainTravelSpeed : 0.0f;
392 if (Runtime.PhaseElapsed + UE_SMALL_NUMBER < HitDelay)
393 {
394 continue;
395 }
396
397 Runtime.TargetHitStates[TargetIndex] = 1;

416 {
417 continue;
418 }
419
420 const float PullDelay = (ChainLength > UE_SMALL_NUMBER)
421 ? ChainRetractDuration * (1.0f - FMath::Clamp(Chain.Targets[TargetIndex].DistanceAlongChain / ChainLength, 0.0f, 1.0f))
422 : 0.0f;
423 if (Runtime.PhaseElapsed + UE_SMALL_NUMBER < PullDelay)
424 {
