25void UDTPlayerProgressionComponent::AddExperience(const int64 Amount)
26{
27 if (!GetOwner() || !GetOwner()->HasAuthority() || Amount <= 0)
28 {
29 return;
30 }
31
32 const int32 OldLevel = FMath::Max(1, CachedHeroLevel);
33 const int64 OldTotalExperience = CachedTotalHeroExperience;
34 const UDTProgressionData* ProgressionData = GetProgressionData();
35 const int32 MaxHeroLevel = ProgressionData ? FMath::Max(1, ProgressionData->MaxHeroLevel) : 100;
36
37 int32 NewLevel = OldLevel;
38 int64 NewExperience = CachedHeroExperience + Amount;
39 int64 NewTotalExperience = CachedTotalHeroExperience + Amount;
40
41 while (NewLevel < MaxHeroLevel)
42 {
43 const int64 RequiredXP = GetExperienceToNextLevelForLevel(NewLevel);
44 if (RequiredXP <= 0 || NewExperience < RequiredXP)
45 {
46 break;
47 }
48
49 NewExperience -= RequiredXP;
50 ++NewLevel;
51 }
52
53 if (NewLevel >= MaxHeroLevel)
54 {
55 const int64 RequiredXPAtCap = GetExperienceToNextLevelForLevel(NewLevel);
56 NewExperience = FMath::Clamp<int64>(NewExperience, 0, FMath::Max<int64>(0, RequiredXPAtCap - 1));
57 }
58
59 CachedHeroLevel = NewLevel;
60 CachedHeroExperience = FMath::Max<int64>(0, NewExperience);
61 CachedTotalHeroExperience = FMath::Max<int64>(0, NewTotalExperience);
62
63 ApplyRuntimeCacheToStats();
64
65 if (NewLevel != OldLevel)
66 {
67 if (UDTSkillTreeComponent* SkillTreeComponent = GetSkillTreeComponent())
68 {
69 SkillTreeComponent->HandleLevelUp(NewLevel, OldLevel);
70 }
71
72 OnHeroLevelChanged.Broadcast(NewLevel, OldLevel);
73 }
74
75 OnHeroExperienceChanged.Broadcast(CachedHeroExperience, GetExperienceToNextLevelForLevel(NewLevel), CachedTotalHeroExperience, CachedTotalHeroExperience - OldTotalExperience);
76}

180int32 UDTPlayerProgressionComponent::RecordCompletedFloorForRun(const FDTDungeonSettings& DungeonSetting)
181{
182 if (!GetOwner() || !GetOwner()->HasAuthority())
183 {
184 return 0;
185 }
186
187 const int32 CompletedFloor = FMath::Max(0, DungeonSetting.GlobalFloor);
188 if (CompletedFloor <= 0)
189 {
190 return 0;
191 }
192
193 FDTModeProgressionSaveEntry* Entry = FindOrAddModeProgressEntry(DungeonSetting.ModeDefinitionTag);
194 check(Entry);
195
196 // EX : your on floor 100 and your friend is on floor 50. On Floor 100 completion you save floor 100 and your friend saves floor 51
197 const int32 ExistingFloor = FMath::Max(0, Entry->HighestCompletedFloor);
198 if (ExistingFloor < CompletedFloor)
199 {
200 Entry->HighestCompletedFloor = ExistingFloor + 1;
201 }
202 else
203 {
204 Entry->HighestCompletedFloor = FMath::Max(ExistingFloor, CompletedFloor);
205 }
206
207 return Entry->HighestCompletedFloor;
208}
