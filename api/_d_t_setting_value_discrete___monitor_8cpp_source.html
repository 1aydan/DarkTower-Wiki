53void UDTSettingValueDiscrete_Monitor::RebuildMonitorOptions()
54{
55 MonitorOptions.Empty();
56
57 FDisplayMetrics DisplayMetrics;
58 FSlateApplication::Get().GetCachedDisplayMetrics(DisplayMetrics);
59
60 for (int32 MonitorIndex = 0; MonitorIndex < DisplayMetrics.MonitorInfo.Num(); ++MonitorIndex)
61 {
62 const FMonitorInfo& Info = DisplayMetrics.MonitorInfo[MonitorIndex];
63
64 // Build a human-readable label: "1: <Name> (W×H)" or "1: Display (W×H)" if name is empty
65 FString MonitorName = Info.Name.IsEmpty() ? FString::Printf(TEXT("Display %d"), MonitorIndex + 1): Info.Name;
66
67 // Append "(Primary)" tag so players can easily identify their main monitor
68 const FString PrimaryTag = Info.bIsPrimary ? TEXT(" (Primary)") : TEXT("");
69
70 const FString Label = FString::Printf(TEXT("%d: %s%s (%dx%d)"), MonitorIndex + 1,*MonitorName,*PrimaryTag,Info.NativeWidth,Info.NativeHeight);
71
72 MonitorOptions.Add(FText::FromString(Label));
73 }
74
75 // Safety fallback — should never be empty on a real machine
76 if (MonitorOptions.IsEmpty())
77 {
78 MonitorOptions.Add(LOCTEXT("MonitorDefault", "Display 1 (Primary)"));
79 }
80}
