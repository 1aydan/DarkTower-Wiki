92void UDTAttributeSet::PreAttributeChange(const FGameplayAttribute& Attribute, float& NewValue)
93{
94 // This is called whenever attributes change, so for max health/mana we want to scale the current totals to match
95 Super::PreAttributeChange(Attribute, NewValue);
96
97 // Handle Max Attributes
98 if (Attribute == GetMaxHealthAttribute())
99 {
100 bIsAdjustingForMaxChange = true;
101 AdjustAttributeForMaxChange(Health, MaxHealth, NewValue, GetHealthAttribute());
102 bIsAdjustingForMaxChange = false;
103 NewValue = FMath::FloorToFloat(NewValue);
104 }
105 else if (Attribute == GetMaxManaAttribute())
106 {
107 bIsAdjustingForMaxChange = true;
108 AdjustAttributeForMaxChange(Mana, MaxMana, NewValue, GetManaAttribute());
109 bIsAdjustingForMaxChange = false;
110 NewValue = FMath::FloorToFloat(NewValue);
111 }
112 else if (Attribute == GetMaxShieldAttribute())
113 {
114 bIsAdjustingForMaxChange = true;
115 AdjustAttributeForMaxChange(Shield, MaxShield, NewValue, GetShieldAttribute());
116 bIsAdjustingForMaxChange = false;
117 NewValue = FMath::FloorToFloat(NewValue);
118 }
119 // Clamp current resources to their max so they never sit above it (e.g. seeded above max on load, or
120 // pushed over by a heal/regen). Skipped during a Max change, where AdjustAttributeForMaxChange re-scales
121 // the resource against the new (not-yet-committed) Max. Only clamps once Max is initialized (> 0) so an
122 // init that sets the resource before its Max doesn't get pinned to 0.
123 else if (Attribute == GetHealthAttribute())
124 {
125 NewValue = FMath::FloorToFloat(NewValue);
126 const float MaxValue = GetMaxHealth();
127 if (!bIsAdjustingForMaxChange && MaxValue > 0.0f)
128 {
129 NewValue = FMath::Clamp(NewValue, 0.0f, MaxValue);
130 }
131 }
132 else if (Attribute == GetManaAttribute())
133 {
134 NewValue = FMath::FloorToFloat(NewValue);
135 const float MaxValue = GetMaxMana();
136 if (!bIsAdjustingForMaxChange && MaxValue > 0.0f)
137 {
138 NewValue = FMath::Clamp(NewValue, 0.0f, MaxValue);
139 }
140 }
141 else if (Attribute == GetShieldAttribute())
142 {
143 NewValue = FMath::FloorToFloat(NewValue);
144 const float MaxValue = GetMaxShield();
145 if (!bIsAdjustingForMaxChange && MaxValue > 0.0f)
146 {
147 NewValue = FMath::Clamp(NewValue, 0.0f, MaxValue);
148 }
149 }
150 else if (Attribute == GetCritChanceAttribute())
151 {
152 NewValue = FMath::Clamp(NewValue, 0.0f, 1.0f);
153 }
154 else if (Attribute == GetCritDamageAttribute())
155 {
156 NewValue = FMath::Clamp(NewValue, 0.0f, 10.0f);
157 }
158 else if (Attribute == GetDamageResistanceAttribute() || Attribute == GetFireResistanceAttribute() || Attribute == GetColdResistanceAttribute() || Attribute == GetLightningResistanceAttribute() || Attribute == GetPoisonResistanceAttribute())
159 {
160 NewValue = FMath::Clamp(NewValue, -10.0f, 0.75f);
161 }
162 else if (Attribute == GetBlockChanceAttribute() || Attribute == GetPhysicalPenetrationAttribute() || Attribute == GetBleedChanceAttribute() || Attribute == GetHealthLeechAttribute())
163 {
164 NewValue = FMath::Clamp(NewValue, 0.0f, 1.0f);
165 }
166 else if (Attribute == GetBlockEffectivenessAttribute())
167 {
168 // Percent-points (e.g. 50 = +50%); only floored at 0 so it can scale past 100%.
169 NewValue = FMath::Max(NewValue, 0.0f);
170 }
171 else if (Attribute == GetCloseCombatDamageReductionAttribute() || Attribute == GetCritDamageReductionAttribute())
172 {
173 NewValue = FMath::Clamp(NewValue, 0.0f, 0.75f);
174 }
175 else if (Attribute == GetDamageOverTimeMultAttribute())
176 {
177 NewValue = FMath::Clamp(NewValue, 0.0f, 10.0f);
178 }
179 else if (Attribute == GetHealthRegenRateAttribute() || Attribute == GetManaRegenRateAttribute())
180 {
181 // Regen rates are per-period amounts that are legitimately fractional (e.g. 0.5/sec) and feed the
182 // periodic regen effects (GE_Regen_Health / GE_Regen_Mana) as an attribute-based magnitude. They must
183 // NOT be floored: the catch-all below would truncate any sub-1 rate to 0, silently disabling regen.
184 // NOTE: both regen GEs must capture this attribute from Target (non-snapshot) so the periodic tick
185 // re-reads the live rate each period. Capturing from Source freezes the value at apply-time (~0,
186 // before stats init) and zeroes out regen — that has regressed more than once.
187 NewValue = FMath::Max(NewValue, 0.0f);
188 }
189 else
190 {
191 // floor all other attributes since they are not meant to be fractional.
192 // Use FloorToFloat (not FloorToInt) so large scaled values (e.g. deep-floor Health/Damage)
193 // don't overflow int32 and wrap negative — that caused enemies to spawn dead at high floors.
194 NewValue = FMath::FloorToFloat(NewValue);
195 }
196}

198void UDTAttributeSet::PostAttributeChange(const FGameplayAttribute& Attribute, float OldValue, float NewValue)
199{
200 Super::PostAttributeChange(Attribute, OldValue, NewValue);
201
202 // Clamp a resource down when its Max is (re)set to a lower value. This covers the load-order case the
203 // PreAttributeChange clamp misses: the resource is set (e.g. Health=128) while Max is still 0, so the
204 // PreAttributeChange guard (MaxValue > 0) skips it, and AdjustAttributeForMaxChange also skips because the
205 // old Max was 0. When Max finally lands (127), pull the resource back into range so it never sits above Max.
206 if (Attribute == GetMaxHealthAttribute() && GetHealth() > NewValue)
207 {
208 SetHealth(NewValue);
209 }
210 else if (Attribute == GetMaxManaAttribute() && GetMana() > NewValue)
211 {
212 SetMana(NewValue);
213 }
214 else if (Attribute == GetMaxShieldAttribute() && GetShield() > NewValue)
215 {
216 SetShield(NewValue);
217 }

483void UDTAttributeSet::HandleDamageAttribute(const FDTAttributeSetExecutionData& ExecutionData)
484{
485 AActor* SourceActor = ExecutionData.SourceActor;
486 AActor* TargetActor = ExecutionData.TargetActor;
487
488 UDTAbilityCoreComponent* SourceCoreComponent = Cast<UDTAbilityCoreComponent>(ExecutionData.SourceCoreComponent);
489 UDTAbilityCoreComponent* TargetCoreComponent = Cast<UDTAbilityCoreComponent>(ExecutionData.TargetCoreComponent);
490
491 const FGameplayEffectContextHandle& GEContext = ExecutionData.Context;
492 FGameplayTagContainer DamageTags = ExecutionData.SourceTags;
493
494 DamageTags.AppendTags(UDTBlueprintLibrary::EffectContextGetContextTags(GEContext));
495
496 const FHitResult* HitResult = GEContext.GetHitResult();
497 AActor* EffectCauser = GEContext.GetEffectCauser();
498
499 if (!TargetActor || !TargetCoreComponent)
500 {
501 UE_LOG(LogDT, Warning, TEXT("UDTAttributeSet::HandleDamageAttribute - TargetActor or TargetCoreComponent is invalid, aborting damage handling"));
502 return;
503 }
504
505 // SS - AE : Attempt to resolve the source component from something like a projectile to have access to damage dealt (TODO : maybe look at another way to do this, but it works)
506 if (!SourceCoreComponent && SourceActor)
507 {
508 if (SourceActor->GetInstigator())
509 {
510 SourceCoreComponent = SourceActor->GetInstigator()->FindComponentByClass<UDTAbilityCoreComponent>();
511 }
512 }
513
514 const float LocalDamageDone = GetDamage();
515 SetDamage(0.f);
516
517 if (LocalDamageDone > 0.f)
518 {
519 const bool bAlive = TargetCoreComponent->IsAlive();
520 const bool bGodMode = TargetCoreComponent->IsGodMode();
521 const bool bDemiGodMode = TargetCoreComponent->IsDemiGodMode();
522
523 if (!bAlive)
524 {
525 UE_LOG(LogDT, VeryVerbose, TEXT("UDTAttributeSet::HandleDamageAttribute - %s character or pawn is NOT alive when receiving damage"), *TargetActor->GetName());
526 return;
527 }
528
529 if (bAlive && !bGodMode)
530 {
531 float NewHealth = GetHealth() - LocalDamageDone;
532 const float ClampMinimumValue = GetClampMinimumValueFor(GetHealthAttribute());
533 const float MaxHealthValue = GetMaxHealth();
534
535 if (bDemiGodMode && NewHealth <= ClampMinimumValue)
536 {
537 NewHealth = MaxHealthValue;
538 }
539
540 SetHealth(FMath::Clamp(NewHealth, ClampMinimumValue, MaxHealthValue));
541
542 const bool bTargetDied = !TargetCoreComponent->IsAlive();
543
544 TargetCoreComponent->HandleDamageEvent(LocalDamageDone, DamageTags, SourceActor, EffectCauser, HitResult, GEContext);
545 TargetCoreComponent->HandleHealthChange(-LocalDamageDone, DamageTags);
546
547 if (SourceCoreComponent)
548 {
549 SourceCoreComponent->HandleDamageDealt(TargetActor, LocalDamageDone, DamageTags, bTargetDied, EffectCauser, HitResult, GEContext);
550
551 if (bTargetDied)
552 {
553 SourceCoreComponent->HandleKill(TargetActor, LocalDamageDone, DamageTags, EffectCauser, HitResult, GEContext);
554 }
555 }
556 }
557 }
558}

575void UDTAttributeSet::HandleManaAttribute(const FDTAttributeSetExecutionData& ExecutionData)
576{
577 UDTAbilityCoreComponent* TargetCoreComponent = Cast<UDTAbilityCoreComponent>(ExecutionData.TargetCoreComponent);
578 const float ClampMinimumValue = GetClampMinimumValueFor(GetManaAttribute());
579
580 SetMana(FMath::Clamp(GetMana(), ClampMinimumValue, GetMaxMana()));
581
