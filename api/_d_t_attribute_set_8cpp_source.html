15void UDTAttributeSet::InitializeAttributeTagMaps()
16{
17 if (bMapsInitialized)
18 {
19 return;
20 }
21
22 for (TFieldIterator<FStructProperty> It(GetClass()); It; ++It)
23 {
24 if (It->Struct != FGameplayAttributeData::StaticStruct())

26 continue;
27 }
28
29 const FString PropName = It->GetName();
30
31 const FGameplayTag StatTag = FGameplayTag::RequestGameplayTag(FName(*FString::Printf(TEXT("Attribute.%s"), *PropName)), false);

35 if (StatTag.IsValid())

44 }
45 }

94void UDTAttributeSet::PreAttributeChange(const FGameplayAttribute& Attribute, float& NewValue)
95{
96 // This is called whenever attributes change, so for max health/mana we want to scale the current totals to match
97 Super::PreAttributeChange(Attribute, NewValue);
98
99 // Handle Max Attributes
100 if (Attribute == GetMaxHealthAttribute())
101 {
102 AdjustAttributeForMaxChange(Health, MaxHealth, NewValue, GetHealthAttribute());
103 NewValue = FMath::FloorToInt(NewValue);
104 }
105 else if (Attribute == GetMaxManaAttribute())
106 {
107 AdjustAttributeForMaxChange(Mana, MaxMana, NewValue, GetManaAttribute());
108 NewValue = FMath::FloorToInt(NewValue);
109 }
110 else if (Attribute == GetMaxShieldAttribute())
111 {
112 AdjustAttributeForMaxChange(Shield, MaxShield, NewValue, GetShieldAttribute());
113 NewValue = FMath::FloorToInt(NewValue);
114 }
115 else if (Attribute == GetCritChanceAttribute())
116 {
117 NewValue = FMath::Clamp(NewValue, 0.0f, 1.0f);
118 }
119 else if (Attribute == GetCritDamageAttribute())
120 {
121 NewValue = FMath::Clamp(NewValue, 0.0f, 10.0f);
122 }
123 else if (Attribute == GetDamageResistanceAttribute() || Attribute == GetFireResistanceAttribute() || Attribute == GetColdResistanceAttribute() || Attribute == GetLightningResistanceAttribute() || Attribute == GetPoisonResistanceAttribute())
124 {
125 NewValue = FMath::Clamp(NewValue, -10.0f, 0.75f);
126 }
127 else if (Attribute == GetBlockChanceAttribute() || Attribute == GetPhysicalPenetrationAttribute() || Attribute == GetBleedChanceAttribute() || Attribute == GetHealthLeechAttribute())
128 {
129 NewValue = FMath::Clamp(NewValue, 0.0f, 1.0f);
130 }
131 else if (Attribute == GetBlockEffectivenessAttribute())
132 {
133 // Percent-points (e.g. 50 = +50%); only floored at 0 so it can scale past 100%.
134 NewValue = FMath::Max(NewValue, 0.0f);
135 }
136 else if (Attribute == GetCloseCombatDamageReductionAttribute() || Attribute == GetCritDamageReductionAttribute())
137 {
138 NewValue = FMath::Clamp(NewValue, 0.0f, 0.75f);
139 }
140 else if (Attribute == GetDamageOverTimeMultAttribute())
141 {
142 NewValue = FMath::Clamp(NewValue, 0.0f, 10.0f);
143 }
144 else
145 {
146 // floor all other attributes to the nearest integer since they are not meant to be fractional
147 NewValue = FMath::FloorToInt(NewValue);
148 }
149}

419void UDTAttributeSet::HandleDamageAttribute(const FGSCAttributeSetExecutionData& ExecutionData)
420{
421 AActor* SourceActor = ExecutionData.SourceActor;
422 AActor* TargetActor = ExecutionData.TargetActor;
423
424 UDTAbilityCoreComponent* SourceCoreComponent = Cast<UDTAbilityCoreComponent>(ExecutionData.SourceCoreComponent);
425 UDTAbilityCoreComponent* TargetCoreComponent = Cast<UDTAbilityCoreComponent>(ExecutionData.TargetCoreComponent);
426
427 const FGameplayEffectContextHandle& GEContext = ExecutionData.Context;
428 FGameplayTagContainer DamageTags = ExecutionData.SourceTags;
429
430 DamageTags.AppendTags(UDTBlueprintLibrary::EffectContextGetContextTags(GEContext));
431
432 const FHitResult* HitResult = GEContext.GetHitResult();
433 AActor* EffectCauser = GEContext.GetEffectCauser();
434
435 if (!TargetActor || !TargetCoreComponent)
436 {
437 UE_LOG(LogDT, Warning, TEXT("UDTAttributeSet::HandleDamageAttribute - TargetActor or TargetCoreComponent is invalid, aborting damage handling"));
438 return;
439 }
440
441 // SS - AE : Attempt to resolve the source component from something like a projectile to have access to damage dealt (TODO : maybe look at another way to do this, but it works)
442 if (!SourceCoreComponent && SourceActor)
443 {
444 if (SourceActor->GetInstigator())
445 {
446 SourceCoreComponent = Cast<UDTAbilityCoreComponent>(UGSCBlueprintFunctionLibrary::GetCompanionCoreComponent(SourceActor->GetInstigator()));
447 }
448 }
449
450 const float LocalDamageDone = GetDamage();
451 SetDamage(0.f);
452
453 if (LocalDamageDone > 0.f)
454 {
455 const bool bAlive = TargetCoreComponent->IsAlive();
456 const bool bGodMode = TargetCoreComponent->IsGodMode();
457
458 if (!bAlive)
459 {
460 UE_LOG(LogDT, VeryVerbose, TEXT("UDTAttributeSet::HandleDamageAttribute - %s character or pawn is NOT alive when receiving damage"), *TargetActor->GetName());
461 return;
462 }
463
464 if (bAlive && !bGodMode)
465 {
466 const float NewHealth = GetHealth() - LocalDamageDone;
467 const float ClampMinimumValue = GetClampMinimumValueFor(GetHealthAttribute());
468 SetHealth(FMath::Clamp(NewHealth, ClampMinimumValue, GetMaxHealth()));
469
470 const bool bTargetDied = !TargetCoreComponent->IsAlive();
471
472 TargetCoreComponent->HandleDamageEvent(LocalDamageDone, DamageTags, SourceActor, EffectCauser, HitResult, GEContext);
473 TargetCoreComponent->HandleHealthChange(-LocalDamageDone, DamageTags);
474
475 if (SourceCoreComponent)
476 {
477 SourceCoreComponent->HandleDamageDealt(TargetActor, LocalDamageDone, DamageTags, bTargetDied, EffectCauser, HitResult, GEContext);
478
479 if (bTargetDied)
480 {
481 SourceCoreComponent->HandleKill(TargetActor, LocalDamageDone, DamageTags, EffectCauser, HitResult, GEContext);
482 }
483 }
484 }
485 }
