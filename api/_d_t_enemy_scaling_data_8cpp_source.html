30int32 UDTEnemyScalingData::CalculateScaledLevel(int32 Floor) const
31{
32 if (Floor <= GlobalEnemyScalingThreshold)
33 {
34 // Before threshold: EnemyLevel = Floor
35 return Floor;
36 }
37
38 // After threshold: EnemyLevel = Threshold + (Floor - Threshold) ^ DR
39 const float ExcessFloors = static_cast<float>(Floor - GlobalEnemyScalingThreshold);
40 const float ScaledExcess = FMath::Pow(ExcessFloors, GlobalEnemyDR);
41 const int32 ScaledLevel = GlobalEnemyScalingThreshold + FMath::RoundToInt(ScaledExcess);
42
43 return ScaledLevel;
44}

46float UDTEnemyScalingData::CalculateScaledLife(float BaseLife, int32 EnemyLevel, float TierMultiplier) const
47{
48 // Formula: Life = (BaseLife × (1 + LifeGrowth)^Level) × TierMultiplier
49 const float LifeMultiplier = FMath::Pow(1.0f + GlobalEnemyLifeGrowth, static_cast<float>(EnemyLevel));
50 return BaseLife * LifeMultiplier * TierMultiplier;
