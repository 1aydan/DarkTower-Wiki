29UDTSettingsShared* UDTLocalPlayer::GetSharedSettings() const
30{
31 if (!SharedSettings)
32 {
33 // On PC it's okay to use the sync load because it only checks the disk
34 // This could use a platform tag to check for proper save support instead
35 bool bCanLoadBeforeLogin = PLATFORM_DESKTOP;
36
37 if (bCanLoadBeforeLogin)
38 {
39 SharedSettings = UDTSettingsShared::LoadOrCreateSettings(this);
40 }
41 else
42 {
43 SharedSettings = UDTSettingsShared::CreateTemporarySettings(this);
44 }
45 }
46

50void UDTLocalPlayer::LoadSharedSettingsFromDisk(bool bForceLoad)
51{
52 FUniqueNetIdRepl CurrentNetId = GetCachedUniqueNetId();
