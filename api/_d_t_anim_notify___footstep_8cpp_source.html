21void UDTAnimNotify_Footstep::Notify(USkeletalMeshComponent* MeshComp, UAnimSequenceBase* Animation, const FAnimNotifyEventReference& EventReference)
22{
23 Super::Notify(MeshComp, Animation, EventReference);
24
25 if (!MeshComp)
26 {
27 return;
28 }
29
30 AActor* Owner = MeshComp->GetOwner();
31 if (!Owner)
32 {
33 return;
34 }
35
36 // Skip anim editor preview actors so scrubbing doesn't fire footstep effects.
37 if (Owner->GetName().StartsWith(TEXT("AnimationEditorPreviewActor")))
38 {
39 return;
40 }
41
42 UWorld* World = MeshComp->GetWorld();
43 if (!World)
44 {
45 return;
46 }
47
48 // Footsteps are cosmetic-only — a dedicated server has no reason to spend the trace or spawn effects.
49 if (World->GetNetMode() == NM_DedicatedServer)
50 {
51 return;
52 }
53
54 UDTFootstepEffectSet* EffectSet = ResolveEffectSet(Owner);
55 if (!EffectSet)
56 {
57 return;
58 }
59
60 // Start slightly above the foot so a foot resting on the ground still produces a downward hit.
61 const FVector FootLocation = MeshComp->DoesSocketExist(FootSocket)
62 ? MeshComp->GetSocketLocation(FootSocket)
63 : MeshComp->GetComponentLocation();
64
65 const FVector TraceStart = FootLocation + FVector::UpVector * TraceStartHeight;
66 const FVector TraceEnd = FootLocation - FVector::UpVector * TraceDistance;
67
68 FCollisionQueryParams QueryParams(SCENE_QUERY_STAT(DTFootstep), /*bTraceComplex=*/false, Owner);
69 QueryParams.bReturnPhysicalMaterial = true;
70
71 FHitResult Hit;
72 const bool bHit = World->LineTraceSingleByChannel(Hit, TraceStart, TraceEnd, TraceChannel.GetValue(), QueryParams);
73
74#if ENABLE_DRAW_DEBUG
75 if (bDrawDebug)
76 {
77 DrawDebugLine(World, TraceStart, TraceEnd, bHit ? FColor::Green : FColor::Red, false, 2.0f, 0, 1.0f);
78 if (bHit)
79 {
80 DrawDebugPoint(World, Hit.ImpactPoint, 8.0f, FColor::Yellow, false, 2.0f);
81 }
82 }
83#endif
84
85 if (!bHit)
86 {
87 return;
88 }
89
90 const EPhysicalSurface Surface = UGameplayStatics::GetSurfaceType(Hit);
91 const FDTFootstepEffect& Effect = EffectSet->GetEffectForSurface(Surface);
92
93 const FVector SpawnLocation = Hit.ImpactPoint;
94
95 if (USoundBase* Sound = Effect.GetRandomSound())
96 {
97 UGameplayStatics::SpawnSoundAtLocation(
98 World,
99 Sound,
100 SpawnLocation,
101 FRotator::ZeroRotator,
102 VolumeMultiplier * Effect.VolumeMultiplier,
103 Effect.PitchMultiplier);
104 }
105
106 if (UNiagaraSystem* Particles = Effect.GetRandomParticles())
107 {
108 const FRotator SpawnRotation = Hit.ImpactNormal.Rotation();
109 UNiagaraFunctionLibrary::SpawnSystemAtLocation(World,Particles, SpawnLocation, SpawnRotation);
110 }
111}
