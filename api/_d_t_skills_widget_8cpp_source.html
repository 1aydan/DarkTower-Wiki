61void UDTSkillsWidget::InitializeSkillTreePanels()
62{
63 UDTSkillTreeComponent* SkillTree = GetSkillTreeComponent();
64 if (!SkillTree || !SkillTreePanelWidgetClass)
65 {
66 return;
67 }
68
69 const UDTGlobalData* GlobalData = UDTBlueprintLibrary::GetGlobalDataFast();
70 if (!GlobalData || !GlobalData->SkillTreeGlobalData)
71 {
72 UE_LOG(LogDT, Warning, TEXT("No SkillTreeGlobalData found in GlobalData"));
73 return;
74 }
75
76 SkillTreePanels.Empty();
77 SkillTree->OnSkillPointsChanged.AddUniqueDynamic(this, &UDTSkillsWidget::HandleSkillPointsChanged);
78
79 TLW_SkillTabs->SetLinkedSwitcher(CAS_SkillContent);
80 TLW_SkillTabs->OnTabSelected.AddUniqueDynamic(this, &UDTSkillsWidget::HandleSkillTabSelected);
81 TLW_SkillTabs->OnTabContentCreated.AddUniqueDynamic(this, &UDTSkillsWidget::HandleSkillTabContentCreated);
82
83 for (const FSkillTreeInstance& TreeInst : SkillTree->GetTreeInstances())
84 {
85 UDTSkillTreeDefinition* TreeDef = GlobalData->SkillTreeGlobalData->GetTreeDefinition(TreeInst.TreeTag);
86 if (!TreeDef)
87 {
88 continue;
89 }
90
91 UDTSkillTreePanelWidget* Panel = CreateWidget<UDTSkillTreePanelWidget>(GetOwningPlayer(), SkillTreePanelWidgetClass);
92 if (!Panel)
93 {
94 continue;
95 }
96
97 Panel->SetSkillTree(TreeDef, SkillTree);
98 SkillTreePanels.Add(Panel);
99
100 FTabEntryInfo TabInfo;
101 TabInfo.TabId = FName(*TreeInst.TreeTag.ToString());
102 TabInfo.TabText = TreeDef->TreeName;
103 TabInfo.TabButtonType = SkillTreeTabButtonClass;
104 TabInfo.CreatedTabContentWidget = Panel;
105
106 if (UTexture2D* Icon = TreeDef->TreeIcon.LoadSynchronous())
107 {
108 TabInfo.IconBrush.SetResourceObject(Icon);
109 }
110
111 TLW_SkillTabs->RegisterDynamicTab(TabInfo);
112 }
113
114 OnSkillPointsUpdated(SkillTree->GetAvailableSkillPoints(), SkillTree->GetTotalSkillPoints());
115}
