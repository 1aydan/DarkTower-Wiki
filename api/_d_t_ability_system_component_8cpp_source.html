969void UDTAbilitySystemComponent::ProcessAbilityInput(float DeltaTime, bool bGamePaused)
970{
971 if (bGamePaused)
972 {
973 ClearAbilityInput();
974 return;
975 }
976
977 static TArray<FGameplayAbilitySpecHandle> AbilitiesToActivate;
978 AbilitiesToActivate.Reset();
979
980 // Process all abilities that activate when the input is held.
981 for (const FGameplayAbilitySpecHandle& SpecHandle : InputHeldSpecHandles)
982 {
983 if (const FGameplayAbilitySpec* AbilitySpec = FindAbilitySpecFromHandle(SpecHandle))
984 {
985 if (AbilitySpec->Ability && !AbilitySpec->IsActive())
986 {
987 const UDTGameplayAbility* AbilityCDO = Cast<UDTGameplayAbility>(AbilitySpec->Ability);
988 if (AbilityCDO && AbilityCDO->GetActivationPolicy() == EDTAbilityActivationPolicy::WhileInputActive)
989 {
990 AbilitiesToActivate.AddUnique(AbilitySpec->Handle);
991 }
992 }
993 }
994 }
995
996 // Process all abilities that had their input pressed this frame.
997 for (const FGameplayAbilitySpecHandle& SpecHandle : InputPressedSpecHandles)
998 {
999 if (FGameplayAbilitySpec* AbilitySpec = FindAbilitySpecFromHandle(SpecHandle))
1000 {
1001 if (AbilitySpec->Ability)
1002 {
1003 AbilitySpec->InputPressed = true;
1004
1005 if (AbilitySpec->IsActive())
1006 {
1007 // Ability is active so pass along the input event.
1008 AbilitySpecInputPressed(*AbilitySpec);
1009 }
1010 else
1011 {
1012 const UDTGameplayAbility* AbilityCDO = Cast<UDTGameplayAbility>(AbilitySpec->Ability);
1013 if (!AbilityCDO || AbilityCDO->GetActivationPolicy() == EDTAbilityActivationPolicy::OnInputTriggered)
1014 {
1015 AbilitiesToActivate.AddUnique(AbilitySpec->Handle);
1016 }
1017 }
1018 }
1019 }
1020 }
1021
1022 // Try to activate all the abilities that are from presses and holds.
1023 // We do it all at once so that held inputs don't activate the ability
1024 // and then also send an input event to the ability because of the press.
1025 for (const FGameplayAbilitySpecHandle& AbilitySpecHandle : AbilitiesToActivate)
1026 {
1027 TryActivateAbility(AbilitySpecHandle);
1028 }
1029
1030 // Process all abilities that had their input released this frame.
1031 for (const FGameplayAbilitySpecHandle& SpecHandle : InputReleasedSpecHandles)
1032 {
1033 if (FGameplayAbilitySpec* AbilitySpec = FindAbilitySpecFromHandle(SpecHandle))
1034 {
1035 if (AbilitySpec->Ability)
1036 {
1037 AbilitySpec->InputPressed = false;
1038
1039 if (AbilitySpec->IsActive())
1040 {
1041 // Ability is active so pass along the input event.
1042 AbilitySpecInputReleased(*AbilitySpec);
1043 }
1044 }
1045 }
1046 }
1047
1048 // Clear the cached ability handles.
1049 InputPressedSpecHandles.Reset();
1050 InputReleasedSpecHandles.Reset();
1051}
