421bool UDTAbilitySystemComponent::IsActivationGroupBlocked(EAbilityActivationGroup Group) const
422{
423 bool bBlocked = false;
424
425 switch (Group)
426 {
427 case EAbilityActivationGroup::Independent:
428 // Independent abilities are never blocked.
429 bBlocked = false;
430 break;
431
432 case EAbilityActivationGroup::Exclusive_Replaceable:
433 case EAbilityActivationGroup::Exclusive_Blocking:
434 // Exclusive abilities can activate if nothing is blocking.
435 bBlocked = (ActivationGroupCounts[(uint8)EAbilityActivationGroup::Exclusive_Blocking] > 0);
436 break;
437
438 default:
439 checkf(false, TEXT("IsActivationGroupBlocked: Invalid ActivationGroup [%d]\n"), (uint8)Group);
440 break;
441 }
442
443 return bBlocked;

446void UDTAbilitySystemComponent::AddAbilityToActivationGroup(EAbilityActivationGroup Group, UDTGameplayAbility* Ability)
447{
448 check(Ability);
449 check(ActivationGroupCounts[(uint8)Group] < INT32_MAX);
450
451 ActivationGroupCounts[(uint8)Group]++;
452
453 const bool bReplicateCancelAbility = false;
454
455 switch (Group)
456 {
457 case EAbilityActivationGroup::Independent:
458 break;
459
460 case EAbilityActivationGroup::Exclusive_Replaceable:
461 case EAbilityActivationGroup::Exclusive_Blocking:
462 CancelActivationGroupAbilities(EAbilityActivationGroup::Exclusive_Replaceable, Ability, bReplicateCancelAbility);
463 break;
464
465 default:
466 checkf(false, TEXT("AddAbilityToActivationGroup: Invalid ActivationGroup [%d]\n"), (uint8)Group);
467 break;
468 }
469
470 const int32 ExclusiveCount = ActivationGroupCounts[(uint8)EAbilityActivationGroup::Exclusive_Replaceable] + ActivationGroupCounts[(uint8)EAbilityActivationGroup::Exclusive_Blocking];
471 if (!ensure(ExclusiveCount <= 1))
472 {
473 UE_LOG(LogDTAbility, Error, TEXT("AddAbilityToActivationGroup: Multiple exclusive abilities are running."));
474 }
475}

678FGameplayEventData UDTAbilitySystemComponent::BuildAbilityEventPayload(const FGameplayTag& EventTag,AActor* Instigator,AActor* Target,AActor* Causer,float Magnitude,const FGameplayTagContainer& ContextTags,const FHitResult& HitResult,const FGameplayEffectContextHandle& ExistingContext) const
679{
680 FGameplayEventData Payload;
681 Payload.EventTag = EventTag;
682 Payload.Instigator = Instigator;
683 Payload.Target = Target;
684 Payload.EventMagnitude = Magnitude;
685
686 // Use existing context or make a new one
687 FGameplayEffectContextHandle ContextHandle = ExistingContext.IsValid() ? ExistingContext : MakeEffectContext();
688
689 ContextHandle.AddInstigator(Instigator, Causer);
690 if (IsValid(Causer))
691 {
692 ContextHandle.AddSourceObject(Causer);
693 }
694
695 if (HitResult.HasValidHitObjectHandle() || HitResult.bBlockingHit || !HitResult.Location.IsNearlyZero())
696 {
697 ContextHandle.AddHitResult(HitResult, true);
698 ContextHandle.AddOrigin(!HitResult.ImpactPoint.IsNearlyZero() ? HitResult.ImpactPoint : HitResult.Location);
699 }
700 else if (IsValid(Target))
701 {
702 ContextHandle.AddOrigin(Target->GetActorLocation());
703 }
704
705 // Set project-level context tags on our custom effect context
706 if (FDTGameplayEffectContext* DTContext = UDTBlueprintLibrary::GetDTEffectContextMutable(ContextHandle))
707 {
708 DTContext->AppendContextTags(ContextTags);
