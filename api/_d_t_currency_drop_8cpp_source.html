96 OwningSubsystem = GetWorld() ? GetWorld()->GetSubsystem<UDTCurrencySubsystem>() : nullptr;

98 // Set owner for replication relevancy
99 APawn* OwningPawn = InOwningPlayer ? InOwningPlayer->GetPawn() : nullptr;
100 SetOwner(OwningPawn);
101
102 // Teleport into the world
103 SetActorLocation(SpawnLocation);
104 SetActorHiddenInGame(false);
105 SetActorEnableCollision(false);
106
107 // Look up visual tier and apply mesh
108 const UDTGlobalData* GlobalData = UDTBlueprintLibrary::GetGlobalDataFast();
109 const UDTCurrencyGlobalData* CurrencyGlobalData = GlobalData ? GlobalData->CurrencyGlobalData : nullptr;
110 if (CurrencyGlobalData)
111 {
112 const FCurrencyDropVisualTier* Tier = CurrencyGlobalData->GetVisualTierForAmount(CurrencyType, Amount);
113 ApplyVisualTier(Tier);
114 }
115
116 APlayerController* LocalPC = GetWorld() ? GetWorld()->GetFirstPlayerController() : nullptr;
117 if (LocalPC && HasAuthority())
118 {
119 const AActor* OwnerActor = GetOwner();
120 const APawn* LocalPawn = LocalPC->GetPawn();
121
122 const bool bIsOwnedByLocalPlayer =
123 (OwnerActor == LocalPC) ||
124 (LocalPawn && OwnerActor == LocalPawn) ||
125 (OwnerActor && LocalPawn && OwnerActor->IsOwnedBy(LocalPawn)) ||
126 (OwnerActor && OwnerActor->IsOwnedBy(LocalPC));
127
128 if (!bIsOwnedByLocalPlayer)
129 {
130 if (DropMeshComponent)
131 {
132 DropMeshComponent->SetVisibility(false, false);
133 }
134
135 if (DropNiagaraComponent)
136 {
137 DropNiagaraComponent->SetVisibility(false, false);
138 }
139
140 return;
141 }
142 }
143
144 OnDropActivated(CurrencyType, Amount);
145
146 ForceNetUpdate();

199void ADTCurrencyDrop::ExecutePickup()
200{
201 if (DropState == EDTCurrencyDropState::PickedUp || DropState == EDTCurrencyDropState::Pooled)
202 {
203 return;
204 }
205
206 DropState = EDTCurrencyDropState::PickedUp;
207
208 APlayerController* PC = OwningPlayer.Get();
209 if (PC)
210 {
211 if (ADTPlayerState* PS = PC->GetPlayerState<ADTPlayerState>())
212 {
213 if (UDTCurrencyComponent* CurrencyComp = PS->GetCurrencyComponent())
214 {
215 CurrencyComp->AddCurrency(CurrencyType, Amount);
216 }
217 }
218
219 if (OwningSubsystem)
220 {
221 OwningSubsystem->OnCurrencyPickedUp.Broadcast(CurrencyType, Amount, PC);
222 }
223 }
224
225 Multicast_OnPickedUp(GetActorLocation(), CurrencyType, Amount);
226
227 ForceNetUpdate();

239void ADTCurrencyDrop::ApplyVisualTier(const FCurrencyDropVisualTier* Tier)
240{
241 if (!DropMeshComponent)

248 DropMeshComponent->SetVisibility(false);
