75void UDTSettingsListEntrySetting_KeyboardInput::ChangeBinding(int32 InKeyBindSlot, FKey InKey)
76{
77 OriginalKeyToBind = InKey;
78 TArray<FName> ActionsForKey;
79 KeyboardInputSetting->GetAllMappedActionsFromKey(InKeyBindSlot, InKey, ActionsForKey);
80 if (!ActionsForKey.IsEmpty())
81 {
82 UKeyAlreadyBoundWarning* KeyAlreadyBoundWarning = CastChecked<UKeyAlreadyBoundWarning>(
83 UCommonUIExtensions::PushContentToLayer_ForPlayer(GetOwningLocalPlayer(), PressAnyKeyLayer, KeyAlreadyBoundWarningPanelClass));
84
85 FString ActionNames;
86 for (FName ActionName : ActionsForKey)
87 {
88 ActionNames += ActionName.ToString() += ", ";
89 }
90
91 FFormatNamedArguments Args;
92 Args.Add(TEXT("InKey"), InKey.GetDisplayName());
93 Args.Add(TEXT("ActionNames"), FText::FromString(ActionNames));
94
95 KeyAlreadyBoundWarning->SetWarningText(FText::Format(LOCTEXT("WarningText", "{InKey} is already bound to {ActionNames} are you sure you want to rebind it?"), Args));
96 KeyAlreadyBoundWarning->SetCancelText(FText::Format(LOCTEXT("CancelText", "Press escape to cancel, or press {InKey} again to confirm rebinding."), Args));
97
98 if (InKeyBindSlot == 1)
99 {
100 KeyAlreadyBoundWarning->OnKeySelected.AddUObject(this, &ThisClass::HandleSecondaryDuplicateKeySelected, KeyAlreadyBoundWarning);
101 }
102 else
103 {
104 KeyAlreadyBoundWarning->OnKeySelected.AddUObject(this, &ThisClass::HandlePrimaryDuplicateKeySelected, KeyAlreadyBoundWarning);
105 }
106 KeyAlreadyBoundWarning->OnKeySelectionCanceled.AddUObject(this, &ThisClass::HandleKeySelectionCanceled, KeyAlreadyBoundWarning);
107 }
108 else
109 {
110 KeyboardInputSetting->ChangeBinding(InKeyBindSlot, InKey);
111 }
112}
