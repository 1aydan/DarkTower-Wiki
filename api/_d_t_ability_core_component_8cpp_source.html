276static FVector ComputeDamageDirection2D(const AActor* Owner, const AActor* SourceActor, const FHitResult* HitResult)
277{
278 if (!Owner)
279 {
280 return FVector::ZeroVector;
281 }
282
283 FVector Direction = FVector::ZeroVector;
284 if (SourceActor && SourceActor != Owner)
285 {
286 Direction = (Owner->GetActorLocation() - SourceActor->GetActorLocation()).GetSafeNormal2D();
287 }
288
289 if (Direction.IsNearlyZero() && HitResult)
290 {
291 Direction = (-HitResult->ImpactNormal).GetSafeNormal2D();
292 }
293
294 return Direction;

367void UDTAbilityCoreComponent::OnAttributeChanged(const FOnAttributeChangeData& Data)
368{
369 const float NewValue = Data.NewValue;
370 const float OldValue = Data.OldValue;
371
372 // Prevent broadcasting attribute changes if New and Old values are the same
373 // (most likely because of clamping in post gameplay effect execute)
374 if (OldValue == NewValue)
375 {
376 return;
377 }
378
379 const FGameplayEffectModCallbackData* ModData = Data.GEModData;
380 FGameplayTagContainer SourceTags = FGameplayTagContainer();
381 if (ModData)
382 {
383 SourceTags = *ModData->EffectSpec.CapturedSourceTags.GetAggregatedTags();
384 }
385
386 // Broadcast attribute change to component
387 OnAttributeChange.Broadcast(Data.Attribute, NewValue - OldValue, SourceTags);
388}

896void UDTAbilityCoreComponent::HandleCooldownOnAbilityCommit(UGameplayAbility* ActivatedAbility)
897{
898 if (!OwnerAbilitySystemComponent)
899 {
900 return;
901 }
902
903 if (!IsValid(ActivatedAbility))
904 {
905 UE_LOG(LogDTAbility, Warning, TEXT("UDTAbilityCoreComponent::HandleCooldownOnAbilityCommit() Activated ability not valid"));
906 return;
907 }
908
909 // Figure out cooldown
910 UGameplayEffect* CooldownGE = ActivatedAbility->GetCooldownGameplayEffect();
911 if (!CooldownGE)
912 {
913 return;
914 }
915
916 if (!ActivatedAbility->IsInstantiated())
917 {
918 return;
919 }
920
921 const FGameplayTagContainer* CooldownTags = ActivatedAbility->GetCooldownTags();
922 if (!CooldownTags || CooldownTags->Num() <= 0)
923 {
924 return;
925 }
926
927 FGameplayAbilityActorInfo ActorInfo = ActivatedAbility->GetActorInfo();
928 const FGameplayAbilitySpecHandle AbilitySpecHandle = ActivatedAbility->GetCurrentAbilitySpecHandle();
929
930 float TimeRemaining = 0.f;
931 float Duration = 0.f;
932 ActivatedAbility->GetCooldownTimeRemainingAndDuration(AbilitySpecHandle, &ActorInfo, TimeRemaining, Duration);
933
934 OnCooldownStart.Broadcast(ActivatedAbility, *CooldownTags, TimeRemaining, Duration);
935
936 // Register delegate to monitor any change to cooldown gameplay tag to be able to figure out when a cooldown expires
937 TArray<FGameplayTag> GameplayTags;
938 CooldownTags->GetGameplayTagArray(GameplayTags);
939 for (const FGameplayTag& GameplayTag : GameplayTags)
940 {
941 OwnerAbilitySystemComponent->RegisterGameplayTagEvent(GameplayTag)
942 .AddUObject(this, &UDTAbilityCoreComponent::OnCooldownGameplayTagChanged, AbilitySpecHandle, Duration);
943
944 GameplayTagBoundToDelegates.AddUnique(GameplayTag);
945 }
946}
