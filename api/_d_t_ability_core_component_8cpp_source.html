73void UDTAbilityCoreComponent::Die()
74{
75 if (OwnerAbilitySystemComponent)
76 {
77 if (OwnerAbilitySystemComponent->HasMatchingGameplayTag(DTGameplayTags::TAG_STATUS_DEAD))
78 {
79 UE_LOG(LogDTAbility, Verbose, TEXT("UDTAbilityCoreComponent::Die() - Attempted to die when already dead"));
80 return;
81 }
82
83 OwnerAbilitySystemComponent->AddLooseGameplayTag(DTGameplayTags::TAG_STATUS_DEAD);
84 }
85
86 if (GetOwner())
87 {
88 SendAbilityEvent(DTGameplayTags::TAG_EVENT_DEATH, MakeGameplayEventData(DTGameplayTags::TAG_EVENT_DEATH, 0.0f, GetOwner(), GetOwner(), GetOwner()));
89 }
90
91 if (GetOwnerRole() == ROLE_Authority)
92 {
93 MulticastHandleDeath();
94 return;
95 }
96
97 OnDeath.Broadcast();
98}

100FGameplayEventData UDTAbilityCoreComponent::MakeGameplayEventData(const FGameplayTag& EventTag, float Magnitude, AActor* Instigator, AActor* Target, AActor* Causer, const FGameplayTagContainer& ContextTags, const FHitResult* HitResult, bool bWasFatal, bool bWasKill, const FGameplayEffectContextHandle& SourceEffectContext) const
101{
102 FGameplayTagContainer ResultTags = ContextTags;
103 FHitResult EventHitResult = HitResult ? *HitResult : FHitResult(Target, nullptr, IsValid(Target) ? Target->GetActorLocation() : FVector::ZeroVector, FVector::ZeroVector);
104
105 if (bWasFatal)
106 {
107 ResultTags.AddTag(DTGameplayTags::TAG_EFFECT_RESULT_FATAL);
108 }
109
110 if (bWasKill)
111 {
112 ResultTags.AddTag(DTGameplayTags::TAG_EFFECT_RESULT_KILL);
113 }
114
115 UDTAbilitySystemComponent* ASC = GetAbilitySystemComponent();
116 if (ASC)
117 {
118 return ASC->BuildAbilityEventPayload(EventTag, Instigator, Target, Causer, Magnitude, ResultTags, EventHitResult, SourceEffectContext);
119 }
120
121 // Fallback: build manually without ASC (shouldn't normally happen)
122 FGameplayEventData Payload;
123 Payload.EventTag = EventTag;
124 Payload.Instigator = Instigator;
125 Payload.Target = Target;
126 Payload.EventMagnitude = Magnitude;
127 return Payload;
128}
