9static FText FormatCountdownText(float Seconds)
10{
11 const int32 TotalSecs = FMath::CeilToInt(FMath::Max(0.f, Seconds));
12 const int32 Mins = TotalSecs / 60;
13 const int32 Secs = TotalSecs % 60;
14 return FText::FromString(FString::Printf(TEXT("%d:%02d"), Mins, Secs));
15}

85void UDTDeathOverlayWidget::InitializeDeathOverlay(float RespawnDuration, int32 InLivesRemaining, APlayerState* InitialSpectateTarget)
86{
87 TotalDuration = FMath::Max(0.f, RespawnDuration);
88 SecondsRemaining = TotalDuration;
89 LivesRemaining = InLivesRemaining;
90 bCountingDown = TotalDuration > 0.f;
91
92 RefreshTimerWidgets(SecondsRemaining, TotalDuration);
93 RefreshLivesText();
94
95 if (bCountingDown)
96 {
97 OnCountdownStarted(TotalDuration);
98 }
99 else
100 {
101 OnCountdownFinished();
102 }
103
104 CurrentSpectatorTarget = InitialSpectateTarget;
105 RefreshSpectatorText(InitialSpectateTarget);
106 if (InitialSpectateTarget)
107 {
108 OnSpectatorTargetChanged(InitialSpectateTarget);
109 }
110
111 OnDeathOverlayInitialized(RespawnDuration, InLivesRemaining, InitialSpectateTarget);
112}
