85void ADTAbilityActor::InitializeAbilitySystem()
86{
87 check(AbilitySystemComponent);
88 AbilitySystemComponent->InitAbilityActorInfo(this, this);
89
90 if (HasAuthority())
91 {
92 for (const TSubclassOf<UGameplayAbility>& AbilityClass : DefaultAbilities)
93 {
94 if (AbilityClass)
95 {
96 AbilitySystemComponent->GiveAbility(FGameplayAbilitySpec(AbilityClass, 1, INDEX_NONE, this));
97 }
98 }
99
100 for (const auto& [AttributeSetClass, DataTable] : DefaultAttributeSets)
101 {
102 if (AttributeSetClass)
103 {
104 UAttributeSet* NewAttributeSet = NewObject<UAttributeSet>(this, AttributeSetClass);
105 AbilitySystemComponent->AddAttributeSetSubobject(NewAttributeSet);
106
107 if (DataTable)
108 {
109 NewAttributeSet->InitFromMetaDataTable(DataTable);
110 }
111 }
112 }
113
114 for (const TSubclassOf<UGameplayEffect>& EffectClass : DefaultEffects)
115 {
116 if (EffectClass)
117 {
118 FGameplayEffectContextHandle EffectContext = AbilitySystemComponent->MakeEffectContext();
119 EffectContext.AddSourceObject(this);
120 AbilitySystemComponent->ApplyGameplayEffectToSelf(EffectClass->GetDefaultObject<UGameplayEffect>(), 1.0f, EffectContext);
121 }
122 }
123 }
124
125 if (AbilitySystemComponent)
126 {
127 for (const FGameplayTag& Tag : DefaultGameplayTags)
128 {
129 AbilitySystemComponent->AddLooseGameplayTag(Tag);
130 }
