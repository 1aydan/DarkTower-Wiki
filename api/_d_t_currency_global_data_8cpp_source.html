62int64 UDTCurrencyGlobalData::CalculateScaledAmount(FGameplayTag CurrencyType, int64 BaseAmount, int32 Floor, const FGameplayTagContainer& TierTags) const
63{
64 const FCurrencyScalingConfig* Scaling = CurrencyScaling.Find(CurrencyType);
65 if (!Scaling || Floor <= 1)
66 {
67 return BaseAmount;
68 }
69
70 const double FloorBonus = static_cast<double>(Floor) * Scaling->AmountPerFloor;
71 double ScaledAmount = static_cast<double>(BaseAmount) + FloorBonus;
72
73 // Apply the highest matching tier multiplier
74 float BestMultiplier = 1.0f;
75 for (const auto& Pair : Scaling->TierMultipliers)
76 {
77 if (TierTags.HasTag(Pair.Key))
78 {
79 BestMultiplier = FMath::Max(BestMultiplier, Pair.Value);
80 }
81 }
82 ScaledAmount *= BestMultiplier;
83
84 return FMath::Max(static_cast<int64>(1), static_cast<int64>(ScaledAmount));
85}
