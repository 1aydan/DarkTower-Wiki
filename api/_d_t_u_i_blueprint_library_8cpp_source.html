33static float EvaluateScalableDescriptionValueForLevel(const FDynamicDescriptionValue& ValueDef, const int32 EvaluationLevel)
34{
35 if (EvaluationLevel <= 0)
36 {
37 return ValueDef.ScalableValue.GetValueAtLevel(0.0f);
38 }
39
40 if (ValueDef.ScalableEvaluationMode == EScalableValueEvaluationMode::Direct)
41 {
42 return ValueDef.ScalableValue.GetValueAtLevel(static_cast<float>(EvaluationLevel));
43 }
44
45 switch (ValueDef.Operation)
46 {
47 case EStatModifierOp::Add:
48 {
49 float AccumulatedValue = 0.0f;
50 for (int32 Level = 1; Level <= EvaluationLevel; ++Level)
51 {
52 AccumulatedValue += ValueDef.ScalableValue.GetValueAtLevel(static_cast<float>(Level));
53 }
54 return AccumulatedValue;
55 }
56
57 case EStatModifierOp::Multiply:
58 {
59 float AccumulatedValue = 1.0f;
60 for (int32 Level = 1; Level <= EvaluationLevel; ++Level)
61 {
62 AccumulatedValue += ValueDef.ScalableValue.GetValueAtLevel(static_cast<float>(Level)) - 1.0f;
63 }
64 return AccumulatedValue;
65 }
66
67 case EStatModifierOp::Override:
68 default:
69 return ValueDef.ScalableValue.GetValueAtLevel(static_cast<float>(EvaluationLevel));
70 }
71}

73static float ReadObjectPropertyValue(const UObject* Object, const FName PropertyName, const float Level)
74{
75 if (!Object || PropertyName.IsNone())
76 {
77 return 0.0f;
78 }
79
80 FProperty* Property = FindFProperty<FProperty>(Object->GetClass(), PropertyName);
81 if (!Property)
82 {
83 UE_LOG(LogDT, Warning, TEXT("ReadObjectPropertyValue: '%s' has no property named '%s'"), *Object->GetName(), *PropertyName.ToString());
84 return 0.0f;
85 }
86
87 const void* ValuePtr = Property->ContainerPtrToValuePtr<void>(Object);
88
89 if (const FNumericProperty* NumericProperty = CastField<FNumericProperty>(Property))
90 {
91 return NumericProperty->IsFloatingPoint()
92 ? static_cast<float>(NumericProperty->GetFloatingPointPropertyValue(ValuePtr))
93 : static_cast<float>(NumericProperty->GetSignedIntPropertyValue(ValuePtr));
94 }
95
96 if (const FBoolProperty* BoolProperty = CastField<FBoolProperty>(Property))
97 {
98 return BoolProperty->GetPropertyValue(ValuePtr) ? 1.0f : 0.0f;
99 }
100
101 if (const FStructProperty* StructProperty = CastField<FStructProperty>(Property))
102 {
103 if (StructProperty->Struct == FScalableFloat::StaticStruct())
104 {
105 return static_cast<const FScalableFloat*>(ValuePtr)->GetValueAtLevel(Level);
106 }
107 }
108
109 UE_LOG(LogDT, Warning, TEXT("ReadObjectPropertyValue: property '%s' on '%s' is not a supported numeric type"), *PropertyName.ToString(), *Object->GetName());
110 return 0.0f;
111}

113static float EvaluateDescriptionValue(const FDynamicDescriptionValue& ValueDef, const float NormalizedValue, const int32 EvaluationLevel = 0, const UObject* PropertyContext = nullptr)
114{
115 if (ValueDef.ValueSource == EDynamicDescriptionValueSource::AbilityProperty)
116 {
117 return ReadObjectPropertyValue(PropertyContext, ValueDef.BoundPropertyName, static_cast<float>(EvaluationLevel));
118 }
119
120 if (ValueDef.ValueSource == EDynamicDescriptionValueSource::ScalableFloat)
121 {
122 if (EvaluationLevel > 0)
123 {
124 return EvaluateScalableDescriptionValueForLevel(ValueDef, EvaluationLevel);
125 }
126
127 return ValueDef.ScalableValue.GetValueAtLevel(FMath::Clamp(NormalizedValue, 0.0f, 1.0f));
128 }
129
130 return FMath::Lerp(ValueDef.MinValue, ValueDef.MaxValue, FMath::Clamp(NormalizedValue, 0.0f, 1.0f));
131}

133static FString FormatDescriptionValuePlainString(const FDynamicDescriptionValue& ValueDef, float RawValue)
134{
135 const bool bIsPercentage = (ValueDef.Operation == EStatModifierOp::Multiply);
136 float DisplayValue = bIsPercentage ? (RawValue - 1.0f) * 100.0f : RawValue;
137
138 FString SignPrefix;
139 if (ValueDef.bShowSign)
140 {
141 SignPrefix = DisplayValue >= 0.0f ? TEXT("+") : TEXT("-");
142 DisplayValue = FMath::Abs(DisplayValue);
143 }
144
145 FString ValueString;
146 if (bIsPercentage || ValueDef.DecimalPlaces > 0)
147 {
148 ValueString = FString::Printf(TEXT("%.*f%s"), ValueDef.DecimalPlaces, DisplayValue, bIsPercentage ? TEXT("%") : TEXT(""));
149 }
150 else
151 {
152 ValueString = FString::Printf(TEXT("%d"), FMath::FloorToInt(DisplayValue));
153 }
154
155 if (!SignPrefix.IsEmpty())
156 {
157 return FString::Printf(TEXT("%s%s"), *SignPrefix, *ValueString);
158 }
159
160 return ValueString;

187static FText FormatResolvedDynamicDescription(const FText& DescriptionTemplate, const TArray<FDynamicDescriptionValue>& DescriptionValues, const TArray<FString>& ResolvedValues)
188{
189 if (DescriptionTemplate.IsEmpty())
190 {
191 return FText::GetEmpty();

193
194 FString DescriptionString = DescriptionTemplate.ToString();
195 for (int32 Index = 0; Index < DescriptionValues.Num(); ++Index)
196 {
197 const FDynamicDescriptionValue& ValueDef = DescriptionValues[Index];
198 const FString PlaceholderName = ValueDef.Placeholder != NAME_None ? ValueDef.Placeholder.ToString() : FString::Printf(TEXT("Value_%d"), Index);
199 const FString Placeholder = FString::Printf(TEXT("{%s}"), *PlaceholderName);
200 const FString ReplacementValue = ResolvedValues.IsValidIndex(Index) ? ResolvedValues[Index] : FString();
201 DescriptionString = DescriptionString.Replace(*Placeholder, *ReplacementValue);
202 }
203
204 return FText::FromString(DescriptionString);

341int32 UDTUIBlueprintLibrary::GetItemAffixDisplayInfo(const FInventoryItemInfo& ItemInfo, TArray<FAffixDisplayInfo>& OutAffixDisplayInfo, bool bGroupBySlotType, int32 DecimalPlaces)
342{
343 OutAffixDisplayInfo.Empty();
344
345 if (ItemInfo.Modifiers.Num() == 0)
346 {
347 return 0;
348 }
349
350 // Recover the rolled tier for display: stored alpha = ItemPower * TierFraction, so
351 // TierFraction = alpha / ItemPower and Tier = round(TierFraction * (N-1)) + 1.
352 int32 AffixTierCount = 10;
353 {
354 bool bGlobalValid = false;
355 if (const UDTGlobalData* GlobalData = UDTBlueprintLibrary::GetGlobalData(bGlobalValid))
356 {
357 if (bGlobalValid && GlobalData->GlobalLootData)
358 {
359 AffixTierCount = FMath::Max(2, GlobalData->GlobalLootData->AffixTierCount);
360 }
361 }
362 }
363
364 auto FillTier = [AffixTierCount, &ItemInfo](FAffixDisplayInfo& Info, const FItemModifierEntry& Mod)
365 {
366 const float ItemPower = FMath::Max(ItemInfo.ItemPower, KINDA_SMALL_NUMBER);
367 const float TierFraction = FMath::Clamp(Mod.NormalizedRollValue / ItemPower, 0.0f, 1.0f);
368 Info.Tier = FMath::Clamp(FMath::RoundToInt(TierFraction * (AffixTierCount - 1)) + 1, 1, AffixTierCount);
369 Info.MaxTier = AffixTierCount;
370 };
371
372 if (bGroupBySlotType)
373 {
374 TArray<FAffixDisplayInfo> PrefixInfos;
375 TArray<FAffixDisplayInfo> SuffixInfos;
376 TArray<FAffixDisplayInfo> ImplicitInfos;
377
378 for (const FItemModifierEntry& ModEntry : ItemInfo.Modifiers)
379 {
380 UDTAffixDefinition* AffixDef = UDefinitionAssetManager::Get().LoadDefinitionByGuid<UDTAffixDefinition>(ModEntry.ModifierGuid);
381 if (!AffixDef)
382 {
383 continue;
384 }
385
386 FAffixDisplayInfo DisplayInfo;
387 DisplayInfo.DisplayName = AffixDef->DisplayName;
388 DisplayInfo.Description = FormatAffixDescription(AffixDef, ModEntry.NormalizedRollValue, DecimalPlaces, ItemInfo.ItemLevel);
389 DisplayInfo.AffixDefinition = AffixDef;
390 DisplayInfo.ModifierEntry = ModEntry;
391 FillTier(DisplayInfo, ModEntry);
392
393 switch (AffixDef->SlotType)
394 {
395 case EAffixSlotType::Prefix:
396 PrefixInfos.Add(DisplayInfo);
397 break;
398 case EAffixSlotType::Suffix:
399 SuffixInfos.Add(DisplayInfo);
400 break;
401 case EAffixSlotType::Implicit:
402 ImplicitInfos.Add(DisplayInfo);
403 break;
404 }
405 }
406
407 OutAffixDisplayInfo.Append(ImplicitInfos);
408 OutAffixDisplayInfo.Append(PrefixInfos);
409 OutAffixDisplayInfo.Append(SuffixInfos);
410 }
411 else
412 {
413 for (const FItemModifierEntry& ModEntry : ItemInfo.Modifiers)
414 {
415 UDTAffixDefinition* AffixDef = UDefinitionAssetManager::Get().LoadDefinitionByGuid<UDTAffixDefinition>(ModEntry.ModifierGuid);
416 if (!AffixDef)
417 {
418 continue;
419 }
420
421 FAffixDisplayInfo DisplayInfo;
422 DisplayInfo.DisplayName = AffixDef->DisplayName;
423 DisplayInfo.Description = FormatAffixDescription(AffixDef, ModEntry.NormalizedRollValue, DecimalPlaces, ItemInfo.ItemLevel);
424 DisplayInfo.AffixDefinition = AffixDef;
425 DisplayInfo.ModifierEntry = ModEntry;
426 FillTier(DisplayInfo, ModEntry);
427
428 OutAffixDisplayInfo.Add(DisplayInfo);
429 }
430 }
431
432 return OutAffixDisplayInfo.Num();
433}

517bool UDTUIBlueprintLibrary::GetSkillTreeNodeTooltipDisplayInfo(const FSkillTreeNode& NodeDefinition, int32 CurrentRank, FSkillTreeNodeTooltipDisplayInfo& OutDisplayInfo)
518{
519 OutDisplayInfo = FSkillTreeNodeTooltipDisplayInfo();
520 if (!NodeDefinition.GetDefinition())
521 {
522 return false;
523 }
524
525 OutDisplayInfo.DisplayName = NodeDefinition.GetNodeName();
526
527 // Ability-granting nodes without an authored description inherit the granted ability's body/values.
528 FText EffectiveDescription;
529 const TArray<FDynamicDescriptionValue>* EffectiveValuesPtr = nullptr;
530 const UObject* PropertyContext = ResolveSkillNodeTooltipSource(NodeDefinition, EffectiveDescription, EffectiveValuesPtr);
531
532 OutDisplayInfo.BaseDescription = EffectiveDescription;
533 OutDisplayInfo.CurrentRank = FMath::Max(CurrentRank, 0);
534 OutDisplayInfo.MaxRank = FMath::Max(NodeDefinition.GetMaxRank(), 0);
535 OutDisplayInfo.bHasNextRank = OutDisplayInfo.CurrentRank < OutDisplayInfo.MaxRank;
536 OutDisplayInfo.NextRank = OutDisplayInfo.bHasNextRank ? ClampDisplayRank(OutDisplayInfo.CurrentRank + 1, OutDisplayInfo.MaxRank) : OutDisplayInfo.MaxRank;
537
538 const TArray<FDynamicDescriptionValue>& TooltipValues = *EffectiveValuesPtr;
539 const int32 CurrentEvaluationRank = OutDisplayInfo.CurrentRank > 0 ? ClampDisplayRank(OutDisplayInfo.CurrentRank, OutDisplayInfo.MaxRank) : 0;
540 const int32 NextEvaluationRank = OutDisplayInfo.bHasNextRank ? ClampDisplayRank(OutDisplayInfo.NextRank, OutDisplayInfo.MaxRank) : 0;
541 const int32 MaxEvaluationRank = OutDisplayInfo.MaxRank > 0 ? ClampDisplayRank(OutDisplayInfo.MaxRank, OutDisplayInfo.MaxRank) : 0;
542 OutDisplayInfo.PrimaryDescription = (OutDisplayInfo.CurrentRank > 0 && OutDisplayInfo.bHasNextRank)
543 ? FormatDynamicDescriptionComparison(
544 OutDisplayInfo.BaseDescription,
545 TooltipValues,
546 RankToNormalizedValue(OutDisplayInfo.CurrentRank, OutDisplayInfo.MaxRank),
547 RankToNormalizedValue(OutDisplayInfo.NextRank, OutDisplayInfo.MaxRank),
548 CurrentEvaluationRank,
549 NextEvaluationRank,
550 PropertyContext)
551 : FText::GetEmpty();
552
553 if (OutDisplayInfo.PrimaryDescription.IsEmpty())
554 {
555 if (OutDisplayInfo.bHasNextRank)
556 {
557 OutDisplayInfo.PrimaryDescription = FormatDynamicDescriptionResolved(
558 OutDisplayInfo.BaseDescription,
559 TooltipValues,
560 RankToNormalizedValue(OutDisplayInfo.NextRank, OutDisplayInfo.MaxRank),
561 NextEvaluationRank, PropertyContext);
562 }
563 else if (OutDisplayInfo.CurrentRank > 0)
564 {
565 OutDisplayInfo.PrimaryDescription = FormatDynamicDescriptionResolved(
566 OutDisplayInfo.BaseDescription,
567 TooltipValues,
568 RankToNormalizedValue(OutDisplayInfo.CurrentRank, OutDisplayInfo.MaxRank),
569 CurrentEvaluationRank, PropertyContext);
570 }
571 else if (OutDisplayInfo.MaxRank > 0)
572 {
573 OutDisplayInfo.PrimaryDescription = FormatDynamicDescriptionResolved(
574 OutDisplayInfo.BaseDescription,
575 TooltipValues,
576 1.0f,
577 MaxEvaluationRank, PropertyContext);
578 }
579 else
580 {
581 OutDisplayInfo.PrimaryDescription = OutDisplayInfo.BaseDescription;
582 }
583 }
584
585 OutDisplayInfo.ValueDisplayInfo.Reserve(TooltipValues.Num());
586 for (int32 Index = 0; Index < TooltipValues.Num(); ++Index)
587 {
588 const FDynamicDescriptionValue& ValueDef = TooltipValues[Index];
589 FSkillTreeNodeTooltipValueDisplayInfo& ValueInfo = OutDisplayInfo.ValueDisplayInfo.AddDefaulted_GetRef();
590 ValueInfo.Placeholder = ValueDef.Placeholder != NAME_None ? ValueDef.Placeholder : FName(*FString::Printf(TEXT("Value_%d"), Index));
591 ValueInfo.DisplayLabel = ValueDef.DisplayLabel;
592 const float CurrentNormalizedValue = RankToNormalizedValue(OutDisplayInfo.CurrentRank, OutDisplayInfo.MaxRank);
593 const float NextNormalizedValue = RankToNormalizedValue(OutDisplayInfo.NextRank, OutDisplayInfo.MaxRank);
594 ValueInfo.CurrentValue = OutDisplayInfo.CurrentRank > 0
595 ? FText::FromString(FormatDescriptionValueString(ValueDef, EvaluateDescriptionValue(ValueDef, CurrentNormalizedValue, CurrentEvaluationRank, PropertyContext)))
596 : FText::GetEmpty();
597 ValueInfo.NextValue = OutDisplayInfo.bHasNextRank
598 ? FText::FromString(FormatDescriptionValueString(ValueDef, EvaluateDescriptionValue(ValueDef, NextNormalizedValue, NextEvaluationRank, PropertyContext)))
599 : FText::GetEmpty();
600 ValueInfo.MaxValue = OutDisplayInfo.MaxRank > 0
601 ? FText::FromString(FormatDescriptionValueString(ValueDef, EvaluateDescriptionValue(ValueDef, 1.0f, MaxEvaluationRank, PropertyContext)))
602 : FText::GetEmpty();
603 }
604
605 return true;
