115 ADTPocketLevelStageManager* PocketLevelStageManager = UDTBlueprintLibrary::GetPocketLevelStageManager(Instance->GetWorld());
116
117 if (!PocketLevelStageManager)
118 {
119 UE_LOG(LogDTInventory, Warning, TEXT("UDTCharacterPreviewWidget::OnInventoryLevelReady - Could not find PocketLevelStageManager in pocket level instance"));
120 return;

123 CharacterPreviewStageManager = PocketLevelStageManager;

126 // Set Diffuse and AlphaMask for the dynamic material of the image
127 auto* ImgMaterial = IMG_CharacterPreview->GetDynamicMaterial();
128 ImgMaterial->SetTextureParameterValue(TEXT("Diffuse"), PocketCaptureInstance->GetOrCreateDiffuseRenderTarget());
129 ImgMaterial->SetTextureParameterValue(TEXT("AlphaMask"), PocketCaptureInstance->GetOrCreateAlphaMaskRenderTarget());
130
131 // Spawn the preview character and mirror any already-bound equipment onto it
132 PocketLevelStageManager->SpawnCharacterInPocketLevel(nullptr);
