169void UDTEndgameTabWidget::HandleFloorRepeatTick()
170{
171 const int32 Current = GetCurrentFloor();
172 const int32 Next = bRepeatIncrement ? Current + 1 : Current - 1;
173 SelectFloor(Next);
174
175 // Stop the timer automatically once the player hits the boundary.
176 const bool bAtBoundary = bRepeatIncrement ? (GetCurrentFloor() >= GetMaxUnlockedFloor())
177 : (GetCurrentFloor() <= 1);
178 if (bAtBoundary)
179 {
180 if (UWorld* World = GetWorld())
181 {
182 World->GetTimerManager().ClearTimer(FloorRepeatTimerHandle);
183 }
184 }
185}

187void UDTEndgameTabWidget::HandleFloorInputTextCommitted(const FText& Text, ETextCommit::Type CommitMethod)
188{
189 if (CommitMethod == ETextCommit::OnCleared)
190 {
191 if (ETB_FloorInput)
192 {
193 ETB_FloorInput->SetText(FText::AsNumber(GetCurrentFloor()));
194 }
195 return;
196 }
197
198 const FString Raw = Text.ToString().TrimStartAndEnd();
199 const int32 Typed = FCString::Atoi(*Raw);
200
201 if (Typed <= 0 || Raw.IsEmpty())
202 {
203 if (ETB_FloorInput)
204 {
205 ETB_FloorInput->SetText(FText::AsNumber(GetCurrentFloor()));
206 }
207
208 return;
209 }
210
211 SelectFloor(Typed);
212}
