155// Combines two limits, always taking the minimum of the two (with special handling for values of <= 0 meaning unlimited)
156float CombineFrameRateLimits(float Limit1, float Limit2)
157{
158 if (Limit1 <= 0.0f)
159 {
160 return Limit2;
161 }
162 else if (Limit2 <= 0.0f)
163 {
164 return Limit1;
165 }
166 else
167 {
168 return FMath::Min(Limit1, Limit2);
169 }

229void UDTSettingsLocal::SetPreferredMonitorIndex(int32 InMonitorIndex)
230{
231 PreferredMonitorIndex = InMonitorIndex;
232
233 // Move the game window to the desired monitor so the next fullscreen commitment lands on it
234 if (GEngine && GEngine->GameViewport)
235 {
236 if (TSharedPtr<SWindow> GameWindow = GEngine->GameViewport->GetWindow())
237 {
238 FDisplayMetrics DisplayMetrics;
239 FSlateApplication::Get().GetCachedDisplayMetrics(DisplayMetrics);
240
241 if (DisplayMetrics.MonitorInfo.IsValidIndex(PreferredMonitorIndex))

244 // Move the window to the top-left of the target monitor's work area so UE picks it up

543void UDTSettingsLocal::SetVolumeForSoundClass(USoundClass* InSoundClass, float InVolume)
544{
545 if (!InSoundClass || !GEngine)
546 {
547 return;
548 }
549
550 // Sanity clamp
551 InVolume = FMath::Clamp(InVolume, 0.0f, 1.0f);
552
553 if (UWorld* World = GEngine->GetCurrentPlayWorld())
554 {
555 if (FAudioDeviceHandle AudioDevice = World->GetAudioDevice())
556 {
557 if (!UserSettingsSoundMix)
558 {
559 if (const UDTAudioSettings* DTAudioSettings = GetDefault<UDTAudioSettings>())
560 {
561 if (UObject* ObjPath = DTAudioSettings->UserSettingsSoundMix.TryLoad())
562 {
563 UserSettingsSoundMix = Cast<USoundMix>(ObjPath);
564 }
565 }
566 }
567
568 if (UserSettingsSoundMix)
569 {
570 AudioDevice->SetSoundMixClassOverride(UserSettingsSoundMix, InSoundClass, InVolume, 1.0f, 0.1f, true);
571 }
572 }
573 }
