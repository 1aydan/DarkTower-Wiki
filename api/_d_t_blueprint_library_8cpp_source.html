138bool UDTBlueprintLibrary::IsActorFacingActor(AActor* SourceActor, AActor* TargetActor, float MaxAngleDegrees)
139{
140 if (!SourceActor || !TargetActor)
141 {
142 return false;
143 }
144
145 FVector SourceToTarget = (TargetActor->GetActorLocation() - SourceActor->GetActorLocation()).GetSafeNormal();
146 FVector SourceForward = SourceActor->GetActorForwardVector().GetSafeNormal();
147
148 float DotProduct = FVector::DotProduct(SourceForward, SourceToTarget);
149 float CosMaxAngle = FMath::Cos(FMath::DegreesToRadians(MaxAngleDegrees));
150
151 return DotProduct >= CosMaxAngle;

155{
156 OutLocation = InLocation;
157
158 UWorld* World = GEngine ? GEngine->GetWorldFromContextObject(WorldContextObject, EGetWorldErrorMode::LogAndReturnNull) : nullptr;
159 if (!World)
160 {
161 return false;

163
164 const FVector Start = InLocation + FVector(0.0f, 0.0f, FMath::Max(0.0f, TraceUp));

175 }
176

304bool UDTBlueprintLibrary::GetBestMatchingTag(const FGameplayTagContainer& TagContainer, const FGameplayTag& ParentTag, FGameplayTag& OutTag)
305{
306 OutTag = FGameplayTag::EmptyTag;
307
308 if (!ParentTag.IsValid() || TagContainer.IsEmpty())
309 {
310 return false;
311 }
312
313 int32 BestDepth = -1;
314
315 for (const FGameplayTag& Tag : TagContainer)
316 {
317 if (Tag.MatchesTag(ParentTag))
318 {
319 // Determine specificity by counting hierarchy separators in the tag name
320 const FString TagString = Tag.GetTagName().ToString();
321 int32 TagDepth = 0;
322 for (const TCHAR Ch : TagString)
323 {
324 if (Ch == TEXT('.'))
325 {
326 ++TagDepth;
327 }
328 }
329
330 if (TagDepth > BestDepth)
331 {
332 BestDepth = TagDepth;
333 OutTag = Tag;
334 }
335 }
336 }
337
338 return OutTag.IsValid();
339}

393FVector UDTBlueprintLibrary::GetSpreadDirection(const FVector& Forward, int32 Index, int32 Count, float TotalSpreadDegrees)
394{
395 const FVector NormalizedForward = Forward.GetSafeNormal();
396 if (Count <= 0)
397 {
398 return FVector::ZeroVector;
399 }
400
401 if (Count == 1)
402 {
403 return NormalizedForward;
404 }
405
406 const int32 ClampedIndex = FMath::Clamp(Index, 0, Count - 1);
407 const float ClampedSpreadDegrees = FMath::Clamp(TotalSpreadDegrees, -360.0f, 360.0f);
408 const bool bFullCircle = FMath::IsNearlyEqual(FMath::Abs(ClampedSpreadDegrees), 360.0f);
409
410 // A closed circle must not include both 0 and 360 degrees because they are the same direction.
411 const float Step = ClampedSpreadDegrees / (bFullCircle ? Count : Count - 1);
412 const float StartAngle = bFullCircle ? 0.0f : -ClampedSpreadDegrees * 0.5f;
413 const float Angle = StartAngle + (ClampedIndex * Step);
414
415 return NormalizedForward.RotateAngleAxis(Angle, FVector::UpVector);
416}

431EDTHitDirection UDTBlueprintLibrary::GetHitDirection(const AActor* SourceActor, const FVector& WorldHitDirection)
432{
433 if (!SourceActor)
434 {
435 return EDTHitDirection::Front;
436 }
437
438 // The montage direction is where the hit came FROM, so compare against the vector toward the attacker.
439 const FVector ToAttacker = (-WorldHitDirection).GetSafeNormal2D();
440 if (ToAttacker.IsNearlyZero())
441 {
442 return EDTHitDirection::Front;
443 }
444
445 const float ForwardDot = static_cast<float>(FVector::DotProduct(SourceActor->GetActorForwardVector().GetSafeNormal2D(), ToAttacker));
446 const float RightDot = static_cast<float>(FVector::DotProduct(SourceActor->GetActorRightVector().GetSafeNormal2D(), ToAttacker));
447
448 if (FMath::Abs(ForwardDot) >= FMath::Abs(RightDot))
449 {
450 return ForwardDot >= 0.0f ? EDTHitDirection::Front : EDTHitDirection::Back;
451 }
452
453 return RightDot >= 0.0f ? EDTHitDirection::Right : EDTHitDirection::Left;
454}
