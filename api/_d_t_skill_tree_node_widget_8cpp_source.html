26void UDTSkillTreeNodeWidget::TryInvestPoint()
27{
28 UDTSkillTreeComponent* SkillTree = GetSkillTreeComponent();
29 if (!SkillTree)
30 {
31 OnInvestFailed(FText::FromString(TEXT("No skill tree component found")));
32 return;
33 }
34
35 FText FailReason;
36 if (!SkillTree->CanInvestPoint(TreeTag, NodeDefinition.GetNodeTag(), FailReason))
37 {
38 OnInvestFailed(FailReason);
39 return;
40 }
41
42 if (SkillTree->InvestPoint(TreeTag, NodeDefinition.GetNodeTag()))
43 {
44 CurrentNodeRank++;
45 bIsMaxRank = CurrentNodeRank >= NodeDefinition.GetMaxRank();
46 OnNodeDataUpdated();
47 OnInvestSucceeded(CurrentNodeRank);
48 }
49}

94UDTAbilityDefinition* UDTSkillTreeNodeWidget::GetNodeAbilityDefinition() const
95{
96 // Prefer the node's authored ability definition, then fall back to the granted ability's own definition.
97 if (UDTAbilityDefinition* AuthoredDefinition = NodeDefinition.GetAbilityDefinition())
98 {
99 return AuthoredDefinition;
100 }
101
102 if (const TSubclassOf<UDTGameplayAbility> AbilityClass = NodeDefinition.GetGrantedAbilityClass())
103 {
104 if (const UDTGameplayAbility* AbilityCDO = AbilityClass->GetDefaultObject<UDTGameplayAbility>())
105 {
106 return AbilityCDO->GetAbilityDefinition();
107 }
108 }
109
110 return nullptr;
111}

118void UDTSkillTreeNodeWidget::PopulateAbilityTooltip(UDTAbilityTooltipWidget* InTooltipWidget) const
119{
120 if (!InTooltipWidget)
121 {
122 return;
123 }
124
125 // The granted ability class is the richest source: its CDO supplies the cooldown/cost GEs and live-scaled values
126 // that a bare definition cannot, so prefer it and only fall back to the authored definition for non-granting nodes.
127 if (const TSubclassOf<UDTGameplayAbility> AbilityClass = NodeDefinition.GetGrantedAbilityClass())
128 {
129 const int32 MaxRank = FMath::Max(NodeDefinition.GetMaxRank(), 1);
130 if (CurrentNodeRank > 0 && CurrentNodeRank < MaxRank)
131 {
132 // Invested with a rank remaining: preview current -> next.
133 InTooltipWidget->SetAbilityClassComparison(AbilityClass, CurrentNodeRank, CurrentNodeRank + 1);
134 }
135 else
136 {
137 // Not yet invested (preview rank 1) or already maxed (show current rank) — a single, non-arrow view.
138 const int32 DisplayRank = CurrentNodeRank > 0 ? CurrentNodeRank : 1;
139 InTooltipWidget->SetAbilityClassComparison(AbilityClass, DisplayRank, DisplayRank);
140 }
141 return;
142 }
143
144 if (UDTAbilityDefinition* AbilityDefinition = GetNodeAbilityDefinition())
145 {
146 InTooltipWidget->SetAbilityDefinition(AbilityDefinition);
147 }
148}

150UWidget* UDTSkillTreeNodeWidget::GetSkillNodeTooltipWidget()
151{
152 // Ability nodes get the rich ability tooltip; everything else gets the generic node tooltip. Both are rebuilt
153 // each hover so they always reflect the node's current rank and the player's live attributes.
154 if (IsAbilityNode() && AbilityTooltipWidgetClass)
155 {
156 if (UDTAbilityTooltipWidget* AbilityTooltip = CreateWidget<UDTAbilityTooltipWidget>(this, AbilityTooltipWidgetClass))
157 {
158 PopulateAbilityTooltip(AbilityTooltip);
159 return AbilityTooltip;
160 }
161 }
162
163 if (NodeTooltipWidgetClass)
164 {
165 if (UUserWidget* NodeTooltip = CreateWidget<UUserWidget>(this, NodeTooltipWidgetClass))
166 {
167 if (NodeTooltip->Implements<UDTSkillNodeTooltipInterface>())
168 {
169 IDTSkillNodeTooltipInterface::Execute_InitSkillNodeTooltip(NodeTooltip, this);
170 }
171 return NodeTooltip;
172 }
173 }
174
175 return nullptr;
176}

219FReply UDTSkillTreeNodeWidget::NativeOnMouseButtonDown(const FGeometry& InGeometry, const FPointerEvent& InMouseEvent)
220{
221 if (InMouseEvent.IsMouseButtonDown(EKeys::LeftMouseButton))
222 {
223 bLeftButtonPressed = true;
224 bDragStartedThisPress = false;
225
226 // Ability nodes arm drag detection; a press that never becomes a drag falls through to tap-to-invest on release.
227 if (CanDragAbility())
228 {
229 return UWidgetBlueprintLibrary::DetectDragIfPressed(InMouseEvent, this, EKeys::LeftMouseButton).NativeReply;
230 }
231
232 // Non-draggable nodes (passives) still capture the press so we receive the mouse-up and can invest.
233 return FReply::Handled();
234 }

239FReply UDTSkillTreeNodeWidget::NativeOnMouseButtonUp(const FGeometry& InGeometry, const FPointerEvent& InMouseEvent)
240{
241 if (InMouseEvent.GetEffectingButton() == EKeys::LeftMouseButton && bLeftButtonPressed)
242 {
243 const bool bWasTap = !bDragStartedThisPress;
244 bLeftButtonPressed = false;
245 bDragStartedThisPress = false;
246
247 // A press-and-release without a drag is a tap: invest a point (TryInvestPoint validates rank/points itself).
248 if (bWasTap)
249 {
250 TryInvestPoint();
251 return FReply::Handled();

253 }

258void UDTSkillTreeNodeWidget::NativeOnDragDetected(const FGeometry& InGeometry, const FPointerEvent& InMouseEvent, UDragDropOperation*& OutOperation)
259{
260 if (!CanDragAbility())
261 {
262 return;
263 }
264
265 // Mark this press as a drag so the following mouse-up is not treated as a tap-to-invest.
266 bDragStartedThisPress = true;
267
268 UDTAbilityDragPayload* Payload = NewObject<UDTAbilityDragPayload>();
269 Payload->AbilityHandle = GetGrantedAbilityHandle();
270 Payload->SourceSlotTag = FGameplayTag(); // Not from a slot
271 Payload->SourceWidget = const_cast<UDTSkillTreeNodeWidget*>(this);
272 Payload->bIsFromSlot = false;
273
274 UDragDropOperation* DragOp = UWidgetBlueprintLibrary::CreateDragDropOperation(UDragDropOperation::StaticClass());
275 DragOp->Payload = Payload;
276 DragOp->DefaultDragVisual = this;
277 OutOperation = DragOp;
278}
