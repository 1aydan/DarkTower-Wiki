22void UDTSkillTreeNodeWidget::TryInvestPoint()
23{
24 UDTSkillTreeComponent* SkillTree = GetSkillTreeComponent();
25 if (!SkillTree)
26 {
27 OnInvestFailed(FText::FromString(TEXT("No skill tree component found")));
28 return;
29 }
30
31 FText FailReason;
32 if (!SkillTree->CanInvestPoint(TreeTag, NodeDefinition.GetNodeTag(), FailReason))
33 {
34 OnInvestFailed(FailReason);
35 return;
36 }
37
38 if (SkillTree->InvestPoint(TreeTag, NodeDefinition.GetNodeTag()))
39 {
40 CurrentNodeRank++;
41 bIsMaxRank = CurrentNodeRank >= NodeDefinition.GetMaxRank();
42 OnNodeDataUpdated();
43 OnInvestSucceeded(CurrentNodeRank);
44 }
45}

85bool UDTSkillTreeNodeWidget::CanDragAbility() const
86{
