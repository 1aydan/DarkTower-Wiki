474 const FIntPoint NewGrid(
475 FMath::RoundToInt((NewTopLeft.X - CellPadding.X) / GridStep.X),
476 FMath::RoundToInt((NewTopLeft.Y - CellPadding.Y) / GridStep.Y));
477 ResetInteractionState();
478 PinnedEditor->SetNodeGridPosition(MovedTag, NewGrid);
479 return FReply::Handled().ReleaseMouseCapture();
480 }

482 if (bLinkDragActive && PinnedEditor.IsValid())

486 const FGameplayTag TargetTag = PinnedEditor->GetNodeTagAtGridPosition(GraphToGridCell(GraphPos, PinnedEditor->GetGridStep()));
487 ResetInteractionState();

492 return FReply::Handled().ReleaseMouseCapture();
493 }
494
495 if (bMoveArmed)
496 {
497 ResetInteractionState();
498 return FReply::Handled().ReleaseMouseCapture();
499 }
500 }

626 void FrameContent(const FVector2D& ViewSize)
627 {
628 const TSharedPtr<FDTSkillTreeAssetEditor> PinnedEditor = AssetEditor.Pin();
629 const UDTSkillTreeDefinition* Def = Definition();
630 if (!PinnedEditor.IsValid() || !Def)
631 {
632 ZoomAmount = 1.0f;
633 ViewOffset = FVector2D(-FrameMargin, -FrameMargin);
634 return;
635 }
636
637 const FVector2D NodeSize = PinnedEditor->GetNodeSize();
638 const TSharedRef<FDTSkillTreeAssetEditor> EditorRef = PinnedEditor.ToSharedRef();
639
640 TArray<FSkillTreeNode> Nodes;
641 Def->GetAllNodes(Nodes);
642
643 bool bHasBounds = false;
644 FVector2D Min(0.0f), Max(0.0f);
645 for (const FSkillTreeNode& Node : Nodes)
646 {
647 const FVector2D TopLeft = GetNodeGraphTopLeft(EditorRef, Node.GetNodeTag());
648 const FVector2D BottomRight = TopLeft + NodeSize;
649 if (!bHasBounds)
650 {
651 Min = TopLeft;
652 Max = BottomRight;
653 bHasBounds = true;
654 continue;
655 }
656 Min.X = FMath::Min(Min.X, TopLeft.X);
657 Min.Y = FMath::Min(Min.Y, TopLeft.Y);
658 Max.X = FMath::Max(Max.X, BottomRight.X);
659 Max.Y = FMath::Max(Max.Y, BottomRight.Y);
660 }
661
662 if (!bHasBounds)
663 {
664 ZoomAmount = 1.0f;
665 ViewOffset = FVector2D(-FrameMargin, -FrameMargin);
666 return;
667 }
668
669 Min -= FVector2D(FrameMargin, FrameMargin);
670 Max += FVector2D(FrameMargin, FrameMargin);
671 const FVector2D ContentSize = (Max - Min).ComponentMax(FVector2D(1.0f, 1.0f));
672
673 ZoomAmount = FMath::Clamp(FMath::Min(ViewSize.X / ContentSize.X, ViewSize.Y / ContentSize.Y), MinZoom, 1.0f);
674 const FVector2D ContentCenter = (Min + Max) * 0.5f;
675 ViewOffset = ContentCenter - (ViewSize / ZoomAmount) * 0.5f;
676 }
