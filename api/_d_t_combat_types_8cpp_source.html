15void FDTCombatAccumulator::ApplyModifier(const FDTCombatModifierSpec& Spec, FDTDamageResult& Result)
16{
17 if (Spec.Operation == EDTCombatModifierOp::IncreasedPct)

19 IncreasedPct += Spec.Magnitude;

23 FDTCombatModifierRecord Record;
24 Record.SourceTag = Spec.SourceTag;
25 Record.Phase = Spec.Phase;
26 Record.Operation = Spec.Operation;
27 Record.Magnitude = Spec.Magnitude;
28 Record.DamageBefore = Damage;
29
30 switch (Spec.Operation)
31 {
32 case EDTCombatModifierOp::AddFlat:
33 Damage += Spec.Magnitude;
34 break;
35 case EDTCombatModifierOp::MoreMultiplier:
36 Damage *= Spec.Magnitude;
37 break;
38 case EDTCombatModifierOp::LessMultiplier:
39 Damage *= FMath::Clamp(Spec.Magnitude, 0.0f, 1.0f);
40 break;
41 case EDTCombatModifierOp::Override:
42 Damage = Spec.Magnitude;
43 bHasOverride = true;
44 OverrideValue = Spec.Magnitude;
45 break;
46 case EDTCombatModifierOp::Penetration:
47 MitigationPenetration += Spec.Magnitude;
48 break;
49 case EDTCombatModifierOp::Reduction:
50 MitigationReduction += Spec.Magnitude;
51 break;
52 default:
53 break;
54 }
55
56 Record.DamageAfter = Damage;
57 Result.AppliedModifiers.Add(Record);
58}

60void FDTCombatAccumulator::FlushAdditiveBucket(FDTDamageResult& Result, EDTCombatModifierPhase Phase)
61{
