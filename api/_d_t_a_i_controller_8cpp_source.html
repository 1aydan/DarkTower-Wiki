135void ADTAIController::UpdateBehaviorFromSignificance(ESignificanceLevel SignificanceLevel)
136{
137 if (SignificanceLevel <= AIDisableThreshold)
138 {
139 if (StateTreeAIComponent && StateTreeAIComponent->IsRunning())
140 {
141 StateTreeAIComponent->StopLogic("Low Significance");
142 }
143 }
144 else
145 {
146 if (StateTreeAIComponent && !StateTreeAIComponent->IsRunning())
147 {
148 StateTreeAIComponent->StartLogic();
149 }
150 }
151
152 if (PerceptionComponent)
153 {
154 const bool bShouldEnablePerception = SignificanceLevel > PerceptionDisableThreshold;
155 PerceptionComponent->SetSenseEnabled(UAISense_Sight::StaticClass(), bShouldEnablePerception);
156 }
157
158 AIUpdateInterval = AIUpdateIntervals[SignificanceLevel];
159 SetActorTickInterval(AIUpdateInterval);
160}

198void ADTAIController::OnTargetPerceptionUpdated(AActor* Actor, FAIStimulus Stimulus)
199{
200 if (!Actor || !AICharacter)
201 {
202 UE_LOG(LogDTAI, Verbose, TEXT("OnTargetPerceptionUpdated called with null Actor or Owner"));
203 return;
204 }
205
206 IDTTargetableInterface* TargetActor = Cast<IDTTargetableInterface>(Actor);
207 if (!TargetActor)
208 {
209 UE_LOG(LogDTAI, Verbose, TEXT("OnTargetPerceptionUpdated called for actor that does not implement IDTTargetableInterface: %s"), *Actor->GetName());
210 return;
211 }
212
213 if (TargetActor->Execute_IsDead(Actor))
214 {
215 UE_LOG(LogDTAI, VeryVerbose, TEXT("OnTargetPerceptionUpdated called for dead actor: %s"), *Actor->GetName());
216 return;
217 }
218
219 if (Stimulus.Type == UAISense::GetSenseID<UAISense_Sight>())
220 {
221 HandleSightUpdate(Actor, Stimulus);
222 }
223 else if (Stimulus.Type == UAISense::GetSenseID<UAISense_Damage>())
224 {
225 HandleDamageUpdate(Actor, Stimulus);
226 }
227 else if (Stimulus.Type == UAISense::GetSenseID<UAISense_Team>())
228 {
229 HandleTeamUpdate(Actor, Stimulus);
230 }
231 else
232 {
233 UE_LOG(LogDTAI, Warning, TEXT("Unhandled perception update for actor: %s"), *Actor->GetName());
234 }
235}
