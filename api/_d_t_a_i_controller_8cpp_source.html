258{
259 if (!Target || !AICharacter)

263
264 const FDTAITargetBroadcastConfig& BroadcastConfig = AICharacter->GetTargetBroadcastConfig();
265 if (!BroadcastConfig.bBroadcastTargetToAllies || BroadcastConfig.BroadcastRadius <= 0.0f)

292{

329void ADTAIController::HandleTeamUpdate(AActor* Actor, FAIStimulus Stimulus)
330{
331 if (!TargetingComponent || !AICharacter)
332 {
333 UE_LOG(LogDTAI, Warning, TEXT("ADTAIController::HandleTeamUpdate - called without TargetingComponent or AICharacter"));
334 return;
335 }
336
337 if (!Stimulus.WasSuccessfullySensed() || Actor == AICharacter)
338 {
339 return;
340 }
341
342 // If we have a target already, ignore this event
343 if (TargetingComponent->HasTarget())
344 {
345 return;
346 }
347
348 const FDTAITargetBroadcastConfig& BroadcastConfig = AICharacter->GetTargetBroadcastConfig();
349 if (!BroadcastConfig.bAcceptAllyTargetBroadcasts)
350 {
351 return;
352 }
353
354 PerceivedActors.AddUnique(Actor);
355
356 // Ally-broadcast targets join the threat table so they compete once threat is in play
357 if (ThreatComponent && AICharacter->GetThreatConfig().bEnableThreat && !ThreatComponent->HasEntryFor(Actor))
358 {
359 ThreatComponent->AddThreat(Actor, AICharacter->GetThreatConfig().SightAggroThreat);
360 }
361
362 TargetingComponent->SetCurrentTarget(Actor);
363}

365void ADTAIController::OnTargetPerceptionUpdated(AActor* Actor, FAIStimulus Stimulus)
366{
367 if (!Actor || !AICharacter)
368 {
369 UE_LOG(LogDTAI, Verbose, TEXT("OnTargetPerceptionUpdated called with null Actor or Owner"));
370 return;
371 }
372
373 IDTTargetableInterface* TargetActor = Cast<IDTTargetableInterface>(Actor);
374 if (!TargetActor)
375 {
376 UE_LOG(LogDTAI, Verbose, TEXT("OnTargetPerceptionUpdated called for actor that does not implement IDTTargetableInterface: %s"), *Actor->GetName());
377 return;
378 }
379
380 if (TargetActor->Execute_IsDead(Actor))
381 {
382 UE_LOG(LogDTAI, VeryVerbose, TEXT("OnTargetPerceptionUpdated called for dead actor: %s"), *Actor->GetName());
383 return;
384 }
385
386 if (Stimulus.Type == UAISense::GetSenseID<UAISense_Sight>())
387 {
388 HandleSightUpdate(Actor, Stimulus);
389 }
390 else if (Stimulus.Type == UAISense::GetSenseID<UAISense_Damage>())
391 {
392 HandleDamageUpdate(Actor, Stimulus);
393 }
394 else if (Stimulus.Type == UAISense::GetSenseID<UAISense_Team>())
395 {
396 HandleTeamUpdate(Actor, Stimulus);
397 }
398 else
399 {
400 UE_LOG(LogDTAI, Warning, TEXT("Unhandled perception update for actor: %s"), *Actor->GetName());
401 }

403void ADTAIController::UpdateTargetSelection()
404{
405 if (!TargetingComponent)
406 {
407 UE_LOG(LogDTAI, Warning, TEXT("UpdateTargetSelection called without TargetingComponent or TargetSelectionStrategy"));
408 return;
409 }
410
411 if (PerceivedActors.IsEmpty())
412 {
413 TargetingComponent->SetCurrentTarget(nullptr);
414 return;
415 }
416
417 // Threat owns targeting while entries exist; preset filtering below remains the fallback
418 if (ThreatComponent && AICharacter && AICharacter->GetThreatConfig().bEnableThreat && ThreatComponent->HasThreatEntries())
419 {
420 AActor* Desired = ThreatComponent->SelectTargetByThreat(TargetingComponent->GetCurrentTarget());
421 if (Desired)
422 {
423 TargetingComponent->SetCurrentTarget(Desired);
424 return;
425 }
426 }
427
428 if (!TargetingComponent->HasTarget())
429 {
430 TargetingComponent->ExecuteAsyncFilteringRequest(PerceivedActors, nullptr, true);
431 }
