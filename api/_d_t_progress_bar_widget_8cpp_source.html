74void UDTProgressBarWidget::SetProgressByValue(const float NewCurrentValue, const float NewMaxValue)
75{
76 MaxValue = FMath::Max(0.0f, NewMaxValue);
77
78 if (MaxValue <= 0.0f)
79 {
80 CurrentValue = 0.0f;
81 Percent = 0.0f;
82 }
83 else
84 {
85 CurrentValue = FMath::Clamp(NewCurrentValue, 0.0f, MaxValue);
86 Percent = FMath::Clamp(CurrentValue / MaxValue, 0.0f, 1.0f);
87 }

139FString UDTProgressBarWidget::FormatValue(const float Value) const
140{
141 if (NumberFormat == EDTProgressBarNumberFormat::CommaSeparated)
142 {
143 return UDamageNumberBlueprintLibrary::FormatNumber(Value);
144 }
145
146 if (NumberFormat == EDTProgressBarNumberFormat::Abbreviated)
147 {
148 const int32 ClampedDecimalPlaces = FMath::Clamp(DecimalPlaces, 0, 3);
149 return UDamageNumberBlueprintLibrary::FormatNumberAbbreviated(Value, ClampedDecimalPlaces);
150 }
151
152 const int32 ClampedDecimalPlaces = FMath::Clamp(DecimalPlaces, 0, 3);
153 switch (ClampedDecimalPlaces)
154 {
155 case 0:
156 return FString::Printf(TEXT("%d"), FMath::FloorToInt32(Value));
157 case 1:
158 return FString::Printf(TEXT("%.1f"), Value);
159 case 2:
160 return FString::Printf(TEXT("%.2f"), Value);
161 default:
162 return FString::Printf(TEXT("%.3f"), Value);
163 }
