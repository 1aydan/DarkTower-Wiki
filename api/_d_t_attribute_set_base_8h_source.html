136 void GetSourceAndTargetFromContext(const FGameplayEffectModCallbackData& Data, TReturnType*& SourceActor, TReturnType*& TargetActor)
137 {
138 const FGameplayEffectContextHandle Context = Data.EffectSpec.GetContext();
139 UAbilitySystemComponent* Source = Context.GetOriginalInstigatorAbilitySystemComponent();
140
141 // Get the Target actor, which should be our owner
142 if (Data.Target.AbilityActorInfo.IsValid() && Data.Target.AbilityActorInfo->AvatarActor.IsValid())
143 {
144 TargetActor = Cast<TReturnType>(Data.Target.AbilityActorInfo->AvatarActor.Get());
145 }
146
147 // Get the Source actor, which should be the damage causer (instigator)
148 if (Source && Source->AbilityActorInfo.IsValid() && Source->AbilityActorInfo->AvatarActor.IsValid())
149 {
150 if (Context.GetEffectCauser())
151 {
152 SourceActor = Cast<TReturnType>(Context.GetEffectCauser());
153 }
154 else
155 {
156 SourceActor = Cast<TReturnType>(Source->AbilityActorInfo->AvatarActor.Get());
157 }
158 }
