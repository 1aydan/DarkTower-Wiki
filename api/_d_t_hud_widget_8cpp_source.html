279FText UDTHudWidget::BuildDungeonSettingsText(const FDTDungeonSettings& DungeonSettings, const UWorld* World)
280{
281 FText ModeLabel;
282 if (DungeonSettings.ModeDefinitionGuid.IsValid())
283 {
284 const UDTGameModeDefinition* Definition = UDefinitionAssetManager::Get().LoadDefinitionByGuid<UDTGameModeDefinition>(DungeonSettings.ModeDefinitionGuid);
285 if (Definition && !Definition->DisplayName.IsEmpty())
286 {
287 ModeLabel = Definition->DisplayName;
288 }
289 else if (DungeonSettings.ModeDefinitionTag.IsValid())
290 {
291 const FString TagStr = DungeonSettings.ModeDefinitionTag.GetTagName().ToString();
292 int32 DotIdx = INDEX_NONE;
293 TagStr.FindLastChar(TEXT('.'), DotIdx);
294 ModeLabel = FText::FromString(FName::NameToDisplayString(DotIdx != INDEX_NONE ? TagStr.Mid(DotIdx + 1) : TagStr, false));
295 }
296 }
297
298 if (ModeLabel.IsEmpty())
299 {
300 ModeLabel = NSLOCTEXT("DarkTower", "UnknownMode", "Unknown");
301 }
302
303 const FText MapLabel = !DungeonSettings.FloorDisplayName.IsEmpty() ? DungeonSettings.FloorDisplayName : FText::FromString(GetMapDisplayName(DungeonSettings.TravelMapName, nullptr));
304 return FText::Format(NSLOCTEXT("DarkTower", "DungeonSettingsFormat", "{0} : {1} - Floor {2}{3}"), ModeLabel, MapLabel, FText::AsNumber(FMath::Max(1, DungeonSettings.GlobalFloor)), DungeonSettings.bMortalMode ? FText::FromString(TEXT("Mortal")): FText::GetEmpty());
305}
