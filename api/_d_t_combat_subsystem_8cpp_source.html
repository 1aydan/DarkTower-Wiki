59bool UDTCombatSubsystem::CalculateDamage(const FDTDamageContext& Context, FDTDamageResult& OutResult) const
60{
61 TRACE_CPUPROFILER_EVENT_SCOPE(UDTCombatSubsystem::CalculateDamage);
62
63 OutResult = FDTDamageResult();
64 OutResult.BaseDamage = Context.BaseDamage;
65
66 if (!Context.SourceActor || !Context.TargetActor)
67 {
68 UE_LOG(LogDTCombat, Verbose, TEXT("CalculateDamage failed: invalid source or target."));
69 return false;
70 }
71
72 if (!Context.SourceActor->HasAuthority())
73 {
74 UE_LOG(LogDTCombat, Verbose, TEXT("CalculateDamage rejected on non-authority source %s."), *GetNameSafe(Context.SourceActor.Get()));
75 return false;
76 }
77
78 UDTAbilitySystemComponent* SourceASC = Cast<UDTAbilitySystemComponent>(UAbilitySystemBlueprintLibrary::GetAbilitySystemComponent(Context.SourceActor));
79 UDTAbilitySystemComponent* TargetASC = Cast<UDTAbilitySystemComponent>(UAbilitySystemBlueprintLibrary::GetAbilitySystemComponent(Context.TargetActor));
80 if (!SourceASC || !TargetASC)
81 {
82 UE_LOG(LogDTCombat, Verbose, TEXT("CalculateDamage failed: source or target ASC missing."));
83 return false;
84 }
85
86 FDTDamageContext DamageContext = Context;
87 UpdateContextTags(DamageContext, SourceASC, TargetASC);
88
89 FGameplayEffectContextHandle EventContext = SourceASC->MakeEffectContext();
90 EventContext.AddInstigator(DamageContext.SourceActor.Get(), DamageContext.EffectCauser.Get());
91 if (DamageContext.HitResult.HasValidHitObjectHandle() || DamageContext.HitResult.bBlockingHit)
92 {
93 EventContext.AddHitResult(DamageContext.HitResult, true);
94 }
95
96 BroadcastDamageEvent(DamageContext, SourceASC, TargetASC, DTGameplayTags::TAG_EVENT_DAMAGE_PRERESOLVE, DamageContext.BaseDamage, DamageContext.ContextTags, EventContext);
97
98 TArray<FDTCombatModifierSpec> Modifiers;
99 GatherModifiers(DamageContext, SourceASC, TargetASC, Modifiers);
100 SortModifiers(Modifiers);
101
102 FDTCombatAccumulator Accumulator;
103 Accumulator.Reset(FMath::Max(0.0f, DamageContext.BaseDamage));
104
105 TArray<FDTCombatModifierSpec> ConsumedSpecs;
106
107 ApplyModifierPhase(DamageContext, EDTCombatModifierPhase::Base, Modifiers, Accumulator, OutResult, ConsumedSpecs);
108 ApplyModifierPhase(DamageContext, EDTCombatModifierPhase::Outgoing, Modifiers, Accumulator, OutResult, ConsumedSpecs);
109 OutResult.PreMitigationDamage = Accumulator.Damage;
110
111 ApplyCrit(DamageContext, Accumulator, OutResult);
112 ApplyModifierPhase(DamageContext, EDTCombatModifierPhase::Crit, Modifiers, Accumulator, OutResult, ConsumedSpecs);
113 ApplyModifierPhase(DamageContext, EDTCombatModifierPhase::Incoming, Modifiers, Accumulator, OutResult, ConsumedSpecs);
114 ApplyModifierPhase(DamageContext, EDTCombatModifierPhase::Mitigation, Modifiers, Accumulator, OutResult, ConsumedSpecs);
115
116 if (!DamageContext.bSkipBuiltInMitigation)
117 {
118 ApplyBlock(DamageContext, TargetASC, Accumulator, OutResult);
119 ApplyMitigation(DamageContext, SourceASC, TargetASC, Accumulator, OutResult);
120 }
121
122 OutResult.MitigatedDamage = Accumulator.Damage;
123
124 ApplyModifierPhase(DamageContext, EDTCombatModifierPhase::Final, Modifiers, Accumulator, OutResult, ConsumedSpecs);
125
126 // A hit that carried real damage always lands for at least 1, so heavy mitigation or a block doesn't round
127 // down to 0 and read as "nothing happened" to the player. Hits with no pre-mitigation damage stay at 0.
128 const float RoundedDamage = FMath::RoundHalfToEven(Accumulator.Damage);
129 const float MinDamage = OutResult.PreMitigationDamage > 0.0f ? 1.0f : 0.0f;
130 OutResult.FinalDamage = FMath::Max(MinDamage, RoundedDamage);
131 OutResult.ResultTags.AddTag(DTGameplayTags::TAG_EFFECT_RESULT_COMBAT);
132
133 ApplyModifierPhase(DamageContext, EDTCombatModifierPhase::Proc, Modifiers, Accumulator, OutResult, ConsumedSpecs);
134
135 for (const FDTCombatModifierSpec& ConsumedSpec : ConsumedSpecs)
136 {
137 UDTAbilitySystemComponent* OwningASC = ConsumedSpec.OwningComponent.Get();
138 if (OwningASC)
139 {
140 FDTCombatModifierHandle Handle;
141 Handle.Id = ConsumedSpec.HandleId;
142 OwningASC->UnregisterCombatModifier(Handle);
143 }
144 }
145
146 FGameplayTagContainer EventTags = DamageContext.ContextTags;
147 EventTags.AppendTags(OutResult.ResultTags);
148
149 BroadcastDamageEvent(DamageContext, SourceASC, TargetASC, DTGameplayTags::TAG_EVENT_DAMAGE_RESOLVED, OutResult.FinalDamage, EventTags, EventContext);
150
151 return !OutResult.bCancelled;
152}

154bool UDTCombatSubsystem::ApplyDamage(const FDTDamageContext& Context, TSubclassOf<UGameplayEffect> DamageEffectClass, FActiveGameplayEffectHandle& OutEffectHandle, FDTDamageResult* OutDamageResult, FGameplayEffectContextHandle EffectContext) const
155{
156 TRACE_CPUPROFILER_EVENT_SCOPE(UDTCombatSubsystem::ApplyDamage);
157
158 OutEffectHandle = FActiveGameplayEffectHandle();
159 if (OutDamageResult)
160 {
161 *OutDamageResult = FDTDamageResult();
162 }
163
164 if (!Context.SourceActor || !Context.TargetActor)
165 {
166 return false;
167 }
168
169 UDTAbilitySystemComponent* SourceASC = Cast<UDTAbilitySystemComponent>(UAbilitySystemBlueprintLibrary::GetAbilitySystemComponent(Context.SourceActor));
170 UAbilitySystemComponent* TargetASC = UAbilitySystemBlueprintLibrary::GetAbilitySystemComponent(Context.TargetActor);
171 if (!SourceASC || !TargetASC)
172 {
173 return false;
174 }
175
176 if (!DamageEffectClass)
177 {
178 const UDTGameSettings* GameSettings = GetDefault<UDTGameSettings>();
179 DamageEffectClass = GameSettings ? GameSettings->GetDefaultDamageEffectClass() : nullptr;
180 if (!DamageEffectClass)
181 {
182 UE_LOG(LogDTCombat, Error, TEXT("ApplyDamage failed: no damage effect class configured."));
183 return false;
184 }
185 }
186
187 FDTDamageContext DamageContext = Context;
188 if (const UGameplayEffect* DamageEffectCDO = DamageEffectClass ? DamageEffectClass->GetDefaultObject<UGameplayEffect>() : nullptr)
189 {
190 const FGameplayTagContainer& DamageEffectTags = DamageEffectCDO->GetAssetTags();
191 DamageContext.ContextTags.AppendTags(DamageEffectTags);
192 }
193
194 FDTDamageResult Result;
195 const bool bDamageResultValid = CalculateDamage(DamageContext, Result);
196 if (OutDamageResult)
197 {
198 *OutDamageResult = Result;
199 }
200
201 if (!bDamageResultValid || Result.bCancelled)
202 {
203 return false;
204 }
205
206 FGameplayEffectContextHandle ContextHandle = EffectContext.IsValid() ? EffectContext : SourceASC->MakeEffectContext();
207 ContextHandle.AddInstigator(DamageContext.SourceActor.Get(), DamageContext.EffectCauser ? DamageContext.EffectCauser.Get() : DamageContext.SourceActor.Get());
208 if (DamageContext.EffectCauser)
209 {
210 ContextHandle.AddSourceObject(DamageContext.EffectCauser.Get());
211 }
212
213 if (DamageContext.HitResult.HasValidHitObjectHandle() || DamageContext.HitResult.bBlockingHit || !DamageContext.HitResult.Location.IsNearlyZero())
214 {
215 ContextHandle.AddHitResult(DamageContext.HitResult, true);
216 ContextHandle.AddOrigin(!DamageContext.HitResult.ImpactPoint.IsNearlyZero() ? DamageContext.HitResult.ImpactPoint : DamageContext.HitResult.Location);
217 }
218
219 FGameplayTagContainer ContextTags = DamageContext.ContextTags;
220 if (DamageContext.DamageType.IsValid())
221 {
222 ContextTags.AddTag(DamageContext.DamageType);
223 }
224
225 ContextTags.AppendTags(Result.ResultTags);
226 FDTGameplayEffectContext* DTContext = UDTBlueprintLibrary::GetDTEffectContextMutable(ContextHandle);
227 if (DTContext)
228 {
229 DTContext->AppendContextTags(ContextTags);
230 DTContext->SetKnockbackMagnitude(DamageContext.KnockbackMagnitude);
231 }
232
233 FGameplayEffectSpecHandle SpecHandle = SourceASC->MakeOutgoingSpec(DamageEffectClass, 1.0f, ContextHandle);
234 if (!SpecHandle.IsValid())
235 {
236 return false;
237 }
238
239 FGameplayEffectSpec* Spec = SpecHandle.Data.Get();
240 Spec->SetSetByCallerMagnitude(DTGameplayTags::TAG_SETBYCALLER_DATA_ATTRIBUTE_DAMAGE, Result.FinalDamage);
241
242 if (DamageContext.DamageType.IsValid())
243 {
244 Spec->AddDynamicAssetTag(DamageContext.DamageType);
245 }
246
247 Spec->AddDynamicAssetTag(DTGameplayTags::TAG_EFFECT_RESULT_COMBAT);
248
249 OutEffectHandle = SourceASC->ApplyGameplayEffectSpecToTarget(*Spec, TargetASC);
250
251 // Restore flat health to the blocker on a successful block (HealthOnBlock). Clamped to MaxHealth by
252 // UDTAttributeSet::PreAttributeChange, so it never overheals.
253 if (Result.BlockHealAmount > 0.0f)
254 {
255 TargetASC->ApplyModToAttribute(UDTAttributeSet::GetHealthAttribute(), EGameplayModOp::Additive, Result.BlockHealAmount);
256 }
257
258 const FVector TargetLocation = DamageContext.TargetActor->GetActorLocation();
259 const FVector HitLocation = DamageContext.HitResult.Location.IsNearlyZero() ? TargetLocation : FVector(DamageContext.HitResult.Location);
260
261 UAISense_Damage::ReportDamageEvent(DamageContext.SourceActor.Get(), DamageContext.TargetActor.Get(), DamageContext.SourceActor.Get(), Result.FinalDamage, TargetLocation, HitLocation, NAME_None);
262
263 if (!DamageContext.bSuppressHitReact)
264 {
265 UDTAbilitySystemComponent* TargetDTASC = Cast<UDTAbilitySystemComponent>(TargetASC);
266 if (TargetDTASC && !TargetDTASC->HasMatchingGameplayTag(DTGameplayTags::TAG_STATUS_DEAD))
267 {
268 FGameplayEventData Payload = TargetDTASC->BuildAbilityEventPayload(DTGameplayTags::TAG_EVENT_CHARACTER_HITREACT, DamageContext.SourceActor.Get(), DamageContext.TargetActor.Get(), DamageContext.EffectCauser.Get(), DamageContext.HitReactMagnitude, FGameplayTagContainer(), DamageContext.HitResult, ContextHandle.Duplicate());
269 TargetDTASC->SendAbilityEvent(DTGameplayTags::TAG_EVENT_CHARACTER_HITREACT, Payload);
270 }
271 }
272
273 return OutEffectHandle.IsValid();
274}

379void UDTCombatSubsystem::ApplyModifierPhase(const FDTDamageContext& Context, EDTCombatModifierPhase Phase, const TArray<FDTCombatModifierSpec>& Modifiers, FDTCombatAccumulator& Accumulator, FDTDamageResult& Result, TArray<FDTCombatModifierSpec>& OutConsumedSpecs)
380{
381 TRACE_CPUPROFILER_EVENT_SCOPE(UDTCombatSubsystem::ApplyModifierPhase);
382
383 for (FDTCombatModifierSpec Spec : Modifiers)
384 {
385 if (Spec.Phase != Phase)
386 {
387 continue;
388 }
389
390 if (Spec.Definition)
391 {
392 if (!Spec.Definition->Matches(Context, Result))
393 {
394 continue;
395 }
396
397 if (Spec.ApplicationChance < 1.0f && FMath::FRand() > Spec.ApplicationChance)
398 {
399 continue;
400 }
401
402 for (const TObjectPtr<UDTCombatModifierRule>& Rule : Spec.Definition->Rules)
403 {
404 if (Rule)
405 {
406 Rule->Apply(Context, Spec);
407 }
408 }
409 }
410
411 Accumulator.ApplyModifier(Spec, Result);
412
413 if (Spec.bConsumeOnApply)
414 {
415 OutConsumedSpecs.Add(Spec);
416 }
417 }
418
419 Accumulator.FlushAdditiveBucket(Result, Phase);
420}

423{
424 if (!Context.ContextTags.HasTag(DTGameplayTags::TAG_EFFECT_CONDITION_CANCRIT) || !Context.SourceActor)
425 {
426 return;
427 }

454void UDTCombatSubsystem::ApplyMitigation(const FDTDamageContext& Context, UDTAbilitySystemComponent* SourceASC, UDTAbilitySystemComponent* TargetASC, FDTCombatAccumulator& Accumulator, FDTDamageResult& Result) const
455{
456 if (!TargetASC || Context.ContextTags.HasTag(DTGameplayTags::TAG_DAMAGE_TYPE_TRUE))
457 {
458 return;
459 }
460
461 const float DamageBefore = Accumulator.Damage;
462 float Armor = TargetASC->GetNumericAttribute(UDTAttributeSet::GetArmorAttribute());
463 const float PhysicalPenetration = SourceASC ? SourceASC->GetNumericAttribute(UDTAttributeSet::GetPhysicalPenetrationAttribute()) : 0.0f;
464 Armor *= 1.0f - FMath::Clamp(PhysicalPenetration + Accumulator.MitigationPenetration, 0.0f, 1.0f);
465
466 const float ArmorConstant = GetArmorConstant();
467 const float ArmorExponent = GetArmorExponent();
468 float ArmorReduction = 0.0f;
469 if (Armor > 0.0f)
470 {
471 const float ArmorPow = FMath::Pow(Armor, ArmorExponent);
472 const float ConstantPow = FMath::Pow(ArmorConstant, ArmorExponent);
473 ArmorReduction = ArmorPow / (ArmorPow + ConstantPow);
474 }
475 const float DamageResistance = FMath::Clamp(TargetASC->GetNumericAttribute(UDTAttributeSet::GetDamageResistanceAttribute()) + Accumulator.MitigationReduction, -10.0f, 0.75f);
476 Accumulator.Damage *= (1.0f - ArmorReduction) * (1.0f - DamageResistance);
477
478 FDTCombatModifierRecord Record;
479 Record.Phase = EDTCombatModifierPhase::Mitigation;
480 Record.Operation = EDTCombatModifierOp::Reduction;
481 Record.Magnitude = DamageBefore - Accumulator.Damage;
482 Record.DamageBefore = DamageBefore;
483 Record.DamageAfter = Accumulator.Damage;
484 Result.AppliedModifiers.Add(Record);
485}

487void UDTCombatSubsystem::ApplyBlock(const FDTDamageContext& Context, UDTAbilitySystemComponent* TargetASC, FDTCombatAccumulator& Accumulator, FDTDamageResult& Result) const
488{
489 // True damage bypasses block, the same as armor/resistance mitigation.
490 if (!TargetASC || Context.ContextTags.HasTag(DTGameplayTags::TAG_DAMAGE_TYPE_TRUE))
491 {
492 return;
493 }
494
495 // Gate: only targets flagged as able to block roll block. A shield (or block ability) grants
496 // Effect.Condition.CanBlock to the wearer's ASC while equipped/active; without it, no block.
497 if (!TargetASC->HasMatchingGameplayTag(DTGameplayTags::TAG_EFFECT_CONDITION_CANBLOCK))
498 {
499 return;
500 }
501
502 // Passive block: roll BlockChance on every incoming hit while able to block.
503 const float BlockChance = FMath::Clamp(TargetASC->GetNumericAttribute(UDTAttributeSet::GetBlockChanceAttribute()), 0.0f, 1.0f);
504 if (BlockChance <= 0.0f || FMath::FRand() > BlockChance)
505 {
506 return;
507 }
508
509 // Successful block. Reduction = min(BaseBlockEffectiveness * (1 + BlockEffectiveness/100), MaxBlockDamageReduction),
510 // where BlockEffectiveness is percent-points (e.g. 50 => the base is scaled by 1.5).
511 const float BlockEffectiveness = FMath::Max(0.0f, TargetASC->GetNumericAttribute(UDTAttributeSet::GetBlockEffectivenessAttribute()));
512 const float Reduction = FMath::Min(GetBaseBlockEffectiveness() * (1.0f + BlockEffectiveness / 100.0f), GetMaxBlockDamageReduction());
513
514 const float DamageBefore = Accumulator.Damage;
515 Accumulator.Damage *= (1.0f - Reduction);
516
517 FDTCombatModifierRecord Record;
518 Record.Phase = EDTCombatModifierPhase::Mitigation;
519 Record.Operation = EDTCombatModifierOp::Reduction;
520 Record.Magnitude = DamageBefore - Accumulator.Damage;
521 Record.DamageBefore = DamageBefore;
522 Record.DamageAfter = Accumulator.Damage;
523 Result.AppliedModifiers.Add(Record);
524
525 Result.ResultTags.AddTag(DTGameplayTags::TAG_EFFECT_RESULT_BLOCK);
526
527 // Flat health restored to the blocker (the target) on a successful block; applied in ApplyDamage.
528 Result.BlockHealAmount = FMath::Max(0.0f, TargetASC->GetNumericAttribute(UDTAttributeSet::GetHealthOnBlockAttribute()));
529}
