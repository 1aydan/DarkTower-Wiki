57bool UDTCombatSubsystem::CalculateDamage(const FDTDamageContext& Context, FDTDamageResult& OutResult) const
58{
59 TRACE_CPUPROFILER_EVENT_SCOPE(UDTCombatSubsystem::CalculateDamage);
60
61 OutResult = FDTDamageResult();
62 OutResult.BaseDamage = Context.BaseDamage;
63
64 if (!Context.SourceActor || !Context.TargetActor)
65 {
66 UE_LOG(LogDTCombat, Verbose, TEXT("CalculateDamage failed: invalid source or target."));
67 return false;
68 }
69
70 if (!Context.SourceActor->HasAuthority())
71 {
72 UE_LOG(LogDTCombat, Verbose, TEXT("CalculateDamage rejected on non-authority source %s."), *GetNameSafe(Context.SourceActor.Get()));
73 return false;
74 }
75
76 UDTAbilitySystemComponent* SourceASC = Cast<UDTAbilitySystemComponent>(UAbilitySystemBlueprintLibrary::GetAbilitySystemComponent(Context.SourceActor));
77 UDTAbilitySystemComponent* TargetASC = Cast<UDTAbilitySystemComponent>(UAbilitySystemBlueprintLibrary::GetAbilitySystemComponent(Context.TargetActor));
78 if (!SourceASC || !TargetASC)
79 {
80 UE_LOG(LogDTCombat, Verbose, TEXT("CalculateDamage failed: source or target ASC missing."));
81 return false;
82 }
83
84 FDTDamageContext DamageContext = Context;
85 UpdateContextTags(DamageContext, SourceASC, TargetASC);
86
87 FGameplayEffectContextHandle EventContext = SourceASC->MakeEffectContext();
88 EventContext.AddInstigator(DamageContext.SourceActor.Get(), DamageContext.EffectCauser.Get());
89 if (DamageContext.HitResult.HasValidHitObjectHandle() || DamageContext.HitResult.bBlockingHit)
90 {
91 EventContext.AddHitResult(DamageContext.HitResult, true);
92 }
93
94 BroadcastDamageEvent(DamageContext, SourceASC, TargetASC, DTGameplayTags::TAG_EVENT_DAMAGE_PRERESOLVE, DamageContext.BaseDamage, DamageContext.ContextTags, EventContext);
95
96 TArray<FDTCombatModifierSpec> Modifiers;
97 GatherModifiers(DamageContext, SourceASC, TargetASC, Modifiers);
98 SortModifiers(Modifiers);
99
100 FDTCombatAccumulator Accumulator;
101 Accumulator.Reset(FMath::Max(0.0f, DamageContext.BaseDamage));
102
103 TArray<FDTCombatModifierSpec> ConsumedSpecs;
104
105 ApplyModifierPhase(DamageContext, EDTCombatModifierPhase::Base, Modifiers, Accumulator, OutResult, ConsumedSpecs);
106 ApplyModifierPhase(DamageContext, EDTCombatModifierPhase::Outgoing, Modifiers, Accumulator, OutResult, ConsumedSpecs);
107 OutResult.PreMitigationDamage = Accumulator.Damage;
108
109 ApplyCrit(DamageContext, Accumulator, OutResult);
110 ApplyModifierPhase(DamageContext, EDTCombatModifierPhase::Crit, Modifiers, Accumulator, OutResult, ConsumedSpecs);
111 ApplyModifierPhase(DamageContext, EDTCombatModifierPhase::Incoming, Modifiers, Accumulator, OutResult, ConsumedSpecs);
112 ApplyModifierPhase(DamageContext, EDTCombatModifierPhase::Mitigation, Modifiers, Accumulator, OutResult, ConsumedSpecs);
113
114 if (!DamageContext.bSkipBuiltInMitigation)
115 {
116 ApplyMitigation(DamageContext, SourceASC, TargetASC, Accumulator, OutResult);
117 }
118
119 OutResult.MitigatedDamage = Accumulator.Damage;
120
121 ApplyModifierPhase(DamageContext, EDTCombatModifierPhase::Final, Modifiers, Accumulator, OutResult, ConsumedSpecs);
122
123 OutResult.FinalDamage = static_cast<float>(FMath::Max(0, FMath::RoundHalfToEven(Accumulator.Damage)));
124 OutResult.ResultTags.AddTag(DTGameplayTags::TAG_EFFECT_RESULT_COMBAT);
125
126 ApplyModifierPhase(DamageContext, EDTCombatModifierPhase::Proc, Modifiers, Accumulator, OutResult, ConsumedSpecs);
127
128 for (const FDTCombatModifierSpec& ConsumedSpec : ConsumedSpecs)
129 {
130 UDTAbilitySystemComponent* OwningASC = ConsumedSpec.OwningComponent.Get();
131 if (OwningASC)
132 {
133 FDTCombatModifierHandle Handle;
134 Handle.Id = ConsumedSpec.HandleId;
135 OwningASC->UnregisterCombatModifier(Handle);
136 }
137 }
138
139 FGameplayTagContainer EventTags = DamageContext.ContextTags;
140 EventTags.AppendTags(OutResult.ResultTags);
141
142 BroadcastDamageEvent(DamageContext, SourceASC, TargetASC, DTGameplayTags::TAG_EVENT_DAMAGE_RESOLVED, OutResult.FinalDamage, EventTags, EventContext);
143
144 return !OutResult.bCancelled;
145}

148{

151 OutEffectHandle = FActiveGameplayEffectHandle();

153 {

195 {
196 return false;
197 }

210 }
211
212 FGameplayTagContainer ContextTags = DamageContext.ContextTags;
213 if (DamageContext.DamageType.IsValid())
214 {
215 ContextTags.AddTag(DamageContext.DamageType);
216 }
217
218 ContextTags.AppendTags(Result.ResultTags);
219 if (FGameplayEffectContext* RawContext = ContextHandle.Get())
220 {
221 if (RawContext->GetScriptStruct() == FDTGameplayEffectContext::StaticStruct())
222 {
223 static_cast<FDTGameplayEffectContext*>(RawContext)->AppendContextTags(ContextTags);
224 }
225 }
226
227 FGameplayEffectSpecHandle SpecHandle = SourceASC->MakeOutgoingSpec(DamageEffectClass, 1.0f, ContextHandle);
228 if (!SpecHandle.IsValid())
229 {
230 return false;
231 }
232
233 FGameplayEffectSpec* Spec = SpecHandle.Data.Get();
234 Spec->SetSetByCallerMagnitude(DTGameplayTags::TAG_SETBYCALLER_DATA_ATTRIBUTE_DAMAGE, Result.FinalDamage);
235
236 if (DamageContext.DamageType.IsValid())
237 {
238 Spec->AddDynamicAssetTag(DamageContext.DamageType);
239 }
240
241 Spec->AddDynamicAssetTag(DTGameplayTags::TAG_EFFECT_RESULT_COMBAT);
242
243 OutEffectHandle = SourceASC->ApplyGameplayEffectSpecToTarget(*Spec, TargetASC);
244
245 const FVector TargetLocation = DamageContext.TargetActor->GetActorLocation();

249

305{
306 Modifiers.Sort([](const FDTCombatModifierSpec& A, const FDTCombatModifierSpec& B)
307 {
308 if (A.Phase != B.Phase)
309 {
310 return static_cast<uint8>(A.Phase) < static_cast<uint8>(B.Phase);
311 }
312 if (A.Priority != B.Priority)
313 {
314 return A.Priority > B.Priority;

320void UDTCombatSubsystem::ApplyModifierPhase(const FDTDamageContext& Context, EDTCombatModifierPhase Phase, const TArray<FDTCombatModifierSpec>& Modifiers, FDTCombatAccumulator& Accumulator, FDTDamageResult& Result, TArray<FDTCombatModifierSpec>& OutConsumedSpecs)
321{
322 TRACE_CPUPROFILER_EVENT_SCOPE(UDTCombatSubsystem::ApplyModifierPhase);
323
324 for (FDTCombatModifierSpec Spec : Modifiers)
325 {
326 if (Spec.Phase != Phase)
327 {
328 continue;
329 }
330
331 if (Spec.Definition)
332 {
333 if (!Spec.Definition->Matches(Context, Result))
334 {
335 continue;
336 }
337
338 if (Spec.ApplicationChance < 1.0f && FMath::FRand() > Spec.ApplicationChance)
339 {
340 continue;
341 }
342
343 for (const TObjectPtr<UDTCombatModifierRule>& Rule : Spec.Definition->Rules)
344 {
345 if (Rule)
346 {
347 Rule->Apply(Context, Spec);
348 }
349 }
350 }
351
352 Accumulator.ApplyModifier(Spec, Result);
353
354 if (Spec.bConsumeOnApply)
355 {
356 OutConsumedSpecs.Add(Spec);
357 }
358 }
359
360 Accumulator.FlushAdditiveBucket(Result, Phase);
361}

363void UDTCombatSubsystem::ApplyCrit(const FDTDamageContext& Context, FDTCombatAccumulator& Accumulator, FDTDamageResult& Result) const
364{
365 if (!Context.ContextTags.HasTag(DTGameplayTags::TAG_EFFECT_CONDITION_CANCRIT) || !Context.SourceActor)
366 {
367 return;
368 }
369
370 UAbilitySystemComponent* SourceASC = UAbilitySystemBlueprintLibrary::GetAbilitySystemComponent(Context.SourceActor);
371 if (!SourceASC)
372 {
373 return;
374 }
375
376 const float CritChance = FMath::Clamp(SourceASC->GetNumericAttribute(UDTAttributeSet::GetCritChanceAttribute()), 0.0f, 1.0f);
377 if (FMath::FRand() > CritChance)
378 {
379 return;
380 }
381
382 const float CritDamage = FMath::Max(0.0f, SourceASC->GetNumericAttribute(UDTAttributeSet::GetCritDamageAttribute()));
383
384 FDTCombatModifierSpec CritSpec;
385 CritSpec.Phase = EDTCombatModifierPhase::Crit;
386 CritSpec.Operation = EDTCombatModifierOp::MoreMultiplier;
387 CritSpec.Magnitude = 1.0f + CritDamage;
388 CritSpec.SourceTag = DTGameplayTags::TAG_EFFECT_RESULT_CRIT;
389
390 Accumulator.ApplyModifier(CritSpec, Result);
391
392 Result.ResultTags.AddTag(DTGameplayTags::TAG_EFFECT_RESULT_CRIT);
393}

395void UDTCombatSubsystem::ApplyMitigation(const FDTDamageContext& Context, UDTAbilitySystemComponent* SourceASC, UDTAbilitySystemComponent* TargetASC, FDTCombatAccumulator& Accumulator, FDTDamageResult& Result) const
396{
397 if (!TargetASC || Context.ContextTags.HasTag(DTGameplayTags::TAG_DAMAGE_TYPE_TRUE))
398 {
399 return;
400 }
401
402 const float DamageBefore = Accumulator.Damage;
403 float Armor = TargetASC->GetNumericAttribute(UDTAttributeSet::GetArmorAttribute());
404 const float PhysicalPenetration = SourceASC ? SourceASC->GetNumericAttribute(UDTAttributeSet::GetPhysicalPenetrationAttribute()) : 0.0f;
405 Armor *= 1.0f - FMath::Clamp(PhysicalPenetration + Accumulator.MitigationPenetration, 0.0f, 1.0f);
406
407 const float ArmorMitigationFactor = GetArmorMitigationFactor();
408 const float ArmorReduction = Armor > 0.0f && Accumulator.Damage > 0.0f ? Armor / (Armor + ArmorMitigationFactor * Accumulator.Damage) : 0.0f;
409 const float DamageResistance = FMath::Clamp(TargetASC->GetNumericAttribute(UDTAttributeSet::GetDamageResistanceAttribute()) + Accumulator.MitigationReduction, -10.0f, 0.75f);
410 Accumulator.Damage *= (1.0f - ArmorReduction) * (1.0f - DamageResistance);
411
412 FDTCombatModifierRecord Record;
413 Record.Phase = EDTCombatModifierPhase::Mitigation;
414 Record.Operation = EDTCombatModifierOp::Reduction;
415 Record.Magnitude = DamageBefore - Accumulator.Damage;
416 Record.DamageBefore = DamageBefore;
417 Record.DamageAfter = Accumulator.Damage;
418 Result.AppliedModifiers.Add(Record);
419}
