175void UDTEquipmentComponent::HandleItemEquipped(FGameplayTag SlotTag, UInventoryEquipmentInstance* Item)
176{
177 if (!Item)
178 {
179 return;
180 }
181
182 UDTEquipmentInstance* DTItem = Cast<UDTEquipmentInstance>(Item);
183 if (!DTItem)
184 {
185 UE_LOG(LogDTInventory, Warning, TEXT("HandleItemEquipped - Item '%s' is not a UDTEquipmentInstance"), *GetNameSafe(Item));
186 return;
187 }
188
189 const bool bIsWeapon = IsWeaponSlot(SlotTag);
190 const bool bIsOffHand = IsOffHandSlot(SlotTag);
191
192 if (HasAuthority())
193 {
194 if (UDTAbilitySystemComponent* ASC = GetAbilitySystemComponent())
195 {
196 DTItem->GrantEquipmentAbilities(ASC);
197
198 // Off-hand stats are fully activation-gated: RefreshOffHandSlot applies/removes the whole
199 // package (base + active) depending on whether the active weapon leaves the off-hand usable
200 if (!bIsOffHand)
201 {
202 DTItem->ApplyStats(ASC, bIsWeapon ? (SlotTag == ActiveWeaponSlot) : true);
203 }
204 }
205 }
206
207 if (bIsOffHand)
208 {
209 // Activate the off-hand only if the current weapon is one-handed
210 RefreshOffHandSlot();
211
212 // If the off-hand stayed inactive, ActivateItem never ran — UI still needs the slot-change event
213 if (!bOffHandActive)
214 {
215 BroadcastEquipmentChange(EEquipmentOp::ItemEquipped, SlotTag, Item);
216 }
217 return;
218 }
219
220 if (!bIsWeapon)
221 {
222 ActivateItem(SlotTag, DTItem);
223 return;
224 }
225
226 if (SlotTag == ActiveWeaponSlot)
227 {
228 RefreshActiveWeaponPresentation(ActiveWeaponSlot, true, false);
229 RefreshOffHandSlot();
230 }
231 else
232 {
233 // Equipped into an inactive weapon slot: no activation, but the slot content changed
234 BroadcastEquipmentChange(EEquipmentOp::ItemEquipped, SlotTag, Item);
235 }
236}

317bool UDTEquipmentComponent::SwapEquippedItems(FGameplayTag FirstSlotTag, FGameplayTag SecondSlotTag)
318{
319 if (!HasAuthority())
320 {
321 Server_SwapEquippedItems(FirstSlotTag, SecondSlotTag);
322 return true;
323 }
324
325 UDTEquipmentInstance* PreviouslyActiveWeapon = GetWeaponInstance(ActiveWeaponSlot);
326 const bool bSuccess = Super::SwapEquippedItems(FirstSlotTag, SecondSlotTag);
327 if (!bSuccess)
328 {
329 return false;
330 }
331
332 const bool bFirstIsWeapon = IsWeaponSlot(FirstSlotTag);
333 const bool bSecondIsWeapon = IsWeaponSlot(SecondSlotTag);
334
335 if (bFirstIsWeapon && bSecondIsWeapon)
336 {
337 if (ActiveWeaponSlot == FirstSlotTag || ActiveWeaponSlot == SecondSlotTag)
338 {
339 UDTEquipmentInstance* CurrentActiveWeapon = GetWeaponInstance(ActiveWeaponSlot);
340 if (PreviouslyActiveWeapon != CurrentActiveWeapon)
341 {
342 if (PreviouslyActiveWeapon)
343 {
344 DeactivateItem(PreviouslyActiveWeapon->GetEquippedSlotTag(), PreviouslyActiveWeapon);
345 }
346
347 ApplyServerActiveWeaponStats(PreviouslyActiveWeapon, CurrentActiveWeapon);
348 RefreshActiveWeaponPresentation(ActiveWeaponSlot, true, false);
349
350 // The active weapon changed (e.g. one-handed swapped for two-handed) — the off-hand may need to toggle
351 RefreshOffHandSlot();
352 }
353 }
354 return true;
355 }
356
357 UDTAbilitySlotComponent* SlotComponent = GetAbilitySlotComponent();
358 if (!SlotComponent)
359 {
360 return true;
361 }
362
363 const TArray<FGameplayTag>* FirstAbilitySlots = GetAbilitySlotsForEquipment(FirstSlotTag);
364 const TArray<FGameplayTag>* SecondAbilitySlots = GetAbilitySlotsForEquipment(SecondSlotTag);
365
366 if (!FirstAbilitySlots || !SecondAbilitySlots)
367 {
368 return true;
369 }
370
371 const int32 PairCount = FMath::Min(FirstAbilitySlots->Num(), SecondAbilitySlots->Num());
372 for (int32 i = 0; i < PairCount; ++i)
373 {
374 const FGameplayTag& FirstAbilitySlot = (*FirstAbilitySlots)[i];
375 const FGameplayTag& SecondAbilitySlot = (*SecondAbilitySlots)[i];
376
377 SlotComponent->SwapSlottedAbilities(FirstAbilitySlot, SecondAbilitySlot);
378
379 UE_LOG(LogDTInventory, Verbose,
380 TEXT("SwapEquippedItems - Swapped ability slots '%s' <-> '%s'"),
381 *FirstAbilitySlot.ToString(), *SecondAbilitySlot.ToString());
382 }
383
384 return true;

558void UDTEquipmentComponent::RefreshOffHandSlot()
559{
560 UDTEquipmentInstance* OffHandItem = GetOffHandItem();
561 if (!OffHandItem)
562 {
563 bOffHandActive = false;
564 return;
565 }
566
567 const bool bShouldBeActive = IsActiveWeaponOneHanded();
568
569 if (bShouldBeActive && !bOffHandActive)
570 {
571 // Activate the off-hand
572 if (HasAuthority())
573 {
574 if (UDTAbilitySystemComponent* ASC = GetAbilitySystemComponent())
575 {
576 OffHandItem->ApplyStats(ASC, true);
577 }
578 }
579
580 ActivateItem(DTGameplayTags::TAG_ITEM_SLOT_WEAPON_OFFHAND, OffHandItem);
581 bOffHandActive = true;
582
583 UE_LOG(LogDTInventory, Verbose, TEXT("RefreshOffHandSlot - Activated off-hand '%s'"), *GetNameSafe(OffHandItem));
584 }
585 else if (!bShouldBeActive && bOffHandActive)
586 {
587 // Deactivate the off-hand
588 DeactivateItem(DTGameplayTags::TAG_ITEM_SLOT_WEAPON_OFFHAND, OffHandItem);
589
590 if (HasAuthority())
591 {
592 if (UDTAbilitySystemComponent* ASC = GetAbilitySystemComponent())
593 {
594 // Full removal — an inactive off-hand contributes no stats at all
595 OffHandItem->RemoveStats(ASC);
596 }
597 }
598
599 bOffHandActive = false;
600
601 UE_LOG(LogDTInventory, Verbose, TEXT("RefreshOffHandSlot - Deactivated off-hand '%s'"), *GetNameSafe(OffHandItem));
602 }

657void UDTEquipmentComponent::SlotAbilitiesForItem(UDTEquipmentInstance* Item) const
658{
659 if (!Item)
660 {
661 return;
662 }
663
664 UDTAbilitySlotComponent* SlotComponent = GetAbilitySlotComponent();
665 if (!SlotComponent)
666 {
667 UE_LOG(LogDTInventory, Warning, TEXT("SlotAbilitiesForItem - No AbilitySlotComponent found"));
668 return;
669 }
670
671 UDTAbilitySystemComponent* ASC = GetAbilitySystemComponent();
672 if (!ASC)
673 {
674 UE_LOG(LogDTInventory, Warning, TEXT("SlotAbilitiesForItem - No AbilitySystemComponent found"));
675 return;
676 }
677
678 TArray<FGameplayAbilitySpecHandle> GrantedHandles;
679 Item->GetGrantedEquipmentAbilitySpecHandles(GrantedHandles);
680
681 const FGameplayTag EquipmentSlotTag = Item->GetEquippedSlotTag();
682 const TArray<FGameplayTag>* MappedAbilitySlots = GetAbilitySlotsForEquipment(EquipmentSlotTag);
683 int32 MappedSlotIndex = 0;
684
685 for (const FGameplayAbilitySpecHandle& Handle : GrantedHandles)
686 {
687 FGameplayTag TargetAbilitySlot;
688
689 if (MappedAbilitySlots && MappedSlotIndex < MappedAbilitySlots->Num())
690 {
691 FGameplayTag MappedSlot = (*MappedAbilitySlots)[MappedSlotIndex];
692 MappedSlotIndex++;
693
694 FText ValidationReason;
695 if (SlotComponent->CanSlotAbility(MappedSlot, Handle, ValidationReason))
696 {
697 TargetAbilitySlot = MappedSlot;
698 }
699 else
700 {
701 UE_LOG(LogDTInventory, Verbose, TEXT("SlotAbilitiesForItem - Mapped slot %s invalid: %s"),
702 *MappedSlot.ToString(), *ValidationReason.ToString());
703 }
704 }
705
706 if (!TargetAbilitySlot.IsValid())
707 {
708 FText FailureReason;
709 TargetAbilitySlot = SlotComponent->FindAvailableSlotForAbility(Handle, FailureReason);
710 if (!TargetAbilitySlot.IsValid())
711 {
712 UE_LOG(LogDTInventory, Verbose, TEXT("SlotAbilitiesForItem - No available slot: %s"), *FailureReason.ToString());
713 continue;
714 }
715 }
716
717 // Weapon/amulet/utility abilities are equipment-granted: flag them so the per-slot player save excludes them.
718 SlotComponent->SlotAbility(TargetAbilitySlot, Handle, /*bFromEquipment*/ true);
719
720 UE_LOG(LogDTInventory, VeryVerbose, TEXT("SlotAbilitiesForItem - Slotted ability (Handle: %s) to slot %s"), *Handle.ToString(), *TargetAbilitySlot.ToString());
721 }
