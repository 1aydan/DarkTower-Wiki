133bool UDTEquipmentPanelWidget::HandleItemDrop(UDTInventoryItemWidget* SourceWidget, UDTInventoryItemWidget* TargetWidget)
134{
135 if (!SourceWidget || !TargetWidget || !EquipmentComponent)
136 {
137 return false;
138 }
139
140 UInventoryItemInstance* SourceItem = SourceWidget->GetItemInstance();
141 UInventoryItemInstance* TargetItem = TargetWidget->GetItemInstance();
142
143 // Case 1: Dragging from inventory to empty equipment slot
144 if (SourceItem && !TargetItem && !SourceWidget->SlotTag.IsValid())
145 {
146 return HandleEquipFromInventory(SourceWidget, TargetWidget, SourceItem);
147 }
148
149 // Case 2: Dragging from inventory to occupied equipment slot (replace/swap)
150 if (SourceItem && TargetItem && !SourceWidget->SlotTag.IsValid())
151 {
152 return HandleEquipWithSwap(SourceWidget, TargetWidget, SourceItem, TargetItem);
153 }
154
155 // Case 3: Swapping items between two equipment slots
156 if (SourceItem && TargetItem && SourceWidget->SlotTag.IsValid())
157 {
158 return HandleSwapEquipment(SourceWidget, TargetWidget, SourceItem, TargetItem);
159 }
160
161 // Case 4: Moving item to empty equipment slot (from another equipment slot)
162 if (SourceItem && !TargetItem && SourceWidget->SlotTag.IsValid())
163 {
164 return HandleMoveEquipment(SourceWidget, TargetWidget, SourceItem);
165 }
166
167 UE_LOG(LogDTInventory, Warning, TEXT("Unhandled equipment drop case"));
168 return false;
169}

171bool UDTEquipmentPanelWidget::HandleEquipFromInventory(UDTInventoryItemWidget* SourceWidget, UDTInventoryItemWidget* TargetWidget, UInventoryItemInstance* SourceItem)
172{
173 UDTInventoryComponent* InventoryComp = GetInventoryComponent();
174 if (!InventoryComp || !EquipmentComponent || !SourceItem)
175 {
176 return false;
177 }
178
179 FGameplayTag TargetSlotTag = TargetWidget->SlotTag;
180
181 if (!EquipmentComponent->CanEquipItemInSlot(TargetSlotTag, SourceItem))
182 {
183 UE_LOG(LogDTInventory, Warning, TEXT("Cannot equip item in slot %s"), *TargetSlotTag.ToString());
184 return false;
185 }
186
187 bool bSuccess = EquipmentComponent->EquipItemFromInventory(TargetSlotTag, SourceItem, InventoryComp);
188
189 if (bSuccess)
190 {
191 UE_LOG(LogDTInventory, Verbose, TEXT("Equipped item %s to slot %s"),
192 *SourceItem->GetItemInfo().ItemName, *TargetSlotTag.ToString());
193 }
194
195 return bSuccess;

198bool UDTEquipmentPanelWidget::HandleEquipWithSwap(UDTInventoryItemWidget* SourceWidget, UDTInventoryItemWidget* TargetWidget, UInventoryItemInstance* SourceItem, UInventoryItemInstance* TargetItem)
199{
200 UDTInventoryComponent* InventoryComp = GetInventoryComponent();
201 if (!InventoryComp || !EquipmentComponent || !SourceItem || !TargetItem)
202 {
203 return false;
204 }
205
206 FGameplayTag TargetSlotTag = TargetWidget->SlotTag;
207
208 if (!EquipmentComponent->CanEquipItemInSlot(TargetSlotTag, SourceItem))
209 {
210 UE_LOG(LogDTInventory, Warning, TEXT("Cannot equip item in slot %s"), *TargetSlotTag.ToString());
211 return false;
212 }
213
214 // Remove the dragged item first so the displaced equipment can land in its vacated slot
215 FInventoryItemInfo SourceItemInfo = SourceItem->GetItemInfo();
216 const int32 SourceSlotIndex = SourceItemInfo.SlotIndex;
217
218 if (!InventoryComp->RemoveItem(SourceItem))
219 {
220 UE_LOG(LogDTInventory, Warning, TEXT("Failed to remove item from inventory"));
221 return false;
222 }
223
224 if (!EquipmentComponent->UnequipItemToInventory(TargetSlotTag, InventoryComp, SourceSlotIndex))
225 {
226 InventoryComp->AddItem(SourceItemInfo);
227 UE_LOG(LogDTInventory, Warning, TEXT("Failed to unequip current item from slot %s"), *TargetSlotTag.ToString());
228 return false;
229 }
230
231 bool bSuccess = EquipmentComponent->EquipItem(TargetSlotTag, SourceItem);
232
233 if (bSuccess)
234 {
235 UE_LOG(LogDTInventory, Verbose, TEXT("Swapped equipment in slot %s with inventory item"), *TargetSlotTag.ToString());
236 }
237 else
238 {
239 InventoryComp->AddItem(SourceItemInfo);
240 UE_LOG(LogDTInventory, Warning, TEXT("Failed to equip item to slot %s"), *TargetSlotTag.ToString());
241 }
242
243 return bSuccess;
244}

267bool UDTEquipmentPanelWidget::HandleMoveEquipment(UDTInventoryItemWidget* SourceWidget, UDTInventoryItemWidget* TargetWidget, UInventoryItemInstance* SourceItem)
268{
269 if (!SourceItem)
270 {
271 return false;
272 }
273
274 if (SourceWidget->IsDefaultEquipmentItem())
275 {
276 UE_LOG(LogDTInventory, Verbose, TEXT("HandleMoveEquipment: Source item is default equipment and cannot be moved"));
277 return false;
278 }
279
280 FGameplayTag SourceSlotTag = SourceWidget->SlotTag;
281 FGameplayTag TargetSlotTag = TargetWidget->SlotTag;
282
283 if (!EquipmentComponent->CanEquipItemInSlot(TargetSlotTag, SourceItem))
284 {
285 UE_LOG(LogDTInventory, Warning, TEXT("Cannot equip item in slot %s"), *TargetSlotTag.ToString());
286 return false;
287 }
288
289 UInventoryItemInstance* UnequippedItem = EquipmentComponent->UnequipItem(SourceSlotTag);
290 if (!UnequippedItem)
291 {
292 return false;
293 }
294
295 bool bSuccess = EquipmentComponent->EquipItem(TargetSlotTag, UnequippedItem);
296
297 if (bSuccess)
298 {
299 UE_LOG(LogDTInventory, Verbose, TEXT("Moved equipment from slot %s to slot %s"),
300 *SourceSlotTag.ToString(), *TargetSlotTag.ToString());
301 }
302
303 return bSuccess;
