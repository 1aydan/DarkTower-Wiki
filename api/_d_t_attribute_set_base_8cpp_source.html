73void UDTAttributeSetBase::AdjustAttributeForMaxChange(FGameplayAttributeData& AffectedAttribute, const FGameplayAttributeData& MaxAttribute, const float NewMaxValue, const FGameplayAttribute& AffectedAttributeProperty) const
74{
75 UAbilitySystemComponent* AbilitySystemComponent = GetOwningAbilitySystemComponent();
76 if (!AbilitySystemComponent)
77 {
78 return;
79 }
80
81 const float CurrentMaxValue = MaxAttribute.GetCurrentValue();
82 const float CurrentValue = AffectedAttribute.GetCurrentValue();
83
84 if (!FMath::IsNearlyEqual(CurrentMaxValue, NewMaxValue) && CurrentMaxValue > 0.f)
85 {
86 // Keep the affected attribute at the same relative percent of its new max
87 const float Ratio = CurrentValue / CurrentMaxValue;
88 const float NewValue = FMath::RoundToFloat(NewMaxValue * Ratio);
89
90 AbilitySystemComponent->ApplyModToAttribute(AffectedAttributeProperty, EGameplayModOp::Override, NewValue);
91 }
92}

94void UDTAttributeSetBase::GetExecutionDataFromMod(const FGameplayEffectModCallbackData& Data, FDTAttributeSetExecutionData& OutExecutionData)
95{
96 OutExecutionData.Context = Data.EffectSpec.GetContext();
97 OutExecutionData.SourceASC = OutExecutionData.Context.GetOriginalInstigatorAbilitySystemComponent();
98 OutExecutionData.SourceTags = *Data.EffectSpec.CapturedSourceTags.GetAggregatedTags();
99 Data.EffectSpec.GetAllAssetTags(OutExecutionData.SpecAssetTags);
100
101 OutExecutionData.TargetActor = Data.Target.AbilityActorInfo->AvatarActor.IsValid() ? Data.Target.AbilityActorInfo->AvatarActor.Get() : nullptr;
102 OutExecutionData.TargetController = Data.Target.AbilityActorInfo->PlayerController.IsValid() ? Data.Target.AbilityActorInfo->PlayerController.Get() : nullptr;
103 OutExecutionData.TargetPawn = Cast<APawn>(OutExecutionData.TargetActor);
104 OutExecutionData.TargetCoreComponent = GetCoreComponent(OutExecutionData.TargetActor);
105
106 if (OutExecutionData.SourceASC && OutExecutionData.SourceASC->AbilityActorInfo.IsValid())
107 {
108 // Get the Source actor, which should be the damage causer (instigator)
109 if (OutExecutionData.SourceASC->AbilityActorInfo->AvatarActor.IsValid())
110 {
111 if (OutExecutionData.Context.GetEffectCauser())
112 {
113 OutExecutionData.SourceActor = OutExecutionData.Context.GetEffectCauser();
114 }
115 else
116 {
117 OutExecutionData.SourceActor = OutExecutionData.SourceASC->AbilityActorInfo->AvatarActor.IsValid()
118 ? OutExecutionData.SourceASC->AbilityActorInfo->AvatarActor.Get()
119 : nullptr;
120 }
121 }
122
123 OutExecutionData.SourceController = OutExecutionData.SourceASC->AbilityActorInfo->PlayerController.IsValid()
124 ? OutExecutionData.SourceASC->AbilityActorInfo->PlayerController.Get()
125 : nullptr;
126 OutExecutionData.SourcePawn = Cast<APawn>(OutExecutionData.SourceActor);
127 OutExecutionData.SourceCoreComponent = GetCoreComponent(OutExecutionData.SourceActor);
128 }
129
130 OutExecutionData.SourceObject = Data.EffectSpec.GetEffectContext().GetSourceObject();
131
132 // Compute the delta between old and new, if it is available
133 OutExecutionData.DeltaValue = 0.f;
134 if (Data.EvaluatedData.ModifierOp == EGameplayModOp::Type::Additive)
135 {
136 OutExecutionData.DeltaValue = Data.EvaluatedData.Magnitude;
137 }
138}
