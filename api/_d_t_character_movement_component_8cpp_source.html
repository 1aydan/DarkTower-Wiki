46const FCharacterGroundInfo& UDTCharacterMovementComponent::GetGroundInfo()
47{
48 if (!CharacterOwner || (GFrameCounter == CachedGroundInfo.LastUpdateFrame))
49 {
50 return CachedGroundInfo;
51 }
52
53 if (MovementMode == MOVE_Walking)
54 {
55 CachedGroundInfo.GroundHitResult = CurrentFloor.HitResult;
56 CachedGroundInfo.GroundDistance = 0.0f;
57 }
58 else
59 {
60 const UCapsuleComponent* CapsuleComp = CharacterOwner->GetCapsuleComponent();
61 check(CapsuleComp);
62
63 const float CapsuleHalfHeight = CapsuleComp->GetUnscaledCapsuleHalfHeight();
64 const ECollisionChannel CollisionChannel = (UpdatedComponent ? UpdatedComponent->GetCollisionObjectType() : ECC_Pawn);
65 const FVector TraceStart(GetActorLocation());
66 const FVector TraceEnd(TraceStart.X, TraceStart.Y, (TraceStart.Z - GroundTraceDistance - CapsuleHalfHeight));
67
68 FCollisionQueryParams QueryParams(SCENE_QUERY_STAT(LyraCharacterMovementComponent_GetGroundInfo), false, CharacterOwner);
69 FCollisionResponseParams ResponseParam;
70 InitCollisionParams(QueryParams, ResponseParam);
71
72 FHitResult HitResult;
73 GetWorld()->LineTraceSingleByChannel(HitResult, TraceStart, TraceEnd, CollisionChannel, QueryParams, ResponseParam);
74
75 CachedGroundInfo.GroundHitResult = HitResult;
76 CachedGroundInfo.GroundDistance = GroundTraceDistance;
77
78 if (MovementMode == MOVE_NavWalking)
79 {
80 CachedGroundInfo.GroundDistance = 0.0f;
81 }
82 else if (HitResult.bBlockingHit)
83 {
84 CachedGroundInfo.GroundDistance = FMath::Max((HitResult.Distance - CapsuleHalfHeight), 0.0f);
85 }
86 }
87
88 CachedGroundInfo.LastUpdateFrame = GFrameCounter;
89
90 return CachedGroundInfo;
91}
