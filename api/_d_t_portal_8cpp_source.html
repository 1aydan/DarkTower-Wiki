154void ADTPortal::OnOverlapBegin(UPrimitiveComponent* OverlappedComp, AActor* OtherActor, UPrimitiveComponent* OtherComp, int32 OtherBodyIndex, bool bFromSweep, const FHitResult& SweepResult)
155{
156 if (!bIsActive || !HasAuthority())
157 {
158 return;
159 }
160
161 APawn* Pawn = Cast<APawn>(OtherActor);
162 if (!Pawn || !Pawn->IsPlayerControlled())
163 {
164 return;
165 }
166
167 TWeakObjectPtr<APawn> WeakPawn(Pawn);
168 if (OverlapTimers.Contains(WeakPawn))
169 {
170 return;
171 }
172
173 FTimerHandle TimerHandle;
174 FTimerDelegate TimerDel;
175 TimerDel.BindUObject(this, &ADTPortal::OnOverlapTimerComplete, Pawn);
176 GetWorld()->GetTimerManager().SetTimer(TimerHandle, TimerDel, OverlapDuration, false);
177 OverlapTimers.Add(WeakPawn, TimerHandle);
178
179 UE_LOG(LogDT, Verbose, TEXT("DTPortal: Pawn %s started overlap timer (%.1fs)"),
180 *Pawn->GetName(), OverlapDuration);
181}

207void ADTPortal::OnOverlapTimerComplete(APawn* Pawn)
208{
209 if (!Pawn || !bIsActive)
210 {
211 return;
212 }
213
214 TWeakObjectPtr<APawn> WeakPawn(Pawn);
215 OverlapTimers.Remove(WeakPawn);
216
217 ADTPlayerController* PC = Cast<ADTPlayerController>(Pawn->GetController());
218 if (!PC)
219 {
220 return;
221 }
222
223 if (bOpenHostMapSetupOnEnter)
224 {
225 OnPlayerEnteredPortal(PC);
226
227 if (PC->IsHost())
228 {
229 FDTHostRequest Request;
230 Request.EntryPoint = HostMapSetupEntryPoint;
231 Request.InitialTab = HostMapSetupInitialTab;
232 Request.TravelMapName = HostMapSetupMapName;
233 PC->Client_ShowHostMapSetup(Request);
234 UE_LOG(LogDT, Log, TEXT("DTPortal: Player %s entered host setup portal. Opening tab %d."), *PC->GetName(), static_cast<int32>(HostMapSetupInitialTab));
235 }
236
237 return;
238 }
239
240 // Broadcast portal entered event
241 UGameplayMessageSubsystem& MessageSubsystem = UGameplayMessageSubsystem::Get(this);
242 FDTDungeonSummaryMessage EnteredMessage;
243 EnteredMessage.CompletedFloor = CachedCompletedFloor;
244 EnteredMessage.DungeonSeed = CachedDungeonSeed;
245 MessageSubsystem.BroadcastMessage(DTGameplayTags::TAG_EVENT_PORTAL_ENTERED, EnteredMessage);
246
247 OnPlayerEnteredPortal(PC);
248
249 // Push summary to this specific player
250 PC->Client_ShowDungeonSummary(CachedCompletedFloor, CachedDungeonSeed);
251
252 UE_LOG(LogDT, Log, TEXT("DTPortal: Player %s entered portal. Showing summary."), *PC->GetName());
253}
