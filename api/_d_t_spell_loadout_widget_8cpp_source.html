68void UDTSpellLoadoutWidget::RebuildSpellBar()
69{
70 if (!WB_SpellBar || !SpellSlotWidgetClass)
71 {
72 return;
73 }
74
75 UDTAbilitySlotComponent* SlotComponent = GetAbilitySlotComponent();
76 if (!SlotComponent)
77 {
78 // Slot component not ready yet (e.g. PlayerState still replicating) — retry shortly.
79 if (UWorld* World = GetWorld())
80 {
81 World->GetTimerManager().SetTimer(InitRetryHandle, this, &UDTSpellLoadoutWidget::RebuildSpellBar, 0.1f, false);
82 }
83 return;
84 }
85
86 WB_SpellBar->ClearChildren();
87
88 // One slot widget per spell slot (the spell-slot tags are the same regardless of which set is being edited;
89 // only the displayed content differs by editing set).
90 const TArray<FAbilitySlot> SpellSlots = SlotComponent->GetSlotsByTags(SpellSlotFilterTags, /*bRequireAll*/ false);
91 for (const FAbilitySlot& SpellSlot : SpellSlots)
92 {
93 UDTSpellLoadoutSlotWidget* SlotWidget = CreateWidget<UDTSpellLoadoutSlotWidget>(GetOwningPlayer(), SpellSlotWidgetClass);
94 if (!SlotWidget)
95 {
96 continue;
97 }
98
99 WB_SpellBar->AddChild(SlotWidget);
100 SlotWidget->SetupSpellSlot(this, SpellSlot.Definition.SlotTag);
101 }
102
103 OnLoadoutUpdated();
104}
