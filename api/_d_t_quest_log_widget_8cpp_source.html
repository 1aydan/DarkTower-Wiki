36void UDTQuestLogWidget::RefreshQuestLog()
37{
38 CachedDisplayEntries.Reset();
39
40 const UQuestSubsystem* QuestSubsystem = ResolveQuestSubsystem();
41 if (!QuestSubsystem)
42 {
43 OnQuestLogRefreshed();
44 return;
45 }
46
47 TArray<FQuestData> FilteredQuests;
48 const TArray<FQuestData> AllQuests = QuestSubsystem->GetAllTrackedQuests(ResolvePlayerState());
49 for (const FQuestData& Entry : AllQuests)
50 {
51 if (ShouldIncludeEntry(Entry))
52 {
53 FilteredQuests.Add(Entry);
54 }
55 }
56
57 FilteredQuests.Sort([](const FQuestData& Left, const FQuestData& Right)
58 {
59 const int32 LeftPriority = GetQuestStateSortPriority(Left.State);
60 const int32 RightPriority = GetQuestStateSortPriority(Right.State);
61 if (LeftPriority != RightPriority)
62 {
63 return LeftPriority < RightPriority;
64 }
65
66 return Left.QuestTag.ToString() < Right.QuestTag.ToString();
67 });
68
69 // Build display-ready entries from sorted quest data
70 CachedDisplayEntries = UQuestBlueprintLibrary::BuildQuestDisplayDataArray(this, FilteredQuests);
71
72 OnQuestLogRefreshed();
73}

116bool UDTQuestLogWidget::ShouldIncludeEntry(const FQuestData& Entry) const
117{
118 if (!bIncludeSharedQuests && Entry.Scope == EQuestScope::Shared)
119 {
120 return false;
121 }
122
123 if (!bIncludeNotAvailableQuests && Entry.State == EQuestState::NotAvailable)
124 {
125 return false;
126 }
127
128 if (!bIncludeCompletedQuests && Entry.State == EQuestState::Completed)
129 {
130 return false;
131 }
132
133 return true;
134}
