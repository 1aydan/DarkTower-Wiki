381 FGameplayTag FindTagByString(const FString& TagString, bool bMatchPartialString)
382 {
383 const UGameplayTagsManager& Manager = UGameplayTagsManager::Get();
384 FGameplayTag Tag = Manager.RequestGameplayTag(FName(*TagString), false);
385
386 if (!Tag.IsValid() && bMatchPartialString)
387 {
388 FGameplayTagContainer AllTags;
389 Manager.RequestAllGameplayTags(AllTags, true);
390
391 for (const FGameplayTag& TestTag : AllTags)
392 {
393 if (TestTag.ToString().Contains(TagString))
394 {
395 UE_LOG(LogDT, Display, TEXT("Could not find exact match for tag [%s] but found partial match on tag [%s]."), *TagString, *TestTag.ToString());
396 Tag = TestTag;
397 break;
398 }
399 }
400 }
401
402 return Tag;
403 }
