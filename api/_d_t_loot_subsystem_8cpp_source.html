207FGameplayTag UDTLootSubsystem::CalculateItemRarity(const FLootContext& Context, float FinalLootBudget) const
208{
209 // Rarity weights scale with the loot budget: Common stays flat (GlobalWeightScalar 0) while rarer
210 // tiers grow, so higher floors / bonuses make Common relatively less common.
211 // CachedRarityWeights is built once (see CacheRarityWeights); only FinalLootBudget varies per roll.
212 if (CachedRarityWeights.Num() == 0)
213 {
214 return DTGameplayTags::TAG_ITEM_RARITY_COMMON;
215 }
216
217 float TotalWeight = 0.0f;
218 for (const FCachedRarityWeight& Rarity : CachedRarityWeights)
219 {
220 TotalWeight += Rarity.GlobalBaseRate * (1.0f + (FinalLootBudget * Rarity.GlobalWeightScalar));
221 }
222
223 if (TotalWeight <= 0.0f)
224 {
225 return DTGameplayTags::TAG_ITEM_RARITY_COMMON;
226 }
227
228 // LootStream is re-seeded per item, so rarity is deterministic for regeneration
229 const float Roll = LootStream.FRandRange(0.0f, TotalWeight);
230 float CurrentWeight = 0.0f;
231
232 for (const FCachedRarityWeight& Rarity : CachedRarityWeights)
233 {
234 CurrentWeight += Rarity.GlobalBaseRate * (1.0f + (FinalLootBudget * Rarity.GlobalWeightScalar));
235 if (Roll <= CurrentWeight)
236 {
237 return Rarity.RarityTag;
238 }
239 }
240
241 return CachedRarityWeights.Last().RarityTag;
242}

267bool UDTLootSubsystem::GenerateLootItem(const FLootTableRow& Row, const FLootContext& Context, int32 ItemIndex, float EffectiveFloor, float FinalLootBudget, FInventoryItemInfo& OutLootInfo)
268{
269 TRACE_CPUPROFILER_EVENT_SCOPE(UDTLootSubsystem::GenerateLootItem)
270
271 int32 OriginalSeed = LootStream.GetInitialSeed();
272
273 float DropRoll = LootStream.FRand();
274 if (DropRoll > Row.DropChance)
275 {
276 return false;
277 }
278
279 if (Row.LootDefinition.IsNull())
280 {
281 UE_LOG(LogDTLoot, Warning, TEXT("GenerateLootItem: LootDefinition is null"));
282 return false;
283 }
284
285 UInventoryItemDefinition* ItemDef = UDefinitionAssetManager::Get().GetAsset(Row.LootDefinition);
286 if (!ItemDef)
287 {
288 UE_LOG(LogDTLoot, Warning, TEXT("GenerateLootItem: Failed to load item definition"));
289 return false;
290 }
291
292 int32 ItemSeed;
293 if (bUseHugeSeedDistribution)
294 {
295 ItemSeed = (ItemIndex == -1) ? OriginalSeed : UDTMathLibrary::GenerateDerivedSeed(OriginalSeed, ItemIndex);
296 }
297 else
298 {
299 ItemSeed = (ItemIndex == -1) ? OriginalSeed : OriginalSeed + (ItemIndex * 7919);
300 }
301
302 LootStream.Initialize(ItemSeed);
303
304 OutLootInfo.Init(ItemDef, ItemSeed);
305
306 if (ItemDef->bStackable)
307 {
308 OutLootInfo.StackCount = CalculateStackCount(Row);
309 }
310
311 if (ItemSupportsLootGen(ItemDef))
312 {
313 // EffectiveFloor and FinalLootBudget are computed once per generation by the caller.
314 const float ItemPower = CalculateItemPower(EffectiveFloor);
315
316 OutLootInfo.RarityTag = CalculateItemRarity(Context, FinalLootBudget);
317 OutLootInfo.ItemPower = ItemPower;
318
319 const float RarityStatMultiplier = GetRarityStatMultiplier(OutLootInfo.RarityTag);
320
321 RollBaseStats(OutLootInfo, ItemPower, RarityStatMultiplier);
322
323 // Provisional item level (floor + rarity, before stat contribution) so affixes can gate on
324 // CanRollAtLevel. Computed up front because the final item level depends on rolled stats.
325 const int32 ProvisionalItemLevel = CalculateBaseItemLevel(EffectiveFloor, OutLootInfo.RarityTag);
326 RollAffixes(OutLootInfo, EffectiveFloor, ItemPower, ProvisionalItemLevel);
327
328 const int32 BaseItemLevel = ProvisionalItemLevel;
329 const float TotalStatValue = CalculateTotalStatValue(OutLootInfo);
330
331 OutLootInfo.ItemLevel = CalculateFinalItemLevel(BaseItemLevel, TotalStatValue);
332 OutLootInfo.RequiredLevel = CalculateLevelRequirement(OutLootInfo.ItemLevel);
333 OutLootInfo.ItemValue = CalculateItemValue(OutLootInfo.ItemLevel, OutLootInfo.RarityTag);
334 }
335 else
336 {
337 OutLootInfo.RarityTag = DTGameplayTags::TAG_ITEM_RARITY_COMMON;
338 OutLootInfo.ItemLevel = 1;
339 OutLootInfo.RequiredLevel = 1;
340 OutLootInfo.ItemValue = CalculateItemValue(OutLootInfo.ItemLevel, OutLootInfo.RarityTag);
341 OutLootInfo.ItemPower = 1.0f;
342 }
343
344 UE_LOG(LogDTLoot, Verbose, TEXT("GenerateLootItem: ItemSeed=%d (OriginalSeed=%d, ItemIndex=%d), Rarity=%s"),
345 ItemSeed, OriginalSeed, ItemIndex, *OutLootInfo.RarityTag.ToString());
346
347 return true;
348}

354
355 OutGeneratedLoot.Empty();
356
357 if (Context.LootTables.Num() == 0 && Context.PreGeneratedLoot.Num() == 0 && Context.ItemDefinitions.Num() == 0)

360 return false;
361 }
362
363 if (!CachedGlobalLootData)
364 {
365 UE_LOG(LogDTLoot, Fatal, TEXT("UDTLootSubsystem::GenerateLootFromContext - CachedGlobalLootData is null and this never should be the case"));
366 return false;
367 }
368
369 InitializeRandomStreams(Context);
370
371 UE_LOG(LogDTLoot, Verbose, TEXT("UDTLootSubsystem::GenerateLootFromContext - Generating loot from context with seed: %d, MinRolls: %d, MaxRolls: %d"), LootStream.GetInitialSeed(), Context.MinLootRolls, Context.MaxLootRolls);
372
373 // Floor + loot budget are constant for this generation (depend only on the context), so compute
374 // them once here and thread them into every GenerateLootItem call (rarity) and the quantity hook.
375 const int32 RawFloor = GetEffectiveFloorFromContext(Context);
376 const float EffectiveFloor = CalculateEffectiveFloor(RawFloor);
377 const float FinalLootBudget = CalculateFinalLootBudget(EffectiveFloor, Context);
378
379 TArray<FLootTableRow*> CombinedRows;
380 TArray<float> CombinedWeights;
381
382 // Keep predefined rows in scope for the entire generation process
383 TArray<FLootTableRow> SyntheticRows;
384 SyntheticRows.Reserve(Context.PreGeneratedLoot.Num() + Context.ItemDefinitions.Num());
385
386 if (Context.PreGeneratedLoot.Num() > 0)
387 {
388 UE_LOG(LogDTLoot, Verbose, TEXT("UDTLootSubsystem::GenerateLootFromContext - Creating synthetic rows from %d pre-generated items"), Context.PreGeneratedLoot.Num());
389
390 for (const FInventoryItemInfo& PreGenItem : Context.PreGeneratedLoot)
391 {
392 if (PreGenItem.ItemDefinition)

398 SyntheticRow.DropChance = 1.0f;
399 SyntheticRow.Weight = 1.0f;
400 SyntheticRow.MinStackCount = PreGenItem.StackCount;
401 SyntheticRow.MaxStackCount = PreGenItem.StackCount;
402
403 CombinedRows.Add(&SyntheticRow);
404 CombinedWeights.Add(SyntheticRow.Weight);
405 }
406 }
407 }
408
409 if (Context.ItemDefinitions.Num() > 0)
410 {
411 if (Context.bGuaranteeAllItems)
412 {
413 UE_LOG(LogDTLoot, Verbose, TEXT("UDTLootSubsystem::GenerateLootFromContext - bGuaranteeAllItems: outputting all %d item definitions directly"), Context.ItemDefinitions.Num());
414
415 for (int32 i = 0; i < Context.ItemDefinitions.Num(); ++i)
416 {
417 const TObjectPtr<UInventoryItemDefinition>& ItemDef = Context.ItemDefinitions[i];
418 if (!ItemDef)
419 {
420 UE_LOG(LogDTLoot, Warning, TEXT("UDTLootSubsystem::GenerateLootFromContext - Null item definition skipped in bGuaranteeAllItems path"));
421 continue;
422 }
423
424 FLootTableRow SyntheticRow;
425 SyntheticRow.LootDefinition = ItemDef.Get();
426 SyntheticRow.DropChance = 1.0f;
427 SyntheticRow.Weight = 1.0f;
428 SyntheticRow.MinStackCount = ItemDef->bStackable ? Context.MinItemStackCount : 1;
429 SyntheticRow.MaxStackCount = ItemDef->bStackable ? Context.MaxItemStackCount : 1;
430
431 FInventoryItemInfo LootInfo;
432 if (GenerateLootItem(SyntheticRow, Context, i, EffectiveFloor, FinalLootBudget, LootInfo))
433 {
434 OutGeneratedLoot.Add(LootInfo);
435 }
436 }
437
438 return OutGeneratedLoot.Num() > 0;
439 }

442
443 for (const TObjectPtr<UInventoryItemDefinition>& ItemDef : Context.ItemDefinitions)
444 {
445 if (!ItemDef)

447 UE_LOG(LogDTLoot, Warning, TEXT("UDTLootSubsystem::GenerateLootFromContext - Encountered null item definition in context.ItemDefinitions"));

449 }
450

453
454 SyntheticRow.LootDefinition = ItemDef.Get();

459 SyntheticRow.MinStackCount = ItemDef->bStackable ? Context.MinItemStackCount : 1;
460 SyntheticRow.MaxStackCount = ItemDef->bStackable ? Context.MaxItemStackCount : 1;
461
462 CombinedRows.Add(&SyntheticRow);
463 CombinedWeights.Add(SyntheticRow.Weight);
464 }
465 }
466
467 for (const TObjectPtr<UDataTable>& LootTable : Context.LootTables)
468 {
469 if (!LootTable)
470 {
471 continue;
472 }
473
474 const TArray<FLootTableRow*>& TableRows = GetOrCacheLootTableRows(LootTable);
475
476 for (FLootTableRow* Row : TableRows)
477 {
478 if (Row && !Row->LootDefinition.IsNull())
479 {
480 CombinedRows.Add(Row);
481 CombinedWeights.Add(Row->Weight);
482 }
483 }
484 }
485
486 if (CombinedRows.Num() == 0)
487 {
488 UE_LOG(LogDTLoot, Warning, TEXT("UDTLootSubsystem::GenerateLootFromContext - No valid loot rows found in any table"));
489 return false;
490 }
491
492 int32 NumRolls = Context.MinLootRolls;
493 if (Context.MaxLootRolls > Context.MinLootRolls)
494 {
495 NumRolls = LootStream.RandRange(Context.MinLootRolls, Context.MaxLootRolls);
496 }
497
498 // SS AE - Special case to handle item regeneration. Must be evaluated against the BASE roll count,
499 // before loot-budget extra rolls are added, so single-item regeneration stays a single item.
500 const bool bSingleItemRegeneration = (Context.SeedOverride != 0 && NumRolls == 1);
501
502 // Loot budget only inflates quantity for loot-table-sourced drops. Explicit ItemDefinitions and
503 // PreGeneratedLoot are scripted/quest drops and must not be multiplied by floor scaling.
504 const bool bLootTableSourced = Context.LootTables.Num() > 0
505 && Context.ItemDefinitions.Num() == 0
506 && Context.PreGeneratedLoot.Num() == 0;
507
508 if (!bSingleItemRegeneration && bLootTableSourced)
509 {
510 const int32 ExtraLootRolls = RollExtraLootRolls(EffectiveFloor, FinalLootBudget);
511 NumRolls += ExtraLootRolls;
512
513 UE_LOG(LogDTLoot, Verbose, TEXT("UDTLootSubsystem::GenerateLootFromContext - LootBudget extra rolls: EffectiveFloor=%.2f, FinalLootBudget=%.4f, ExtraLootRolls=%d, NumRolls=%d"),
514 EffectiveFloor, FinalLootBudget, ExtraLootRolls, NumRolls);
515 }
516
517 for (int32 RollIndex = 0; RollIndex < NumRolls; ++RollIndex)
518 {
519 int32 SelectedIndex = SelectWeightedIndex(CombinedRows, CombinedWeights);
520
521 if (SelectedIndex >= 0 && SelectedIndex < CombinedRows.Num())
522 {
523 FLootTableRow* SelectedRow = CombinedRows[SelectedIndex];
524
525 if (SelectedRow)
526 {
527 FInventoryItemInfo LootInfo;
528
529 int32 EffectiveItemIndex = bSingleItemRegeneration ? -1 : RollIndex;
530
531 if (GenerateLootItem(*SelectedRow, Context, EffectiveItemIndex, EffectiveFloor, FinalLootBudget, LootInfo))
532 {
533 OutGeneratedLoot.Add(LootInfo);
534 UE_LOG(LogDTLoot, Verbose, TEXT("UDTLootSubsystem::GenerateLootFromContext - Roll %d succeeded, generated %s"), RollIndex, *GetNameSafe(LootInfo.ItemDefinition));
535 }
536 else
537 {
538 UE_LOG(LogDTLoot, Verbose, TEXT("UDTLootSubsystem::GenerateLootFromContext - Roll %d failed drop chance"), RollIndex);
539 }
540 }
541 }
542 }
543
544 return OutGeneratedLoot.Num() > 0;
545}

673 return GetCurrentDungeonFloor();

676float UDTLootSubsystem::CalculateEffectiveFloor(int32 RawFloor) const
677{
678 // Formula: IF Floor ≤ SoftCap THEN Floor ELSE SoftCap + (Floor − SoftCap)^DR
679
680 const int32 SoftCap = CachedGlobalLootData->GlobalFloorSoftCap;
681 const float DR = CachedGlobalLootData->GlobalFloorDR;
682
683 if (RawFloor <= SoftCap)
684 {
685 return static_cast<float>(RawFloor);
686 }
687
688 const float ExcessFloor = static_cast<float>(RawFloor - SoftCap);
689 const float EffectiveFloor = static_cast<float>(SoftCap) + FMath::Pow(ExcessFloor, DR);
690
691 UE_LOG(LogDTLoot, Verbose, TEXT("CalculateEffectiveFloor: Raw=%d, SoftCap=%d, DR=%.2f, Effective=%.2f"),
692 RawFloor, SoftCap, DR, EffectiveFloor);
693
694 return EffectiveFloor;
695}

697float UDTLootSubsystem::CalculateItemPower(float EffectiveFloor) const
698{
699 // Formula: MinItemPower + (1 - MinItemPower) * (1 - EXP(-EffectiveFloor / GlobalItemPowerScalingRate))
700
701 const float ScalingRate = CachedGlobalLootData->GlobalItemPowerScalingRate;
702 const float MinItemPower = CachedGlobalLootData->MinItemPower;
703 const float Curve = 1.0f - FMath::Exp(-EffectiveFloor / ScalingRate);
704 const float ItemPower = MinItemPower + ((1.0f - MinItemPower) * Curve);
705
706 UE_LOG(LogDTLoot, Verbose, TEXT("CalculateItemPower: EffectiveFloor=%.2f, ScalingRate=%.2f, MinItemPower=%.4f, ItemPower=%.4f"),
707 EffectiveFloor, ScalingRate, MinItemPower, ItemPower);
708
709 return ItemPower;
710}

712float UDTLootSubsystem::CalculateLootBudget(float EffectiveFloor) const
713{
714 // Formula: MinLoot + (1 - MinLoot) * (1 - EXP(-EffectiveFloor / LootBudgetScalingRate))
715
716 const float ScalingRate = CachedGlobalLootData->LootBudgetScalingRate;
717 const float MinLoot = CachedGlobalLootData->MinLoot;
718 const float Curve = 1.0f - FMath::Exp(-EffectiveFloor / ScalingRate);
719 const float LootBudget = MinLoot + ((1.0f - MinLoot) * Curve);
720
721 UE_LOG(LogDTLoot, Verbose, TEXT("CalculateLootBudget: EffectiveFloor=%.2f, ScalingRate=%.2f, MinLoot=%.4f, LootBudget=%.4f"),
722 EffectiveFloor, ScalingRate, MinLoot, LootBudget);
723
724 return LootBudget;
725}

779int32 UDTLootSubsystem::RollExtraLootRolls(float EffectiveFloor, float FinalLootBudget) const
780{
781 // Integer part is guaranteed; fractional part is a drop chance for one more.
782 // e.g. 0.5 -> 50% of 1 extra; 1.5 -> 1 guaranteed + 50% of a 2nd. One LootStream draw (deterministic).
783 const float Expected = GetExpectedExtraLootRolls(EffectiveFloor, FinalLootBudget);
784
785 int32 ExtraLootRolls = FMath::FloorToInt32(Expected);
786 const float Fraction = Expected - static_cast<float>(ExtraLootRolls);
787 const float Roll = LootStream.FRand();
788 if (Fraction > 0.0f && Roll < Fraction)
789 {
790 ++ExtraLootRolls;
791 }
792
793 ExtraLootRolls = FMath::Min(ExtraLootRolls, CachedGlobalLootData->MaxExtraLootRolls);
794
795 UE_LOG(LogDTLoot, Verbose, TEXT("RollExtraLootRolls: EffectiveFloor=%.2f, FinalLootBudget=%.4f, Expected=%.3f, Fraction=%.3f, Roll=%.3f, ExtraLootRolls=%d (Max=%d)"),
796 EffectiveFloor, FinalLootBudget, Expected, Fraction, Roll, ExtraLootRolls, CachedGlobalLootData->MaxExtraLootRolls);
797
798 return ExtraLootRolls;
799}

831float UDTLootSubsystem::RollAffixTierAlpha(float EffectiveFloor, float ItemPower) const
832{
833 const int32 TierCount = FMath::Max(2, CachedGlobalLootData->AffixTierCount);
834
835 // Axis 1 - tier roll (luck). Weight per tier strictly decreases (exp falloff), so the top tier is
836 // always the rarest. Depth raises FloorPressure, flattening the falloff so higher tiers get likelier;
837 // MaxFloorPressure caps the flattening so top tiers stay rare even deep.
838 const float FloorPressure = FMath::Clamp(EffectiveFloor / CachedGlobalLootData->AffixTierFloorScale, 0.0f, CachedGlobalLootData->MaxFloorPressure);
839 const float Decay = CachedGlobalLootData->BaseTierDecay * (1.0f - FloorPressure);
840
841 float TotalWeight = 0.0f;
842 TArray<float, TInlineAllocator<16>> CumulativeWeights;
843 CumulativeWeights.Reserve(TierCount);
844 for (int32 Tier = 1; Tier <= TierCount; ++Tier)
845 {
846 TotalWeight += FMath::Exp(-static_cast<float>(Tier - 1) * Decay);
847 CumulativeWeights.Add(TotalWeight);
848 }
849
850 // Weighted pick (single LootStream draw -> deterministic for regeneration).
851 const float Roll = LootStream.FRandRange(0.0f, TotalWeight);
852 int32 RolledTier = TierCount;
853 for (int32 i = 0; i < CumulativeWeights.Num(); ++i)
854 {
855 if (Roll <= CumulativeWeights[i])
856 {
857 RolledTier = i + 1;
858 break;
859 }
860 }
861
862 const float TierFraction = static_cast<float>(RolledTier - 1) / static_cast<float>(TierCount - 1);
863
864 // Axis 2 - floor magnitude scale. ItemPower scales the tier's value down at low floors (no one-shot)
865 // and up with depth. Stored as the affix's NormalizedRollValue; equip-time Lerp(Min,Max,alpha) is unchanged.
866 const float EffectiveAlpha = ItemPower * TierFraction;
867
868 UE_LOG(LogDTLoot, Verbose, TEXT("RollAffixTierAlpha: EffectiveFloor=%.2f, ItemPower=%.4f, FloorPressure=%.3f, Decay=%.3f, RolledTier=%d/%d, TierFraction=%.3f, EffectiveAlpha=%.4f"),
869 EffectiveFloor, ItemPower, FloorPressure, Decay, RolledTier, TierCount, TierFraction, EffectiveAlpha);
870
871 return EffectiveAlpha;
872}

940

944 }
945

965 for (int32 i = 0; i < NumItems; ++i)

971 ADTItemActor* SpawnedLoot = SpawnSingleLoot(Loot[i], ItemSpawnLocation, OwningPlayer);

974 SpawnedActors.Add(SpawnedLoot);
975 }
976 }

1069int32 UDTLootSubsystem::RollForcedBaseStats(FInventoryItemInfo& ItemInfo, TArray<FBaseStatTableRow*>& StatPool, TArray<float>& Weights, float ItemPower, float RarityStatMultiplier)
1070{
1071 int32 NumRolled = 0;
1072
1073 const UInventoryItemDefinition* ItemDef = ItemInfo.ItemDefinition;
1074 const UDTInventoryItemFragment_Equipment* EquipFragment = ItemDef ? ItemDef->FindFragment<UDTInventoryItemFragment_Equipment>() : nullptr;
1075 if (!EquipFragment || EquipFragment->ForcedBaseStats.Num() == 0)
1076 {
1077 return NumRolled;
1078 }
1079
1080 for (const FItemStatEntry& ForcedEntry : EquipFragment->ForcedBaseStats)
1081 {
1082 if (!ForcedEntry.StatTag.IsValid())
1083 {
1084 continue;
1085 }
1086
1087 // Find this stat in the pool so we can remove it from random rolling
1088 int32 ForcedStatIndex = INDEX_NONE;
1089 for (int32 i = 0; i < StatPool.Num(); ++i)
1090 {
1091 if (StatPool[i]->StatTag == ForcedEntry.StatTag)
1092 {
1093 ForcedStatIndex = i;
1094 break;
1095 }
1096 }
1097
1098 if (ForcedEntry.StatRollValue != 0.0f)
1099 {
1100 // Value pre-assigned — use it directly without rolling
1101 ItemInfo.BaseStats.Add(FItemStatEntry(ForcedEntry.StatTag, ForcedEntry.StatRollValue));
1102 NumRolled++;
1103
1104 UE_LOG(LogDTLoot, Verbose, TEXT("RollForcedBaseStats: Applied pre-assigned value %.4f for stat %s"),
1105 ForcedEntry.StatRollValue, *ForcedEntry.StatTag.ToString());
1106 }
1107 else
1108 {
1109 // StatRollValue is 0 — roll from table
1110 if (ForcedStatIndex == INDEX_NONE)
1111 {
1112 UE_LOG(LogDTLoot, Warning, TEXT("RollForcedBaseStats: Forced stat %s not found in stat pool and has no pre-assigned value"), *ForcedEntry.StatTag.ToString());
1113 continue;
1114 }
1115
1116 CalculateStatValue(ItemInfo, StatPool[ForcedStatIndex], ItemPower, RarityStatMultiplier);
1117 NumRolled++;
1118
1119 UE_LOG(LogDTLoot, Verbose, TEXT("RollForcedBaseStats: Rolled stat %s from table"), *ForcedEntry.StatTag.ToString());
1120 }
1121
1122 // Remove from pool regardless so it can't be double-rolled
1123 if (ForcedStatIndex != INDEX_NONE)
1124 {
1125 StatPool.RemoveAt(ForcedStatIndex);
1126 Weights.RemoveAt(ForcedStatIndex);
1127 }
1128 }
1129
1130 return NumRolled;
1131}

1133void UDTLootSubsystem::CalculateStatValue(
1134 FInventoryItemInfo& ItemInfo,
1135 const FBaseStatTableRow* StatRow,
1136 float ItemPower,
1137 float RarityStatMultiplier)
1138{
1139 if (!StatRow)
1140 {
1141 return;
1142 }
1143
1144 const float BaseStatMin = StatRow->MinValue;
1145 const float BaseStatMax = StatRow->MaxValue;
1146 const float Range = BaseStatMax - BaseStatMin;
1147
1148 // A zero or inverted range (Min >= Max) is a data-config error. Don't roll a random value — store the
1149 // fixed Min (normalized 0) deterministically and warn so the bad stat row gets noticed and fixed.
1150 if (Range <= 0.0f)
1151 {
1152 UE_LOG(LogDTLoot, Warning, TEXT("CalculateStatValue: stat %s has an invalid range (Min=%.2f >= Max=%.2f); storing Min with no roll. Fix the stat table row."),
1153 *StatRow->StatTag.ToString(), BaseStatMin, BaseStatMax);
1154 ItemInfo.BaseStats.Add(FItemStatEntry(StatRow->StatTag, 0.0f));
1155 return;
1156 }
1157
1158 // ------------------------------------------------------------
1159 // SCALED WINDOW + LINEAR ROLL (with built-in minimum)
1160 // ------------------------------------------------------------
1161 // ItemPower scales the top of the achievable window; MinStatScale floors the bottom of the
1162 // window relative to that top, so even low rolls retain some strength.
1163 const float MinStatScale = CachedGlobalLootData->MinStatScale;
1164 const float ScaledMax = BaseStatMin + (Range * ItemPower);
1165 const float ScaledMin = BaseStatMin + (Range * ItemPower * MinStatScale);
1166
1167 const float RollFactor = CalculateRollFactor();
1168 const float FinalStatRoll = ScaledMin + (RollFactor * (ScaledMax - ScaledMin));
1169
1170 // Rarity scales the whole rolled value (e.g. Legendary 1.4 = +40%).
1171 const float ScaledStatRoll = FinalStatRoll * RarityStatMultiplier;
1172
1173 // BaseStats store a normalized roll that CalculateActualStatValue re-lerps against MinValue/MaxValue.
1174 // Only the LOWER bound is clamped: a rarity multiplier > 1 intentionally yields a normalized value
1175 // above 1.0 so the re-lerp extrapolates PAST the stat's table Max (rarity genuinely exceeds the base
1176 // ceiling). With no rarity bonus (multiplier 1.0) ScaledStatRoll <= ScaledMax <= Max, so the value
1177 // stays within [0,1] exactly as before. (Range > 0 is guaranteed by the early-out above.)
1178 const float NormalizedValue = FMath::Max(0.0f, (ScaledStatRoll - BaseStatMin) / Range);
1179
1180 ItemInfo.BaseStats.Add(FItemStatEntry(StatRow->StatTag, NormalizedValue));
1181
1182 UE_LOG(LogDTLoot, Verbose,
1183 TEXT("Rolled base stat %s: FinalStatRoll=%.2f, RarityMult=%.2f, ScaledStatRoll=%.2f, NormalizedValue=%.4f (scaledMin=%.2f, scaledMax=%.2f, itemPower=%.4f, minStatScale=%.2f, rollFactor=%.4f)"),
1184 *StatRow->StatTag.ToString(),
1185 FinalStatRoll,
1186 RarityStatMultiplier,
1187 ScaledStatRoll,
1188 NormalizedValue,
1189 ScaledMin,
1190 ScaledMax,
1191 ItemPower,
1192 MinStatScale,
1193 RollFactor);
1194}

1370 {

1430void UDTLootSubsystem::CacheRarityWeights()
1431{
1432 CachedRarityWeights.Empty();
1433
1434 if (!CachedGlobalLootData)
1435 {
1436 return;
1437 }
1438
1439 // Canonical selection order, lowest rarity first.
1440 static const TArray<FGameplayTag> RarityTagOrder = {
1441 DTGameplayTags::TAG_ITEM_RARITY_COMMON,
1442 DTGameplayTags::TAG_ITEM_RARITY_UNCOMMON,
1443 DTGameplayTags::TAG_ITEM_RARITY_RARE,
1444 DTGameplayTags::TAG_ITEM_RARITY_EPIC,
1445 DTGameplayTags::TAG_ITEM_RARITY_LEGENDARY,
1446 DTGameplayTags::TAG_ITEM_RARITY_ARTIFACT
1447 };
1448
1449 CachedRarityWeights.Reserve(RarityTagOrder.Num());
1450
1451 for (const FGameplayTag& RarityTag : RarityTagOrder)
1452 {
1453 // Validate against the raw map so we can warn about unconfigured tiers (GetRarityData would
1454 // silently return a default with base rate 1, collapsing selection toward a uniform spread).
1455 const FLootRarityData* Data = CachedGlobalLootData->LootRarityDataMap.Find(RarityTag);
1456 if (!Data)
1457 {
1458 UE_LOG(LogDTLoot, Warning, TEXT("CacheRarityWeights: Rarity %s is missing from LootRarityDataMap - using default weight (rarity distribution will be skewed)."),
1459 *RarityTag.ToString());
1460 }
1461 else if (Data->GlobalBaseRate <= 0.0f)
1462 {
1463 UE_LOG(LogDTLoot, Warning, TEXT("CacheRarityWeights: Rarity %s has a non-positive GlobalBaseRate (%.2f) and will effectively never drop."),
1464 *RarityTag.ToString(), Data->GlobalBaseRate);
1465 }
1466
1467 FCachedRarityWeight Cached;
1468 Cached.RarityTag = RarityTag;
1469 Cached.GlobalBaseRate = Data ? Data->GlobalBaseRate : 1.0f;
1470 Cached.GlobalWeightScalar = Data ? Data->GlobalWeightScalar : 0.0f;
1471 Cached.StatMultiplier = Data ? Data->StatMultiplier : 1.0f;
1472 CachedRarityWeights.Add(Cached);
1473 }
1474
1475 // Detect the "unconfigured" case: identical base rates and zero scalars across all tiers means the
1476 // rarity fields are still at their struct defaults (e.g. the asset predates these fields), which
1477 // produces a UNIFORM distribution - every rarity, including Legendary/Artifact, drops ~equally.
1478 bool bAllScalarsZero = true;
1479 bool bAllBaseRatesEqual = true;
1480 const float FirstBaseRate = CachedRarityWeights.Num() > 0 ? CachedRarityWeights[0].GlobalBaseRate : 0.0f;
1481 for (const FCachedRarityWeight& Rarity : CachedRarityWeights)
1482 {
1483 if (!FMath::IsNearlyZero(Rarity.GlobalWeightScalar))
1484 {
1485 bAllScalarsZero = false;
1486 }
1487 if (!FMath::IsNearlyEqual(Rarity.GlobalBaseRate, FirstBaseRate))
1488 {
1489 bAllBaseRatesEqual = false;
1490 }
1491 }
1492
1493 if (bAllScalarsZero && bAllBaseRatesEqual)
1494 {
1495 UE_LOG(LogDTLoot, Warning, TEXT("CacheRarityWeights: All rarities share GlobalBaseRate=%.2f and GlobalWeightScalar=0 - rarity weighting is UNCONFIGURED and will be UNIFORM (~%.0f%% each, including Legendary/Artifact). Set per-rarity GlobalBaseRate/GlobalWeightScalar on the loot data asset."),
1496 FirstBaseRate, 100.0f / FMath::Max(1, CachedRarityWeights.Num()));
1497 }
1498
1499 // Resolved weights at budget 0 (floor 1) and budget 1 (deep) so the configured curve is visible in logs.
1500 for (const FCachedRarityWeight& Rarity : CachedRarityWeights)
1501 {
1502 const float WeightAtZero = Rarity.GlobalBaseRate;
1503 const float WeightAtOne = Rarity.GlobalBaseRate * (1.0f + Rarity.GlobalWeightScalar);
1504 UE_LOG(LogDTLoot, Log, TEXT("CacheRarityWeights: %s baseRate=%.2f scalar=%.2f statMult=%.2f -> weight@budget0=%.2f, weight@budget1=%.2f"),
1505 *Rarity.RarityTag.ToString(), Rarity.GlobalBaseRate, Rarity.GlobalWeightScalar, Rarity.StatMultiplier, WeightAtZero, WeightAtOne);
1506
1507 if (Rarity.StatMultiplier <= 0.0f)
1508 {
1509 UE_LOG(LogDTLoot, Warning, TEXT("CacheRarityWeights: Rarity %s has a non-positive StatMultiplier (%.2f) - base stats will roll to zero."),
1510 *Rarity.RarityTag.ToString(), Rarity.StatMultiplier);
1511 }
1512 }
1513}
