49void UDTAbilityTooltipWidget::RefreshTooltip()
50{
51 if (TXT_AbilityName)
52 {
53 TXT_AbilityName->SetText(GetTooltipTitle());
54 }
55
56 if (RTXT_Description)
57 {
58 RTXT_Description->SetText(GetTooltipDescription());
59 }
60
61 if (Img_Icon)
62 {
63 UTexture2D* Icon = GetTooltipIcon();
64 Img_Icon->SetBrushFromTexture(Icon);
65 Img_Icon->SetVisibility(Icon ? ESlateVisibility::HitTestInvisible : ESlateVisibility::Collapsed);
66 }
67
68 OnTooltipDataUpdated();
69}
