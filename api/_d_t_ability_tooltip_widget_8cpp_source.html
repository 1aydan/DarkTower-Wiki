138const FGameplayAbilitySpec* UDTAbilityTooltipWidget::FindGrantedSpec() const
139{
140 const UAbilitySystemComponent* ASC = GetPlayerASC();
141 if (!ASC)
142 {
143 return nullptr;
144 }
145
146 const UClass* AbilityClass = SourceAbility ? SourceAbility->GetClass() : nullptr;
147 const FGameplayTag AbilityTag = AbilityDefinition ? AbilityDefinition->AbilityTag : FGameplayTag();
148
149 for (const FGameplayAbilitySpec& Spec : ASC->GetActivatableAbilities())
150 {
151 const UDTGameplayAbility* SpecAbility = Cast<UDTGameplayAbility>(Spec.Ability);
152 if (!SpecAbility)
153 {
154 continue;
155 }
156
157 const bool bClassMatch = AbilityClass && SpecAbility->GetClass() == AbilityClass;
158 const bool bTagMatch = AbilityTag.IsValid() && SpecAbility->GetAbilityDefinition() && SpecAbility->GetAbilityDefinition()->AbilityTag == AbilityTag;
159 if (bClassMatch || bTagMatch)
160 {
161 return &Spec;
162 }
163 }
164
165 return nullptr;
166}

222float UDTAbilityTooltipWidget::GetManaCostAtLevel(int32 Level) const
223{
224 UAbilitySystemComponent* ASC = GetPlayerASC();
225 const UGameplayEffect* CostGE = SourceAbility ? SourceAbility->GetCostGameplayEffect() : nullptr;
226 if (!ASC || !CostGE)
227 {
228 return 0.0f;
229 }
230
231 FGameplayEffectSpec CostSpec(CostGE, ASC->MakeEffectContext(), static_cast<float>(Level));
232 CostSpec.CalculateModifierMagnitudes();
233
234 const FGameplayAttribute ManaAttribute = UDTAttributeSet::GetManaAttribute();
235 for (int32 ModIdx = 0; ModIdx < CostSpec.Modifiers.Num(); ++ModIdx)
236 {
237 if (CostSpec.Def && CostSpec.Def->Modifiers.IsValidIndex(ModIdx) && CostSpec.Def->Modifiers[ModIdx].Attribute == ManaAttribute)
238 {
239 return FMath::Abs(CostSpec.GetModifierMagnitude(ModIdx));
240 }
241 }
242
243 return 0.0f;
244}

276FText UDTAbilityTooltipWidget::GetFriendlyTagName(FGameplayTag Tag) const
277{
278 if (!Tag.IsValid())
279 {
280 return FText::GetEmpty();
281 }
282
283 if (const FText* Override = TagDisplayNames.Find(Tag))
284 {
285 return *Override;
286 }
287
288 // Fall back to a humanized version of the tag's leaf segment (e.g. Equipment.Weapon.TwoHandedBow -> "Two Handed Bow").
289 const FString TagString = Tag.GetTagName().ToString();
290 FString Leaf = TagString;
291 int32 DotIndex = INDEX_NONE;
292 if (TagString.FindLastChar(TEXT('.'), DotIndex))
293 {
294 Leaf = TagString.RightChop(DotIndex + 1);
295 }
296
297 return FText::FromString(FName::NameToDisplayString(Leaf, /*bIsBool*/ false));
298}

311FText UDTAbilityTooltipWidget::FormatValueComparison(const FText& Format, float CurrentValue, float NextValue) const
312{
313 // Value unchanged across the upgrade (e.g. a flat cooldown): show a single value, empty when zero.
314 if (FMath::IsNearlyEqual(CurrentValue, NextValue))
315 {
316 if (CurrentValue <= 0.0f)
317 {
318 return FText::GetEmpty();
319 }
320
321 return FText::Format(Format, FText::AsNumber(CurrentValue));
322 }
323
324 // Real change: render "Current -> Next" in the value slot, matching the description's upgrade arrow.
325 const FText Comparison = FText::Format(LOCTEXT("ValueUpgradeFormat", "{0} -> {1}"), FText::AsNumber(CurrentValue), FText::AsNumber(NextValue));
326 return FText::Format(Format, Comparison);
327}

372FText UDTAbilityTooltipWidget::GetTooltipRequirementsText() const
373{
374 const FGameplayTagContainer Requirements = GetTooltipRequirements();
375 if (Requirements.IsEmpty())
376 {
377 return FText::GetEmpty();
378 }
379
380 TArray<FText> RequirementNames;
381 RequirementNames.Reserve(Requirements.Num());
382 for (const FGameplayTag& RequirementTag : Requirements)
383 {
384 RequirementNames.Add(GetFriendlyTagName(RequirementTag));
385 }
386
387 return FText::Format(RequirementsFormat, FText::Join(FText::FromString(TEXT(", ")), RequirementNames));
