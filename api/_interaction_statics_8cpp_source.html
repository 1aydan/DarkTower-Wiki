99int32 UInteractionStatics::SelectBestInteractableIndex(AActor* Interactor, const TArray<AActor*>& Candidates, const TArray<float>& Priorities)
100{
101 if (!Interactor || Candidates.Num() == 0)
102 {
103 return INDEX_NONE;
104 }
105
106 if (Candidates.Num() == 1)
107 {
108 return 0;
109 }
110
111 const ADTPlayerCharacter* PlayerCharacter = Cast<ADTPlayerCharacter>(Interactor);
112 const bool bMouse = PlayerCharacter && PlayerCharacter->IsUsingMouseAim();
113
114 if (bMouse)
115 {
116 if (APlayerController* PlayerController = Cast<APlayerController>(PlayerCharacter->GetController()))
117 {
118 FHitResult CursorHit;
119 if (PlayerController->GetHitResultUnderCursor(DT_TraceChannel_Interaction, false, CursorHit) && CursorHit.bBlockingHit)
120 {
121 const int32 HitIndex = Candidates.IndexOfByKey(CursorHit.GetActor());
122 if (HitIndex != INDEX_NONE)
123 {
124 return HitIndex;
125 }
126 }
127 }
128
129 // Otherwise the candidate nearest the cursor's ground point.
130 const FVector CursorPoint = PlayerCharacter->GetAimGroundLocation();
131 int32 BestIndex = 0;
132 float BestDistSq = TNumericLimits<float>::Max();
133 for (int32 Index = 0; Index < Candidates.Num(); ++Index)
134 {
135 if (!Candidates[Index])
136 {
137 continue;
138 }
139
140 const float DistSq = FVector::DistSquared(Candidates[Index]->GetActorLocation(), CursorPoint);
141 if (DistSq < BestDistSq)
142 {
143 BestDistSq = DistSq;
144 BestIndex = Index;
145 }
146 }
147
148 return BestIndex;
149 }
150
151 // Gamepad / non-mouse: highest priority, then facing (in front beats behind), then nearest.
152 const FVector Origin = Interactor->GetActorLocation();
153 const FVector Forward = Interactor->GetActorForwardVector().GetSafeNormal2D();
154
155 int32 BestIndex = INDEX_NONE;
156 float BestPriority = -TNumericLimits<float>::Max();
157 bool bBestInFront = false;
158 float BestDistSq = TNumericLimits<float>::Max();
159 for (int32 Index = 0; Index < Candidates.Num(); ++Index)
160 {
161 if (!Candidates[Index])
162 {
163 continue;
164 }
165
166 const FVector ToTarget = Candidates[Index]->GetActorLocation() - Origin;
167 const float DistSq = ToTarget.SizeSquared();
168 const bool bInFront = FVector::DotProduct(ToTarget.GetSafeNormal2D(), Forward) >= 0.0f;
169 const float Priority = Priorities.IsValidIndex(Index) ? Priorities[Index] : 0.0f;
170
171 bool bBetter;
172 if (BestIndex == INDEX_NONE)
173 {
174 bBetter = true;
175 }
176 else if (Priority != BestPriority)
177 {
178 bBetter = Priority > BestPriority;
179 }
180 else if (bInFront != bBestInFront)
181 {
182 bBetter = bInFront;
183 }
184 else
185 {
186 bBetter = DistSq < BestDistSq;
187 }
188
189 if (bBetter)
190 {
191 BestIndex = Index;
192 BestPriority = Priority;
193 bBestInFront = bInFront;
194 BestDistSq = DistSq;
195 }
196 }
197 return BestIndex;
198}
