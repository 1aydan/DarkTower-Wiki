111void UDTAbilityBarWidget::OnAbilitySlotChanged(FGameplayTag SlotTag, FGameplayAbilitySpecHandle NewAbility, FGameplayAbilitySpecHandle OldAbility)
112{
113 if (!WB_AbilityBar || !AbilitySlotComponent.Get())
114 {
115 UE_LOG(LogDTUI, Warning, TEXT("DTAbilityBarWidget::OnAbilitySlotChanged - Missing WB_AbilityBar or AbilitySlotComponent"));
116 return;
117 }
118
119 // Get the updated slot
120 FAbilitySlot UpdatedSlot;
121 if (!AbilitySlotComponent->GetAbilitySlot(SlotTag, UpdatedSlot))
122 {
123 UE_LOG(LogDTUI, Warning, TEXT("DTAbilityBarWidget::OnAbilitySlotChanged - Could not find slot for tag %s"), *SlotTag.ToString());
124 return;
125 }
126
127 for (int32 i = 0; i < WB_AbilityBar->GetChildrenCount(); ++i)
128 {
129 UDTAbilityWidget* AbilityWidget = Cast<UDTAbilityWidget>(WB_AbilityBar->GetChildAt(i));
130 if (AbilityWidget && AbilityWidget->GetSlotTag() == SlotTag)
131 {
132 UE_LOG(LogDTUI, Verbose, TEXT("DTAbilityBarWidget::OnAbilitySlotChanged - Found widget at index %d for slot %s"), i, *SlotTag.ToString());
133
134 InitializeAbilityWidget(AbilityWidget, UpdatedSlot);
135 return;
136 }
137 }
138
139 UE_LOG(LogDTUI, Verbose, TEXT("DTAbilityBarWidget::OnAbilitySlotChanged - Widget not found for slot %s, doing full refresh"), *SlotTag.ToString());
140
141 RefreshAbilityBar();
142}

151void UDTAbilityBarWidget::CreateSlotWidgets()
152{
153 if (!AbilitySlotComponent.Get())
154 {
155 UE_LOG(LogDTUI, Warning, TEXT("DTAbilityBarWidget::CreateSlotWidgets - Missing required component(s)"));
156 return;
157 }
158
159 ClearAbilityWidgets();
160
161 TArray<FAbilitySlot> MatchingSlots;
162 if (SlotFilterTags.Num() > 0)
163 {
164 MatchingSlots = AbilitySlotComponent->GetSlotsByTags(SlotFilterTags, bRequireAllFilterTags);
165 }
166 else
167 {
168 MatchingSlots = AbilitySlotComponent->GetAbilitySlots();
169 }
170
171 UE_LOG(LogDTUI, VeryVerbose, TEXT("DTAbilityBarWidget::CreateSlotWidgets - Creating %d slot widgets"), MatchingSlots.Num());
172
173 for (const FAbilitySlot& AbilitySlot : MatchingSlots)
174 {
175 UDTAbilityWidget* AbilityWidget = Cast<UDTAbilityWidget>(CreateWidget(GetOwningPlayer(), AbilityButtonWidgetClass));
176
177 if (AbilityWidget)
178 {
179 InitializeAbilityWidget(AbilityWidget, AbilitySlot);
180
181 WB_AbilityBar->AddChildToWrapBox(AbilityWidget);
182 }
183 }
