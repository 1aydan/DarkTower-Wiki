9int64 UDTProgressionData::GetXPToLevel(const int32 Level) const
10{
11 const int32 ClampedLevel = FMath::Max(1, Level);
12 const double ExponentialWeight = FMath::Max(0.0, static_cast<double>(LevelCurveExponentialWeight));
13 const double ExponentialBase = FMath::Max(0.0001, static_cast<double>(LevelCurveExponentialBase));
14 const double ExponentialExponentDivisor = FMath::Max(0.0001, static_cast<double>(LevelCurveExponentDivisor));
15 const int64 TotalXPDivisor = static_cast<int64>(FMath::Max(1, LevelCurveTotalXPDivisor));
16
17 int64 SummedTerms = 0;
18
19 for (int32 N = 1; N < ClampedLevel; ++N)
20 {
21 const double Term = static_cast<double>(N)
22 + ExponentialWeight
23 * FMath::Pow(ExponentialBase, static_cast<double>(N) / ExponentialExponentDivisor);
24
25 if (Term >= static_cast<double>(MAX_int64))
26 {
27 return MAX_int64;
28 }
29
30 const int64 FlooredTerm = FMath::FloorToInt64(Term);
31 if (SummedTerms > MAX_int64 - FlooredTerm)
32 {
33 return MAX_int64;
34 }
35
36 SummedTerms += FlooredTerm;
37 }
38
39 return SummedTerms / TotalXPDivisor;
40}

42int64 UDTProgressionData::GetXPToNextLevel(const int32 CurrentLevel) const
43{
44 const int32 ClampedLevel = FMath::Max(1, CurrentLevel);
45 const int64 CurrentTotalXP = GetXPToLevel(ClampedLevel);
46 const int64 NextTotalXP = GetXPToLevel(ClampedLevel + 1);
47 return FMath::Max<int64>(1, NextTotalXP - CurrentTotalXP);
48}

89FDTDungeonSettings UDTProgressionData::BuildDungeonSettingsFromConfig(const FDTHostConfig& HostConfig) const
90{
91 FDTDungeonSettings DungeonSettings;
92
93 DungeonSettings.ModeDefinitionTag = HostConfig.ModeDefinitionTag;
94 DungeonSettings.bMortalMode = HostConfig.bMortalMode;
95 DungeonSettings.GlobalFloor = HostConfig.StartingFloor;
96 DungeonSettings.BiomeFloor = 1;
97 DungeonSettings.DungeonSeed = HostConfig.DungeonSeed;
98
99 if (HostConfig.ModeDefinitionTag.MatchesTag(DTGameplayTags::TAG_GAMEMODE_CAMPAIGN))
100 {
101 const UDTCampaignDefinition* CampaignData = FindGameModeDefinitionOfType<UDTCampaignDefinition>(HostConfig.ModeDefinitionTag);
102 if (CampaignData)
103 {
104 ApplyCampaignSettings(CampaignData, HostConfig, DungeonSettings);
105 }
106 }
107 else if (HostConfig.ModeDefinitionTag.MatchesTag(DTGameplayTags::TAG_GAMEMODE_ENDGAME))
108 {
109 const UDTEndgameDefinition* EndgameDefinition = FindGameModeDefinitionOfType<UDTEndgameDefinition>(HostConfig.ModeDefinitionTag);
110 if (EndgameDefinition)
111 {
112 ApplyEndgameSettings(EndgameDefinition, HostConfig, DungeonSettings);
113 }
114 }
115
116 return DungeonSettings;
117}

135{

150void UDTProgressionData::ApplyEndgameSettings(const UDTEndgameDefinition* EndgameDefinition, const FDTHostConfig& SanitizedConfig, FDTDungeonSettings& OutSettings)
151{
152 OutSettings.ModeDefinitionTag = EndgameDefinition->GameModeTag.IsValid() ? EndgameDefinition->GameModeTag : SanitizedConfig.ModeDefinitionTag;
153 OutSettings.ModeDefinitionGuid = EndgameDefinition->AssetGuid;
154 OutSettings.BiomeFloor = ((SanitizedConfig.StartingFloor - 1) % 10) + 1;
155 OutSettings.DungeonSeed = SanitizedConfig.DungeonSeed != 0 ? SanitizedConfig.DungeonSeed : SanitizedConfig.StartingFloor;
156
157 if (const UDTBiomeDefinition* EndgameBiome = EndgameDefinition->GetBiomeForFloor(SanitizedConfig.StartingFloor, OutSettings.DungeonSeed))
158 {
159 if (const FDTBiomeFloorDefinition* FloorDefinition = EndgameBiome->GetFloorDefinition(OutSettings.BiomeFloor))
160 {
161 OutSettings.TravelMapName = GetTravelMapName(*FloorDefinition);
162 OutSettings.FloorDisplayName = FloorDefinition->FloorDisplayName;
163 }
