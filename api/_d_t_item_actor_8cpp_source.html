50void ADTItemActor::BeginPlay()
51{
52 Super::BeginPlay();
53
54 if (HasAuthority())
55 {
56 if (VisibilityType == EItemVisibilityType::DroppedItem)
57 {
58 InitDroppedItem();
59 }
60 else
61 {
62 // SS AE - force for sanity
63 bOnlyRelevantToOwner = true;
64 }
65
66 if (VisibilityType == EItemVisibilityType::WorldPlaced)
67 {
68 InitWorldPlacedItem();
69 }
70
71 ItemCollisionComponent->OnComponentHit.AddDynamic(this, &ADTItemActor::OnItemHit);
72
73 // TODO : maybe make this optional?
74 SetItemPhysics(true);
75 }
76
77 // personal loot setup
78 if (VisibilityType == EItemVisibilityType::PersonalLoot)
79 {
80 APlayerController* LocalPC = GetWorld() ? GetWorld()->GetFirstPlayerController() : nullptr;
81 if (LocalPC && HasAuthority())
82 {
83 const AActor* OwnerActor = GetOwner();
84 const APawn* LocalPawn = LocalPC->GetPawn();
85
86 const bool bIsOwnedByLocalPlayer =
87 (OwnerActor == LocalPC) ||
88 (LocalPawn && OwnerActor == LocalPawn) ||
89 (OwnerActor && LocalPawn && OwnerActor->IsOwnedBy(LocalPawn)) ||
90 (OwnerActor && OwnerActor->IsOwnedBy(LocalPC));
91
92 if (!bIsOwnedByLocalPlayer)
93 {
94 if (ItemMeshComponent)
95 {
96 ItemMeshComponent->SetVisibility(false, false);
97 }
98
99 if (ItemSkeletalMeshComponent)
100 {
101 ItemSkeletalMeshComponent->SetVisibility(false, false);
102 }
103
104 if (ItemParticleComponent)
105 {
106 ItemParticleComponent->SetVisibility(false, false);
107 }
108
109 return;
110 }
111
112 OwningPlayer = LocalPC;
113 }
114 }
115
116 if (ItemInfo.ItemDefinition)
117 {
118 SetItemVisuals();
119 }
120
121 TryRegisterLoot();
122}

159void ADTItemActor::ApplyItemVisuals()
160{
161 if (!ItemInfo.ItemDefinition)
162 {
163 return;
164 }
165
166 const UInventoryItemFragment_World* WorldFragment = ItemInfo.ItemDefinition->FindFragment<UInventoryItemFragment_World>();
167 if (!WorldFragment)
168 {
169 return;
170 }
171
172 if (!WorldFragment->StaticMesh.IsNull())
173 {
174 if (UStaticMesh* LoadedMesh = WorldFragment->StaticMesh.LoadSynchronous())
175 {
176 ItemMeshComponent->SetStaticMesh(LoadedMesh);
177 ItemMeshComponent->SetRelativeTransform(WorldFragment->VisualTransformOffset);
178 ItemMeshComponent->SetVisibility(true);
179 ItemMeshComponent->SetHiddenInGame(false);
180 ItemSkeletalMeshComponent->SetVisibility(false);
181 }
182 }
183 else if (!WorldFragment->SkeletalMesh.IsNull())
184 {
185 if (USkeletalMesh* LoadedMesh = WorldFragment->SkeletalMesh.LoadSynchronous())
186 {
187 ItemSkeletalMeshComponent->SetSkeletalMesh(LoadedMesh);
188 ItemSkeletalMeshComponent->SetRelativeTransform(WorldFragment->VisualTransformOffset);
189 ItemSkeletalMeshComponent->SetVisibility(true);
190 ItemSkeletalMeshComponent->SetHiddenInGame(false);
191 ItemMeshComponent->SetVisibility(false);
192 }
193 }
194 else
195 {
196 ItemMeshComponent->SetVisibility(false);
197 ItemSkeletalMeshComponent->SetVisibility(false);
198 }
199
200 if (!WorldFragment->NiagaraEffect.IsNull())
201 {
202 if (UNiagaraSystem* LoadedEffect = WorldFragment->NiagaraEffect.LoadSynchronous())
203 {
204 ItemParticleComponent->SetRelativeTransform(WorldFragment->VisualTransformOffset);
205 ItemParticleComponent->SetAsset(LoadedEffect);
206 ItemParticleComponent->Activate();
207 }
208 }

335bool ADTItemActor::IsRelevantToInteractor(const AActor* Interactor) const
336{
337 if (!Interactor)
338 {
339 return false;
340 }
341
342 // Non-personal loot is interactable by anyone.
343 if (VisibilityType != EItemVisibilityType::PersonalLoot)
344 {
345 return true;
346 }
347
348 const AActor* OwnerActor = GetOwner();
349 if (!OwnerActor)
350 {
351 return false;
352 }
353
354 const APawn* InteractorPawn = Cast<APawn>(Interactor);
355 const AController* InteractorController = InteractorPawn ? InteractorPawn->GetController() : Cast<AController>(Interactor);
356
357 return (OwnerActor == Interactor) ||
358 (InteractorController && OwnerActor == InteractorController) ||
359 (InteractorPawn && OwnerActor->IsOwnedBy(InteractorPawn)) ||
360 (InteractorController && OwnerActor->IsOwnedBy(InteractorController));
361}
