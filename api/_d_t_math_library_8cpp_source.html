36int32 UDTMathLibrary::GenerateRandomSeed()
37{
38 // Use high-resolution time as entropy source
39 // FPlatformTime::Cycles64() provides a high-resolution timestamp
40 uint64 TimeSeed = FPlatformTime::Cycles64();
41
42 // Mix in additional entropy from FMath::Rand() to avoid identical seeds
43 // if called multiple times in the same cycle
44 TimeSeed ^= static_cast<uint64>(FMath::Rand()) << 32;
45
46 // Run through SplitMix64 to produce high-quality seed
47 uint64 Result = SplitMix64(TimeSeed);
48
49 // Return as 32-bit value (still vastly better distribution than FMath::Rand alone)
50 return static_cast<int32>(Result & 0xFFFFFFFF);
51}
