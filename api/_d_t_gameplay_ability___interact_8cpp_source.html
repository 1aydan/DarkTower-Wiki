51void UDTGameplayAbility_Interact::SetFocusedActor(AActor* NewFocus)
52{
53 AActor* OldFocus = FocusedActor.Get();
54 if (OldFocus == NewFocus)
55 {
56 return;
57 }
58
59 AActor* Avatar = GetAvatarActorFromActorInfo();
60 if (OldFocus && OldFocus->Implements<UDTInteractableTarget>())
61 {
62 IDTInteractableTarget::Execute_OnInteractionFocusLost(OldFocus, Avatar);
63 }
64 if (NewFocus && NewFocus->Implements<UDTInteractableTarget>())
65 {
66 IDTInteractableTarget::Execute_OnInteractionFocusGained(NewFocus, Avatar);
67 }
68 FocusedActor = NewFocus;
69
70 // Broadcast the focus change for HUD (interaction prompt / loot tooltip). Local player only - this
71 // path is already gated to the locally-controlled player by the callers.
72 FDTInteractionMessage Message;
73 Message.FocusedActor = NewFocus;
74 Message.bHasFocus = (NewFocus != nullptr);
75 Message.Prompt = (NewFocus && NewFocus->Implements<UDTInteractableTarget>()) ? IDTInteractableTarget::Execute_GetInteractionText(NewFocus) : FText::GetEmpty();
76 Message.Instigator = Cast<APawn>(Avatar);
77 UGameplayMessageSubsystem::Get(this).BroadcastMessage(DTGameplayTags::TAG_EVENT_HUD_INTERACTIONFOCUSCHANGED, Message);
78}

91void UDTGameplayAbility_Interact::TriggerInteraction()
92{
93 if (CurrentOptions.Num() == 0)
94 {
95 return;
96 }
97
98 UAbilitySystemComponent* AbilitySystem = GetAbilitySystemComponentFromActorInfo();
99 if (AbilitySystem)
100 {
101 const FInteractionOption& InteractionOption = CurrentOptions[0];
102
103 AActor* Instigator = GetAvatarActorFromActorInfo();
104 AActor* InteractableTargetActor = UInteractionStatics::GetActorFromInteractableTarget(InteractionOption.InteractableTarget);
105
106 // Direct-interact option (no backing ability system): call the interface directly and finish.
107 if (!InteractionOption.TargetAbilitySystem)
108 {
109 if (InteractableTargetActor && InteractableTargetActor->Implements<UDTInteractableTarget>() && IDTInteractableTarget::Execute_CanInteract(InteractableTargetActor, Instigator))
110 {
111 IDTInteractableTarget::Execute_Interact(InteractableTargetActor, Instigator);
112 }
113
114 return;
115 }
116
117 // Allow the target to customize the event data we're about to pass in, in case the ability needs custom data
118 // that only the actor knows.
119 FGameplayEventData Payload;
120 Payload.EventTag = TAG_Ability_Interaction_Activate;
121 Payload.Instigator = Instigator;
122 Payload.Target = InteractableTargetActor;
123
124 // If needed we allow the interactable target to manipulate the event data so that for example, a button on the wall
125 // may want to specify a door actor to execute the ability on, so it might choose to override Target to be the
126 // door actor.
127 if (UObject* CustomizeTarget = InteractionOption.InteractableTarget.GetObject())
128 {
129 IDTInteractableTarget::Execute_CustomizeInteractionEventData(CustomizeTarget, TAG_Ability_Interaction_Activate, Payload);
130 }
131
132 // Grab the target actor off the payload we're going to use it as the 'avatar' for the interaction, and the
133 // source InteractableTarget actor as the owner actor.
134 AActor* TargetActor = const_cast<AActor*>(ToRawPtr(Payload.Target));
135
136 // The actor info needed for the interaction.
137 FGameplayAbilityActorInfo ActorInfo;
138 ActorInfo.InitFromActor(InteractableTargetActor, TargetActor, InteractionOption.TargetAbilitySystem);
139
140 // Trigger the ability using event tag.
141 const bool bSuccess = InteractionOption.TargetAbilitySystem->TriggerAbilityFromGameplayEvent(
142 InteractionOption.TargetInteractionAbilityHandle,
143 &ActorInfo,
144 TAG_Ability_Interaction_Activate,
145 &Payload,
146 *InteractionOption.TargetAbilitySystem
147 );
148 }
