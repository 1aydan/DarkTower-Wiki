107void ADTPlayerCharacter::Tick(float DeltaTime)
108{
109 Super::Tick(DeltaTime);
110
111 if (!IsLocallyControlled() || !PlayerController)
112 {
113 return;
114 }
115
116 // Keep the cached aim point fresh for reticles / cast-at-location targeting, even while rotation is blocked.
117 UpdateAimLocation();
118
119 // TODO : Possibly rework this?
120 if (HasMatchingGameplayTag(DTGameplayTags::TAG_BLOCK_ROTATION))
121 {
122 UE_LOG(LogTemp, VeryVerbose, TEXT("ADTPlayerCharacter::Tick - Rotation blocked by TAG_BLOCK_ROTATION"));
123 return;
124 }
125
126 const FRotator OldRotation = GetActorRotation();
127
128 if (bUsingMouse)
129 {
130 // Face the cursor: derive the target yaw from the cached aim location and interp toward it.
131 const FRotator AimRot = UKismetMathLibrary::FindLookAtRotation(GetActorLocation(), CurrentAimLocation);
132 const float AngleDifference = FMath::FindDeltaAngleDegrees(AimAngle, AimRot.Yaw);
133 AimAngle = FMath::FInterpTo(AimAngle, AimAngle + AngleDifference, DeltaTime, AimRotationInterpSpeed);
134 }
135
136 PlayerController->SetControlRotation(FRotator(OldRotation.Pitch, AimAngle, OldRotation.Roll));
137}

226void ADTPlayerCharacter::UpdateAimLocation()
227{
228 const FVector Origin = GetActorLocation();
229 FVector AimPoint = Origin;
230
231 if (bUsingMouse && PlayerController)
232 {
233 // Mouse: project the cursor onto the character's horizontal plane. This is the stable 2D aim used for
234 // facing - it ignores vertical geometry (walls) so the character doesn't turn to face hovered surfaces.
235 float MouseX, MouseY;
236 FVector WorldLocation, WorldDirection;
237 if (PlayerController->GetMousePosition(MouseX, MouseY) &&
238 PlayerController->DeprojectScreenPositionToWorld(MouseX, MouseY, WorldLocation, WorldDirection) &&
239 !FMath::IsNearlyZero(WorldDirection.Z))
240 {
241 const float T = (Origin.Z - WorldLocation.Z) / WorldDirection.Z;
242 AimPoint = WorldLocation + WorldDirection * T;
243 }
244 }
245 else
246 {
247 // Gamepad: no cursor, so project along the current aim direction at a fixed distance.
248 const FVector AimDirection = FRotator(0.0f, AimAngle, 0.0f).Vector();
249 AimPoint = Origin + AimDirection * DefaultAimDistance;
250 }
251
252 // Snap only the Z to the ground beneath the aim point. The XY (and therefore facing/aim direction) is
253 // preserved, so this yields a valid ground location for targeting without changing how the character aims.
254 if (const UWorld* World = GetWorld())
255 {
256 const FVector TraceStart(AimPoint.X, AimPoint.Y, Origin.Z + 1000.0f);
257 const FVector TraceEnd(AimPoint.X, AimPoint.Y, Origin.Z - 1000.0f);
258 FHitResult GroundHit;
259 FCollisionQueryParams Params(TEXT("DTAimGround"), false, this);
260 if (World->LineTraceSingleByChannel(GroundHit, TraceStart, TraceEnd, AimGroundTraceChannel, Params) && GroundHit.bBlockingHit)
261 {
262 AimPoint.Z = GroundHit.ImpactPoint.Z;
263 }
264 }
265
266 CurrentAimLocation = AimPoint;
267}

269FVector ADTPlayerCharacter::GetAimGroundLocation(float MaxDistance) const
270{
271 FVector AimPoint = CurrentAimLocation;
272
273 // Apply the ability-specific range clamp (the cache is the raw, unclamped aim point).
274 if (MaxDistance > 0.0f)
275 {
276 const FVector Origin = GetActorLocation();
277 const FVector Offset = AimPoint - Origin;
278 if (Offset.Size2D() > MaxDistance)
279 {
280 const FVector Clamped = Offset.GetClampedToMaxSize2D(MaxDistance);
281 AimPoint.X = Origin.X + Clamped.X;
282 AimPoint.Y = Origin.Y + Clamped.Y;
283 }
284 }
285
286 return AimPoint;
287}
