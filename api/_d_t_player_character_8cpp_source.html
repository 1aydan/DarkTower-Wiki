227void ADTPlayerCharacter::UpdateAimLocation()
228{
229 const FVector Origin = GetActorLocation();
230 FVector AimPoint = Origin;
231
232 if (bUsingMouse && PlayerController)
233 {
234 float MouseX, MouseY;
235 FVector WorldLocation, WorldDirection;
236 if (PlayerController->GetMousePosition(MouseX, MouseY) &&
237 PlayerController->DeprojectScreenPositionToWorld(MouseX, MouseY, WorldLocation, WorldDirection) &&
238 !FMath::IsNearlyZero(WorldDirection.Z))
239 {
240 const float T = (Origin.Z - WorldLocation.Z) / WorldDirection.Z;
241 AimPoint = WorldLocation + WorldDirection * T;
242 }
243 }
244 else
245 {
246 const FVector AimDirection = FRotator(0.0f, AimAngle, 0.0f).Vector();
247 AimPoint = Origin + AimDirection * DefaultAimDistance;
248 }
249
250 if (const UWorld* World = GetWorld())
251 {
252 const FVector TraceStart(AimPoint.X, AimPoint.Y, Origin.Z + 1000.0f);
253 const FVector TraceEnd(AimPoint.X, AimPoint.Y, Origin.Z - 1000.0f);
254 FHitResult GroundHit;
255 FCollisionQueryParams Params(TEXT("DTAimGround"), false, this);
256 if (World->LineTraceSingleByChannel(GroundHit, TraceStart, TraceEnd, AimGroundTraceChannel, Params) && GroundHit.bBlockingHit)
257 {
258 AimPoint.Z = GroundHit.ImpactPoint.Z;
259 }
260 }
261
262 CurrentAimLocation = AimPoint;
263}

265FVector ADTPlayerCharacter::GetAimGroundLocation(float MaxDistance) const
266{
267 FVector AimPoint = CurrentAimLocation;
268
269 // Apply the ability-specific range clamp (the cache is the raw, unclamped aim point).
270 if (MaxDistance > 0.0f)
271 {
272 const FVector Origin = GetActorLocation();
273 const FVector Offset = AimPoint - Origin;
274 if (Offset.Size2D() > MaxDistance)
275 {
276 const FVector Clamped = Offset.GetClampedToMaxSize2D(MaxDistance);
277 AimPoint.X = Origin.X + Clamped.X;
278 AimPoint.Y = Origin.Y + Clamped.Y;
279 }
280 }
281
282 return AimPoint;
283}

372
373 FDTPlayerDeathMessage Message;
374 Message.Verb = DTGameplayTags::TAG_EVENT_PLAYER_DEATH;
375 Message.Instigator = this;
376 Message.Target = this;
377 Message.DeadPlayerState = PS;
378 Message.bMortalMode = bMortalMode;
379
