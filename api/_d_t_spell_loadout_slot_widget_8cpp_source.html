37void UDTSpellLoadoutSlotWidget::RefreshSpellDisplay()
38{
39 UDTAbilityDefinition* Def = nullptr;
40
41 if (OwningLoadout.IsValid())
42 {
43 const TSubclassOf<UGameplayAbility> SpellClass = OwningLoadout->GetSpellInSlot(SpellSlotTag);
44 if (const UDTGameplayAbility* SpellCDO = SpellClass ? SpellClass->GetDefaultObject<UDTGameplayAbility>() : nullptr)
45 {
46 Def = SpellCDO->GetAbilityDefinition();
47 }
48 }
49
50 if (Def)

62bool UDTSpellLoadoutSlotWidget::NativeOnDrop(const FGeometry& InGeometry, const FDragDropEvent& InDragDropEvent, UDragDropOperation* InOperation)
63{
64 UDTAbilityDragPayload* Payload = InOperation ? Cast<UDTAbilityDragPayload>(InOperation->Payload) : nullptr;
65 if (!Payload || !Payload->AbilityHandle.IsValid() || !OwningLoadout.IsValid() || !SpellSlotTag.IsValid())
66 {
67 return false;
68 }
69
70 // Write to the editing weapon set's stored loadout (NOT the live slot path the base widget uses).
71 OwningLoadout->SetSpellInSlot(SpellSlotTag, Payload->AbilityHandle);
72 RefreshSpellDisplay();
73 return true;
74}
