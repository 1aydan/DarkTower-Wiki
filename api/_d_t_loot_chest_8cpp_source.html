91void ADTLootChest::OpenChest(APlayerController* InstigatingPC)
92{
93 if (bOpened)
94 {
95 return;
96 }
97
98 bOpened = true;
99 InstigatingController = InstigatingPC;
100 OnRep_Opened();
101
102 APawn* InstigatingPawn = InstigatingPC ? InstigatingPC->GetPawn() : nullptr;
103 Multicast_NotifyChestOpened(InstigatingPawn);
104
105 // Calculate spawn transform for loot
106 const FVector SpawnOrigin = GetActorTransform().TransformPosition(SpawnLocation);
107 FTransform SpawnTransformWithLocation = GetActorTransform();
108 SpawnTransformWithLocation.SetLocation(SpawnOrigin);
109
110 TArray<ADTItemActor*> SpawnedLoot;
111
112 if (LootSourceComponent)
113 {
114 LootSourceComponent->DropLootAtTransform(SpawnTransformWithLocation, SpawnedLoot, InstigatingPC);
115 }
116
117 for (int32 i = 0; i < SpawnedLoot.Num(); ++i)
118 {
119 LaunchLootActor(SpawnedLoot[i], i, SpawnedLoot.Num());
120 }
121
122 ForceNetUpdate();

125void ADTLootChest::LaunchLootActor(ADTItemActor* LootActor, int32 ItemIndex, int32 TotalItems)
126{
127 if (!IsValid(LootActor))
128 {
129 return;
130 }
131
132 UPrimitiveComponent* PrimitiveComp = Cast<UPrimitiveComponent>(LootActor->GetRootComponent());
133 if (!PrimitiveComp)
134 {
135 return;
136 }
137
138 const FRotator ChestRotation = GetActorRotation();
139
140 float HorizontalAngle = 0.0f;
141 if (TotalItems > 1)
142 {
143 const float Step = SpreadAngle / (TotalItems - 1);
144 HorizontalAngle = (ItemIndex * Step) - (SpreadAngle * 0.5f);
145 }
146
147 const float AngleVariation = FMath::FRandRange(-10.0f, 10.0f);
148 HorizontalAngle += AngleVariation;
149
150 const FRotator LaunchRotation = FRotator(LaunchAngle, ChestRotation.Yaw + HorizontalAngle, 0.0f);
151 const FVector LaunchDirection = LaunchRotation.Vector();
152
153 if (bDebug)
154 {
155 DrawDebugLine(GetWorld(), LootActor->GetActorLocation(),
156 LootActor->GetActorLocation() + LaunchDirection * 200.0f,
157 FColor::Green, false, 5.0f, 0, 2.0f);
158 }
159
160 const float SpeedVariation = FMath::FRandRange(1.0f - LaunchSpeedVariation, 1.0f + LaunchSpeedVariation);
161 const float FinalSpeed = LaunchSpeed * SpeedVariation;
162 const FVector LaunchVelocity = LaunchDirection.GetSafeNormal() * FinalSpeed;
163
164 PrimitiveComp->AddImpulse(LaunchVelocity, NAME_None, true);
165}
