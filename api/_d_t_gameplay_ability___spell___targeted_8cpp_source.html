39void UDTGameplayAbility_Spell_Targeted::StartInteractiveTargeting()
40{
41 // No target actor configured: fall back to the avatar's aim point and cast immediately.
42 if (!TargetActorClass)
43 {
44 EndAbility(CurrentSpecHandle, CurrentActorInfo, CurrentActivationInfo, true, false);
45 return;
46 }
47
48 WaitTargetDataTask = UAbilityTask_WaitTargetData::WaitTargetData(this, TEXT("SpellTargeting"), ConfirmationType, TargetActorClass);
49 if (!WaitTargetDataTask)
50 {
51 UE_LOG(LogDTAbility, Warning, TEXT("UDTGameplayAbility_Spell_Targeted::StartInteractiveTargeting - Failed to create WaitTargetData task; casting at avatar."));
52 BeginSpellCast();
53 return;
54 }
55
56 WaitTargetDataTask->ValidData.AddDynamic(this, &UDTGameplayAbility_Spell_Targeted::OnTargetDataReady);
57 WaitTargetDataTask->Cancelled.AddDynamic(this, &UDTGameplayAbility_Spell_Targeted::OnTargetDataCancelled);
58
59 AGameplayAbilityTargetActor* SpawnedTargetActor = nullptr;
60 if (WaitTargetDataTask->BeginSpawningActor(this, TargetActorClass, SpawnedTargetActor))
61 {
62 WaitTargetDataTask->FinishSpawningActor(this, SpawnedTargetActor);
63 }
64
65 WaitTargetDataTask->ReadyForActivation();
66}

121bool UDTGameplayAbility_Spell_Targeted::ExtractLocation(const FGameplayAbilityTargetDataHandle& Data, FVector& OutLocation)
122{
123 for (int32 Index = 0; Index < Data.Num(); ++Index)
124 {
125 const FGameplayAbilityTargetData* TargetData = Data.Get(Index);
126 if (!TargetData)
127 {
128 continue;
129 }
130
131 if (TargetData->HasHitResult())
132 {
133 const FHitResult* HitResult = TargetData->GetHitResult();
134 OutLocation = HitResult->bBlockingHit ? HitResult->ImpactPoint : HitResult->TraceEnd;
135 return true;
136 }
137
138 if (TargetData->HasEndPoint())
139 {
140 OutLocation = TargetData->GetEndPoint();
141 return true;
142 }
143 }
144
145 return false;
