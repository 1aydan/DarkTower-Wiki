247FTargetingSourceContext UDTTargetingComponent::ResolveSourceContext(EDTTargetSourceMode Mode) const
248{
249 FTargetingSourceContext Context = CreateSourceContext();
250
251 AActor* SourceActor = Context.SourceActor;
252 if (!SourceActor)
253 {
254 return Context;
255 }
256
257 // Default the request origin to the avatar's location; modes below may override it.
258 Context.SourceLocation = SourceActor->GetActorLocation();
259
260 switch (Mode)
261 {
262 case EDTTargetSourceMode::Self:
263 break;
264
265 case EDTTargetSourceMode::AimLocation:
266 case EDTTargetSourceMode::AimDirection:
267 {
268 FVector AimLocation;
269 FRotator AimRotation;
270 if (ResolveAimPoint(SourceActor, AimLocation, AimRotation))
271 {
272 Context.SourceLocation = AimLocation;
273 }
274 break;
275 }
276
277 case EDTTargetSourceMode::CurrentTarget:
278 if (CurrentTarget)
279 {
280 Context.SourceLocation = CurrentTarget->GetActorLocation();
281 }
282 break;
283
284 default:
285 break;
286 }
287
288 return Context;
289}

291bool UDTTargetingComponent::ResolveAimPoint(AActor* SourceActor, FVector& OutLocation, FRotator& OutRotation) const
292{
293 if (!SourceActor)
294 {
295 return false;
296 }
297
298 AController* Controller = nullptr;
299 if (const APawn* Pawn = Cast<APawn>(SourceActor))
300 {
301 Controller = Pawn->GetController();
302 }
303 else
304 {
305 Controller = Cast<AController>(GetOwner());
306 }
307
308 if (!Controller)
309 {
310 return false;
311 }
312
313 // Player: trace from the camera view point along the aim direction.
314 if (Controller->IsPlayerController())
315 {
316 FVector ViewLocation;
317 FRotator ViewRotation;
318 Controller->GetPlayerViewPoint(ViewLocation, ViewRotation);
319 OutRotation = ViewRotation;
320
321 const UWorld* World = GetWorld();
322 if (!World)
323 {
324 OutLocation = SourceActor->GetActorLocation();
325 return true;
326 }
327
328 const FVector TraceEnd = ViewLocation + ViewRotation.Vector() * MaxAimDistance;
329 FHitResult Hit;
330 FCollisionQueryParams Params(TEXT("DTAimTrace"), false, SourceActor);
331 if (World->LineTraceSingleByChannel(Hit, ViewLocation, TraceEnd, AimTraceChannel, Params))
332 {
333 OutLocation = Hit.ImpactPoint;
334 }
335 else
336 {
337 OutLocation = TraceEnd;
338 }
339 return true;
340 }
341
342 // AI: use the controller's focal point as the aim location.
343 if (const AAIController* AIController = Cast<AAIController>(Controller))
344 {
345 OutLocation = AIController->GetFocalPoint();
346 OutRotation = (OutLocation - SourceActor->GetActorLocation()).Rotation();
347 return true;
348 }
349
350 return false;
351}

368FTargetingRequestHandle UDTTargetingComponent::CreateTargetingHandle(UTargetingPreset* TargetingPreset, const TArray<AActor*>& InitialTargets) const
369{
370 FTargetingRequestHandle InvalidHandle;
371
372 // Use default preset if none provided
373 UTargetingPreset* PresetToUse = TargetingPreset ? TargetingPreset : DefaultTargetingPreset.Get();
374 if (!PresetToUse)
375 {
376 UE_LOG(LogDT, Warning, TEXT("DTTargetingComponent::CreateTargetingHandle - No targeting preset provided and no default preset set!"));
377 return InvalidHandle;
378 }
379
380 UTargetingSubsystem* Subsystem = GetTargetingSubsystem();
381 if (!Subsystem)
382 {
383 UE_LOG(LogDT, Warning, TEXT("DTTargetingComponent::CreateTargetingHandle - Could not get targeting subsystem!"));
384 return InvalidHandle;
385 }
386
387 // Create the source context
388 FTargetingSourceContext SourceContext = CreateSourceContext();
389
390 // Create the targeting handle
391 FTargetingRequestHandle Handle = UTargetingSubsystem::MakeTargetRequestHandle(PresetToUse, SourceContext);
392
393 if (!Handle.IsValid())
394 {
395 return InvalidHandle;
396 }
397
398 // Setup initial targets if provided
399 if (InitialTargets.Num() > 0)
400 {
401 SetupInitialTargetsForRequest(Handle, InitialTargets);
402 }
403
404 return Handle;

457FTargetingSourceContext UDTTargetingComponent::CreateSourceContext() const
458{
459 FTargetingSourceContext SourceContext;
460 AActor* Owner = GetOwner();
461 if (AController* Controller = Cast<AController>(Owner))
462 {
463 if (APawn* ControlledPawn = Controller->GetPawn())
464 {
465 SourceContext.SourceActor = ControlledPawn;
466 SourceContext.InstigatorActor = Controller;
467 }
468 else
469 {
470 // No pawn, fall back to controller
471 SourceContext.SourceActor = Controller;
472 SourceContext.InstigatorActor = Controller;
473 }
474 }
475 else

478 SourceContext.SourceActor = Owner;
479 SourceContext.InstigatorActor = Owner;
480 }
481
482 return SourceContext;

485void UDTTargetingComponent::SetupInitialTargetsForRequest(FTargetingRequestHandle RequestHandle, const TArray<AActor*>& InitialTargets) const
486{
487 if (!RequestHandle.IsValid() || InitialTargets.Num() == 0)
488 {
489 return;
490 }
491
492 FTargetingDefaultResultsSet& TargetingResults = FTargetingDefaultResultsSet::FindOrAdd(RequestHandle);
493 for (AActor* Target : InitialTargets)
494 {
495 if (!Target)
496 {
497 continue;
498 }
499
500 // Check if this target is already in the results
501 bool bAddResult = !TargetingResults.TargetResults.ContainsByPredicate([Target](const FTargetingDefaultResultData& Data) -> bool
502 {
503 return (Data.HitResult.GetActor() == Target);
504 });
505
506 if (bAddResult)
507 {
508 FTargetingDefaultResultData* ResultData = new(TargetingResults.TargetResults) FTargetingDefaultResultData();
509 ResultData->HitResult.HitObjectHandle = FActorInstanceHandle(Target);
510 ResultData->HitResult.Location = Target->GetActorLocation();
511 }
512 }

515void UDTTargetingComponent::HandleTargetingRequestComplete(FTargetingRequestHandle TargetingHandle, bool bAutoSetFirstTarget)
516{
517 UTargetingSubsystem* Subsystem = GetTargetingSubsystem();
518 if (!Subsystem)
519 {
520 OnTargetingComplete.Broadcast(TArray<AActor*>(), false);
521 return;
522 }
523
524 TArray<AActor*> FoundTargets;
525 Subsystem->GetTargetingResultsActors(TargetingHandle, FoundTargets);
526
527 LastFoundTargets = FoundTargets;
528
529 if (FoundTargets.Num() <= 0)
530 {
531 ClearTarget();
532 OnTargetingComplete.Broadcast(FoundTargets, false);
533 return;
534 }
535
536 if (bAutoSetFirstTarget && !HasTarget())
537 {
538 SetCurrentTarget(FoundTargets[0]);
539 }
540
541 OnTargetingComplete.Broadcast(FoundTargets, true);

577void UDTTargetingComponent::UpdateAutoTargeting()
578{
579 if (!DefaultTargetingPreset)
580 {
581 UE_LOG(LogDT, Warning, TEXT("DTTargetingComponent::UpdateAutoTargeting - No default targeting preset!"));
582 StopAutoTargeting();
583 return;
584 }
585
586 // TODO : Add seperate validation function and events for target death etc
587 if (CurrentTarget && AutoTargetingMaxDistance > 0.0f)
588 {
589 float Distance = FVector::Dist(GetOwner()->GetActorLocation(), CurrentTarget->GetActorLocation());
590 if (Distance > AutoTargetingMaxDistance)
591 {
592 UE_LOG(LogDT, Verbose, TEXT("DTTargetingComponent::UpdateAutoTargeting - Target out of range (%.2f > %.2f), clearing"), Distance, AutoTargetingMaxDistance);
593 ClearTarget();
594 }
595 else if (!bContinuousAutoTargeting)
596 {
597 return;
598 }
599 }
600 else if (!bContinuousAutoTargeting && HasTarget())
601 {
602 return;
603 }
604
605 if (bUseAsyncForAutoTargeting)
606 {
607 ExecuteAsyncTargetingRequest(nullptr, true);
608 }
609 else
610 {
611 ExecuteTargetingRequest(nullptr, true);
612 }
