19TArray<FDTBiomeEntryInfo> UDTCampaignTabWidget::GetBiomeInfos() const
20{
21 TArray<FDTBiomeEntryInfo> Result;
22
23 const UDTCampaignDefinition* CampaignData = GetCampaignData();
24 if (!CampaignData)
25 {
26 return Result;
27 }
28
29 const int32 MaxAccessibleFloor = GetMaxUnlockedFloor();
30
31 for (int32 BiomeIdx = 0; BiomeIdx < CampaignData->Biomes.Num(); ++BiomeIdx)
32 {
33 const UDTBiomeDefinition* Biome = CampaignData->Biomes[BiomeIdx];
34
35 FDTBiomeEntryInfo Info;
36 Info.BiomeIndex = BiomeIdx;
37 Info.DisplayName = Biome ? Biome->DisplayName
38 : FText::FromString(FString::Printf(TEXT("Biome %d"), BiomeIdx + 1));
39 Info.GlobalFloorStart = BiomeIdx * UDTCampaignDefinition::FloorsPerBiome + 1;
40 Info.TotalFloors = UDTCampaignDefinition::FloorsPerBiome;
41 Info.PreviewImage = Biome ? Biome->PreviewImage : nullptr;
42 Info.bIsSelected = (BiomeIdx == SelectedBiomeIndex);
43 Info.UnlockedFloors = FMath::Clamp(MaxAccessibleFloor - Info.GlobalFloorStart + 1, 0, Info.TotalFloors);
44
45 Result.Add(Info);
46 }
47
48 return Result;
49}

51TArray<FDTFloorEntryInfo> UDTCampaignTabWidget::GetFloorsForSelectedBiome() const
52{
53 TArray<FDTFloorEntryInfo> Result;
54
55 const UDTCampaignDefinition* CampaignData = GetCampaignData();
56 if (!CampaignData || !CampaignData->Biomes.IsValidIndex(SelectedBiomeIndex))
57 {
58 return Result;
59 }
60
61 const UDTBiomeDefinition* Biome = CampaignData->Biomes[SelectedBiomeIndex];
62 const int32 GlobalStart = SelectedBiomeIndex * UDTCampaignDefinition::FloorsPerBiome + 1;
63 const int32 MaxAccessibleFloor = GetMaxUnlockedFloor();
64 const int32 CurrentFloor = GetCurrentFloor();
65
66 Result.Reserve(UDTCampaignDefinition::FloorsPerBiome);
67 for (int32 LocalIdx = 0; LocalIdx < UDTCampaignDefinition::FloorsPerBiome; ++LocalIdx)
68 {
69 FDTFloorEntryInfo Info;
70 Info.LocalFloor = LocalIdx + 1;
71 Info.GlobalFloor = GlobalStart + LocalIdx;
72 Info.BiomeIndex = SelectedBiomeIndex;
73 Info.bUnlocked = Info.GlobalFloor <= MaxAccessibleFloor;
74 Info.bIsCurrent = Info.GlobalFloor == CurrentFloor;
75
76 if (Biome)
77 {
78 if (const FDTBiomeFloorDefinition* FloorDef = Biome->GetFloorDefinition(Info.LocalFloor))
79 {
80 Info.FloorType = FloorDef->FloorType;
81 }
82 }
83
84 Result.Add(Info);
85 }
86
87 return Result;
88}

115void UDTCampaignTabWidget::SetSelectedBiomeIndex(int32 NewBiomeIndex)
116{
117 const UDTCampaignDefinition* CampaignData = GetCampaignData();
118 const int32 NumBiomes = CampaignData ? CampaignData->Biomes.Num() : 0;
119 if (NumBiomes <= 0)
120 {
121 SelectedBiomeIndex = INDEX_NONE;
122 return;
123 }
124
125 const int32 Clamped = FMath::Clamp(NewBiomeIndex, 0, NumBiomes - 1);
126 if (Clamped == SelectedBiomeIndex)
127 {
128 return;
129 }
130
131 SelectedBiomeIndex = Clamped;
132 Refresh();
133}

135void UDTCampaignTabWidget::ClampSelectedBiomeIndex()
136{
137 const UDTCampaignDefinition* CampaignData = GetCampaignData();
138 const int32 NumBiomes = CampaignData ? CampaignData->Biomes.Num() : 0;
139 if (NumBiomes <= 0)
140 {
141 SelectedBiomeIndex = INDEX_NONE;
142 return;
143 }
144
145 if (SelectedBiomeIndex == INDEX_NONE)
146 {
147 const int32 Floor = FMath::Max(1, GetCurrentFloor());
148 SelectedBiomeIndex = FMath::Clamp((Floor - 1) / UDTCampaignDefinition::FloorsPerBiome, 0, NumBiomes - 1);
149 }
150 else
151 {
152 SelectedBiomeIndex = FMath::Clamp(SelectedBiomeIndex, 0, NumBiomes - 1);
153 }

180void UDTCampaignTabWidget::Refresh()
181{
182 const int32 CurrentFloor = GetCurrentFloor();
183 const int32 MaxFloor = GetMaxUnlockedFloor();
184
185 if (TXT_SelectedFloor)
186 {
187 TXT_SelectedFloor->SetText(FText::AsNumber(CurrentFloor));
188 }
189
190 if (TXT_MaxFloor)
191 {
192 TXT_MaxFloor->SetText(FText::AsNumber(MaxFloor));
193 }
194
195 if (TXT_ZoneTitle)
196 {
197 FText BiomeName = FText::GetEmpty();
198 for (const FDTBiomeEntryInfo& Biome : GetBiomeInfos())
199 {
200 const int32 BiomeEnd = Biome.GlobalFloorStart + Biome.TotalFloors - 1;
201 if (CurrentFloor >= Biome.GlobalFloorStart && CurrentFloor <= BiomeEnd)
202 {
203 BiomeName = Biome.DisplayName;
204 break;
205 }
206 }
207
208 TXT_ZoneTitle->SetText(FText::Format(NSLOCTEXT("DarkTower", "ZoneTitleFormat", "{0} : Floor - {1} {2}"), BiomeName, FText::AsNumber(CurrentFloor), GetPendingConfig().bMortalMode ? FText::FromString(TEXT("Mortal")) : FText::GetEmpty()
209 ));
210 }
211
212 if (CB_MortalMode)
213 {
214 CB_MortalMode->SetIsChecked(GetPendingConfig().bMortalMode);
215 }
216
217 RebuildBiomeEntries();
218 RebuildFloorEntries();
219}
