158void UDTGameplayAbility_ChainAbility::UpdateChainTargets()
159{
160 ActiveChainTraces.Reset();
161
162 AActor* Caster = ActiveCaster.Get();
163 if (!IsValid(Caster))
164 {
165 return;
166 }
167
168 FName AttachSocketName = NAME_None;
169 FVector Start = FVector::ZeroVector;
170 USceneComponent* AttachComponent = GetChainStartPoint(AttachSocketName, Start);
171
172 ActiveChainAttachComponent = AttachComponent;
173 ActiveChainAttachSocketName = AttachSocketName;
174
175 const TArray<FVector> Directions = UDTBlueprintLibrary::GetSpreadDirections(Caster->GetActorForwardVector(), GetChainAmount(), SpreadDegrees);
176
177 TArray<AActor*> ActorsToIgnore;
178 ActorsToIgnore.Add(Caster);
179
180 ActiveChainTraces.Reserve(Directions.Num());
181 for (const FVector& Direction : Directions)
182 {
183 FChainTrace& Chain = ActiveChainTraces.AddDefaulted_GetRef();
184 Chain.Start = Start;
185 Chain.Direction = Direction.GetSafeNormal();
186 Chain.End = GetChainEndPoint(Chain.Start, Chain.Direction, ActorsToIgnore);
187 }
188
189 TSet<TObjectPtr<AActor>> ClaimedTargets;
190
191 for (int32 ChainIndex = 0; ChainIndex < ActiveChainTraces.Num(); ++ChainIndex)
192 {
193 FChainTrace& Chain = ActiveChainTraces[ChainIndex];
194
195 TArray<FHitResult> Hits;
196
197 UKismetSystemLibrary::SphereTraceMultiForObjects(this, Chain.Start, Chain.End, TraceRadius, TargetObjectTypes, false, ActorsToIgnore, DebugTraceType, Hits, true);
198
199 const float ChainLength = FMath::Max(FVector::DotProduct(Chain.End - Chain.Start, Chain.Direction), 0.0f);
200
201 TSet<TObjectPtr<AActor>> ChainTargets;
202
203 for (const FHitResult& Hit : Hits)
204 {
205 AActor* Target = Hit.GetActor();
206 if (!IsValidChainTarget(Target, Chain.Start) || ClaimedTargets.Contains(Target) || ChainTargets.Contains(Target))
207 {
208 continue;
209 }
210
211 const float DistanceAlongChain = FVector::DotProduct(Target->GetActorLocation() - Chain.Start, Chain.Direction);
212 if (DistanceAlongChain < 0.0f || DistanceAlongChain > ChainLength)
213 {
214 continue;
215 }
216
217 ChainTargets.Add(Target);
218
219 FChainTarget& ChainTarget = Chain.Targets.AddDefaulted_GetRef();
220
221 ChainTarget.Target = Target;
222 ChainTarget.DistanceAlongChain = DistanceAlongChain;
223 ChainTarget.HookLocation = Chain.Start + (Chain.Direction * DistanceAlongChain);
224 ChainTarget.Hit = Hit;
225 }
226
227 Chain.Targets.Sort([](const FChainTarget& A, const FChainTarget& B)
228 {
229 return A.DistanceAlongChain < B.DistanceAlongChain;
230 });
231
232 if (!Chain.Targets.IsEmpty())
233 {
234 for (const FChainTarget& PullTarget : Chain.Targets)
235 {
236 ClaimedTargets.Add(PullTarget.Target);
237 }
238 }
239
240 }

260bool UDTGameplayAbility_ChainAbility::PullTarget(int32 ChainIndex, int32 TargetIndex, const FChainTrace& Chain, const FChainTarget& PullTargetData)
261{
262 ACharacter* TargetCharacter = Cast<ACharacter>(PullTargetData.Target);
263 AActor* Caster = ActiveCaster.Get();
264 if (!TargetCharacter || !IsValid(Caster) || !Caster->HasAuthority() || !IsValidChainTarget(TargetCharacter, Chain.Start))
265 {
266 return false;
267 }
268
269 const float StopDistance = FMath::Clamp(PullStopDistance, 0.0f, PullTargetData.DistanceAlongChain);
270 if (PullTargetData.DistanceAlongChain <= StopDistance + UE_SMALL_NUMBER || PullStrength <= UE_SMALL_NUMBER)
271 {
272 return false;
273 }
274
275 const FVector HookStopLocation = Chain.Start + (Chain.Direction * StopDistance);
276 const FVector TargetOffsetFromHook = TargetCharacter->GetActorLocation() - PullTargetData.HookLocation;
277 const FVector PullDestination = HookStopLocation + TargetOffsetFromHook;
278
279 UGameplayStatics::FSuggestProjectileVelocityParameters VelocityParams(this, TargetCharacter->GetActorLocation(), PullDestination, PullStrength);
280 VelocityParams.bFavorHighArc = false;
281 VelocityParams.TraceOption = ESuggestProjVelocityTraceOption::DoNotTrace;
282 VelocityParams.bAcceptClosestOnNoSolutions = true;
283
284 const UCapsuleComponent* Capsule = TargetCharacter->GetCapsuleComponent();
285 if (Capsule)
286 {
287 VelocityParams.CollisionRadius = Capsule->GetScaledCapsuleRadius();
288 }
289
290 FVector LaunchVelocity;
291 if (!UGameplayStatics::SuggestProjectileVelocity(VelocityParams, LaunchVelocity))
292 {
293 LaunchVelocity = (PullDestination - TargetCharacter->GetActorLocation()).GetSafeNormal() * PullStrength;
294 }
295
296 UCharacterMovementComponent* MovementComponent = TargetCharacter->GetCharacterMovement();
297 if (MovementComponent)
298 {
299 MovementComponent->StopMovementImmediately();
300 }
301
302 TargetCharacter->LaunchCharacter(LaunchVelocity, true, true);
303 K2_OnChainTargetPulled(TargetCharacter, ChainIndex, TargetIndex, PullTargetData.HookLocation, PullDestination, LaunchVelocity);
304
305 return true;
