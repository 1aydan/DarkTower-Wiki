71TMap<FGameplayTag, float> FStatCollectionResult::AggregateStatsByTag(const TArray<FCollectedStatModifier>& Stats) const
72{
73 TMap<FGameplayTag, float> Result;
74
75 TMap<FGameplayTag, TArray<FCollectedStatModifier>> GroupedMods;
76 for (const FCollectedStatModifier& Mod : Stats)
77 {
78 if (Mod.StatTag.IsValid())
79 {
80 GroupedMods.FindOrAdd(Mod.StatTag).Add(Mod);
81 }
82 }
83
84 for (const auto& Pair : GroupedMods)
85 {
86 const FGameplayTag& Tag = Pair.Key;
87 const TArray<FCollectedStatModifier>& Mods = Pair.Value;
88
89 float OverrideValue = 0.0f;
90 bool bHasOverride = false;
91 float AddTotal = 0.0f;
92 float MultiplyTotal = 0.0f;
93
94 for (const FCollectedStatModifier& Mod : Mods)
95 {
96 switch (Mod.Operation)
97 {
98 case EStatModifierOp::Override:
99 OverrideValue = Mod.Value;
100 bHasOverride = true;
101 break;
102 case EStatModifierOp::Add:
103 AddTotal += Mod.Value;
104 break;
105 case EStatModifierOp::Multiply:
106 MultiplyTotal += Mod.Value;
107 break;
108 }
109 }
110
111 float Total = bHasOverride ? OverrideValue : 0.0f;
112 Total += AddTotal;
113 Total += Total * (MultiplyTotal / 100.0f);
114
115 const bool bIsPercentageOnly = !bHasOverride && AddTotal == 0.0f && MultiplyTotal != 0.0f;
116 if (!bIsPercentageOnly)
117 {
118 Total = static_cast<float>(FMath::FloorToInt32(Total));
119 }
120
121 Result.Add(Tag, Total);
122 }
123
124 return Result;
125}

127FStatCollectionResult FDTEquipmentStatHelper::CollectAllStats(const FInventoryItemInfo& ItemInfo, const UInventoryItemDefinition* ItemDef, const UDTGlobalLootData* GlobalLootData)
128{
129 FStatCollectionResult Result;
130
131 if (!ItemDef || !GlobalLootData)
132 {
133 return Result;
134 }
135
136 const FGameplayTag ItemTypeTag = ItemDef->GetItemTypeTag();
137
138 for (const FItemStatEntry& StatEntry : ItemInfo.BaseStats)
139 {
140 FBaseStatTableRow StatRow;
141 GlobalLootData->GetBaseStatRow(StatEntry.StatTag, ItemTypeTag, StatRow);
142
143 if (!StatRow.StatTag.IsValid())
144 {
145 UE_LOG(LogDTInventory, Warning, TEXT("FDTEquipmentStatHelper::CollectAllStats - Could not resolve base stat %s"), *StatEntry.StatTag.ToString());
146 continue;
147 }
148
149 const float ActualValue = UDTLootSubsystem::CalculateActualStatValue(StatEntry.StatRollValue, StatRow);
150 Result.StatModifiers.Emplace(StatEntry.StatTag, ActualValue, StatRow.ApplicationType, StatRow.Operation);
151 }
152
153 for (const FItemModifierEntry& ModEntry : ItemInfo.Modifiers)
154 {
155 UDTAffixDefinition* AffixDef = UDefinitionAssetManager::Get().LoadDefinitionByGuid<UDTAffixDefinition>(ModEntry.ModifierGuid);
156 if (!AffixDef)
157 {
158 UE_LOG(LogDTInventory, Warning, TEXT("FDTEquipmentStatHelper::CollectAllStats - Could not resolve affix with GUID %s"), *ModEntry.ModifierGuid.ToString());
159 continue;
160 }
161
162 const float Alpha = ModEntry.NormalizedRollValue;
163 const EStatApplicationType AffixAppType = AffixDef->ApplicationType;
164
165 if (AffixDef->AffixGameplayEffect)
166 {
167 TMap<FGameplayTag, float> SetByCallerValues;
168 for (const FStatModifierDef& StatMod : AffixDef->StatModifiers)
169 {
170 if (StatMod.SetByCallerTag.IsValid())
171 {
172 float Value = AffixDef->CalculateStatModifierValue(StatMod, Alpha, ItemInfo.ItemLevel);
173
174 if (StatMod.Operation != EStatModifierOp::Multiply)
175 {
176 Value = static_cast<float>(FMath::FloorToInt32(Value));
177 }
178
179 SetByCallerValues.Add(StatMod.SetByCallerTag, Value);
180 }
181 }
182
183 Result.AffixGEs.Emplace(AffixDef->AffixGameplayEffect, SetByCallerValues, AffixAppType);
184 }
185 else if (AffixDef->StatModifiers.Num() > 0)
186 {
187 UE_LOG(LogDTInventory, Warning, TEXT("FDTEquipmentStatHelper::CollectAllStats - Affix '%s' has StatModifiers but no AffixGameplayEffect configured. Stats will not be applied."), *AffixDef->GetName());
188 }
189
190 if (AffixDef->AffixAbilitySet)
191 {
192 Result.AbilitySetsToGrant.Emplace(AffixDef->AffixAbilitySet, AffixAppType);
193 }
194
195 for (UDTCombatModifier* CombatModifier : AffixDef->CombatModifiers)
196 {
197 if (!CombatModifier)
198 {
199 continue;
200 }
201
202 FCollectedCombatModifier CollectedModifier(CombatModifier, AffixAppType);
203 if (!CombatModifier->MagnitudeModifierID.IsNone())
204 {
205 CollectedModifier.MagnitudeOverride = AffixDef->GetStatModifierValueByID(CombatModifier->MagnitudeModifierID, Alpha, ItemInfo.ItemLevel);
206 CollectedModifier.bUseMagnitudeOverride = true;
207 }
208
209 Result.CombatModifiers.Add(CollectedModifier);
210 }
211 }
212
213 return Result;
214}

216FActiveGameplayEffectHandle FDTEquipmentStatHelper::ApplyAggregatedStatsAsEffect(UAbilitySystemComponent* AbilitySystem,const TMap<FGameplayTag, float>& AggregatedStats,TSubclassOf<UGameplayEffect> StatEffectClass,UDTAttributeSet* AttributeSet,UObject* SourceObject)
217{
218 if (!AbilitySystem || !StatEffectClass || !AttributeSet || AggregatedStats.Num() == 0)
219 {
220 return FActiveGameplayEffectHandle();
221 }
222
223 FGameplayEffectContextHandle Context = AbilitySystem->MakeEffectContext();
224 Context.AddSourceObject(SourceObject);
225
226 FGameplayEffectSpecHandle Spec = AbilitySystem->MakeOutgoingSpec(StatEffectClass, 1.0f, Context);
227 if (!Spec.IsValid())
228 {
229 return FActiveGameplayEffectHandle();
230 }
231
232 for (const auto& Pair : AttributeSet->GetStatToSetByCallerMap())
233 {
234 Spec.Data->SetSetByCallerMagnitude(Pair.Value, AggregatedStats.FindRef(Pair.Key));
235 }
236
237 return AbilitySystem->ApplyGameplayEffectSpecToSelf(*Spec.Data.Get());
238}

240int32 FDTEquipmentStatHelper::ApplyAffixGEsAsEffects(UAbilitySystemComponent* AbilitySystem, const TArray<FCollectedAffixGE>& AffixGEs, UObject* SourceObject, TArray<FActiveGameplayEffectHandle>& OutHandles)
241{
242 if (!AbilitySystem || AffixGEs.Num() == 0)
243 {
244 return 0;
245 }
246
247 int32 AppliedCount = 0;
248 for (const FCollectedAffixGE& AffixGE : AffixGEs)
249 {
250 if (!AffixGE.GEClass)
251 {
252 continue;
253 }
254
255 FGameplayEffectContextHandle Context = AbilitySystem->MakeEffectContext();
256 Context.AddSourceObject(SourceObject);
257
258 FGameplayEffectSpecHandle Spec = AbilitySystem->MakeOutgoingSpec(AffixGE.GEClass, 1.0f, Context);
259 if (!Spec.IsValid())
260 {
261 UE_LOG(LogDTInventory, Warning, TEXT("FDTEquipmentStatHelper::ApplyAffixGEsAsEffects - Failed to create spec for GE '%s'"), *AffixGE.GEClass->GetName());
262 continue;
263 }
264
265 for (const auto& Pair : AffixGE.SetByCallerValues)
266 {
267 if (Pair.Key.IsValid())
268 {
269 Spec.Data->SetSetByCallerMagnitude(Pair.Key, Pair.Value);
270 }
271 }
272
273 FActiveGameplayEffectHandle Handle = AbilitySystem->ApplyGameplayEffectSpecToSelf(*Spec.Data.Get());
274 if (Handle.IsValid())
275 {
276 OutHandles.Add(Handle);
277 AppliedCount++;
278 }
279 else
280 {
281 UE_LOG(LogDTInventory, Warning, TEXT("FDTEquipmentStatHelper::ApplyAffixGEsAsEffects - Failed to apply GE '%s'"), *AffixGE.GEClass->GetName());
282 }
283 }
284
285 return AppliedCount;
