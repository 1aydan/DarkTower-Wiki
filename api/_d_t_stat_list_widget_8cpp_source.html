42void UDTStatListWidget::RefreshStatList()
43{
44 StatItems.Reset();
45
46 if (!LV_StatList)
47 {
48 return;
49 }
50
51 LV_StatList->ClearListItems();
52
53 if (!StatDisplayTable)
54 {
55 OnStatListRefreshed();
56 return;
57 }
58
59 // Resolve current stat values
60 const UStatSubsystem* StatSub = ResolveStatSubsystem();
61 const UPlayerStatComponent* PlayerStats = StatSub ? StatSub->GetPlayerStats(ResolvePlayerState()) : nullptr;
62 const UWorldStatComponent* WorldStats = StatSub ? StatSub->GetWorldStats() : nullptr;
63
64 const bool bHasCategoryFilter = FilterCategories.Num() > 0;
65 const bool bHasWhitelist = AllowedStats.Num() > 0;
66
67 // Iterate all rows in the DataTable
68 const FString ContextString(TEXT("DTStatListWidget::RefreshStatList"));
69 TArray<FDTStatDisplayRow*> Rows;
70 StatDisplayTable->GetAllRows<FDTStatDisplayRow>(ContextString, Rows);
71
72 for (const FDTStatDisplayRow* Row : Rows)
73 {
74 if (!Row || !Row->StatTag.IsValid())
75 {
76 continue;
77 }
78
79 // Apply stat whitelist
80 if (bHasWhitelist && !AllowedStats.HasTagExact(Row->StatTag))
81 {
82 continue;
83 }
84
85 // Apply category filter
86 if (bHasCategoryFilter && Row->CategoryTag.IsValid() && !FilterCategories.HasTagExact(Row->CategoryTag))
87 {
88 continue;
89 }
90
91 // Read current value from appropriate source
92 double CurrentValue = 0.0;
93 if (Row->bReadFromWorldStats)
94 {
95 if (WorldStats)
96 {
97 WorldStats->GetNumericStat(Row->StatTag, CurrentValue);
98 }
99 }
100 else
101 {
102 if (PlayerStats)
103 {
104 PlayerStats->GetNumericStat(Row->StatTag, CurrentValue);
105 }
106 }
107
108 UDTStatListItemData* Item = UDTStatListItemData::CreateFromRow(this, *Row, CurrentValue);
109 if (Item)
110 {
111 StatItems.Add(Item);
112 }
113 }
114
115 // Sort by category then by sort order within category
116 StatItems.Sort([](const UDTStatListItemData& A, const UDTStatListItemData& B)
117 {
118 if (A.CategoryTag != B.CategoryTag)
119 {
120 return A.CategoryTag.GetTagName().LexicalLess(B.CategoryTag.GetTagName());
121 }
122 return A.SortOrder < B.SortOrder;
123 });
124
125 // Feed items to the ListView
126 TArray<UObject*> ListObjects;
127 ListObjects.Reserve(StatItems.Num());
128 for (UDTStatListItemData* Item : StatItems)
129 {
130 ListObjects.Add(Item);
131 }
132 LV_StatList->SetListItems(ListObjects);
133
134 OnStatListRefreshed();
135}

185void UDTStatListWidget::BindStatDelegates()
186{
187 UStatSubsystem* StatSub = ResolveStatSubsystem();
188 if (!StatSub)
189 {
190 return;
191 }
192
193 if (!bStatDelegatesBound)
194 {
195 if (UPlayerStatComponent* PlayerStats = StatSub->GetPlayerStats(ResolvePlayerState()))
196 {
197 PlayerStats->OnStatChanged.AddDynamic(this, &UDTStatListWidget::HandleStatChanged);
198 bStatDelegatesBound = true;
199 }
200 }
201
202 if (!bWorldStatDelegatesBound)
203 {
204 if (UWorldStatComponent* WorldStats = StatSub->GetWorldStats())
205 {
206 WorldStats->OnStatChanged.AddDynamic(this, &UDTStatListWidget::HandleWorldStatChanged);
207 bWorldStatDelegatesBound = true;
208 }
