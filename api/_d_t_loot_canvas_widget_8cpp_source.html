222void UDTLootCanvasWidget::ResolveOverlaps(TArray<TPair<UDTLootLabelWidget*, FVector2D>>& LabelPositions) const
223{
224 TRACE_CPUPROFILER_EVENT_SCOPE(UDTLootCanvasWidget::ResolveOverlaps)
225
226 const int32 Num = LabelPositions.Num();
227 if (Num < 2)
228 {
229 return;
230 }
231
232 struct FPlacedEntry { FVector2D Min; FVector2D Max; };
233
234 TArray<FPlacedEntry, TInlineAllocator<32>> PlacedEntries;
235 PlacedEntries.Reserve(Num);
236
237 TArray<FVector2D, TInlineAllocator<32>> Candidates;
238
239 for (int32 i = 0; i < Num; ++i)
240 {
241 const FVector2D Origin = LabelPositions[i].Value;
242 const FVector2D Size = LabelPositions[i].Key->GetDesiredSize();
243 const float HalfW = Size.X * 0.5f;
244
245 auto MakeAABB = [&](const FVector2D& Pos) -> FPlacedEntry
246 {
247 return { FVector2D(Pos.X - HalfW, Pos.Y - Size.Y), FVector2D(Pos.X + HalfW, Pos.Y) };
248 };
249
250 auto CandidateFits = [&](const FVector2D& Cand) -> bool
251 {
252 const FPlacedEntry C = { MakeAABB(Cand).Min - LabelPadding * 0.5f, MakeAABB(Cand).Max + LabelPadding * 0.5f };
253 for (const FPlacedEntry& P : PlacedEntries)
254 {
255 if (C.Max.X > P.Min.X && C.Min.X < P.Max.X && C.Max.Y > P.Min.Y && C.Min.Y < P.Max.Y)
256 {
257 return false;
258 }
259 }
260 return true;
261 };
262
263 Candidates.Reset();
264 Candidates.Add(Origin);
265 for (const FPlacedEntry& P : PlacedEntries)
266 {
267 Candidates.Add(FVector2D(Origin.X, P.Min.Y - LabelPadding.Y));
268 Candidates.Add(FVector2D(Origin.X, P.Max.Y + Size.Y + LabelPadding.Y));
269 }
270
271 FVector2D BestPos = Origin;
272 float BestCost = TNumericLimits<float>::Max();
273
274 for (const FVector2D& Cand : Candidates)
275 {
276 if (CandidateFits(Cand))
277 {
278 const float Cost = FVector2D::DistSquared(Cand, Origin);
279 if (Cost < BestCost)
280 {
281 BestPos = Cand;
282 BestCost = Cost;
283 }
284 }
285 }
286
287 if (BestCost == TNumericLimits<float>::Max() && !PlacedEntries.IsEmpty())
288 {
289 BestPos = FVector2D(Origin.X, PlacedEntries.Last().Max.Y + Size.Y + LabelPadding.Y);
290 }
291
292 LabelPositions[i].Value = BestPos;
293 PlacedEntries.Add(MakeAABB(BestPos));
294 }
