164void UDTSaveSubsystem::PopulateSaveGameForCurrentPlayer(UDTSaveGame* SaveGame)
165{
166 APlayerState* LocalPlayerState = GetLocalPlayerState();
167 if (!SaveGame || !LocalPlayerState)
168 {
169 return;
170 }
171
172 UWorld* World = GetWorld();
173 if (!World)
174 {
175 return;
176 }
177
178 if (LocalPlayerState->Implements<USaveable>())
179 {
180 ISaveable::Execute_SaveToSaveGame(LocalPlayerState, SaveGame);
181 }
182
183 TArray<AActor*> SaveableActors;
184 UGameplayStatics::GetAllActorsWithInterface(World, USaveable::StaticClass(), SaveableActors);
185
186 for (AActor* Actor : SaveableActors)
187 {
188 if (!Actor || !Actor->Implements<USaveable>())
189 {
190 continue;
191 }
192
193 // Skip player state (already saved above)
194 if (Cast<APlayerState>(Actor))
195 {
196 continue;
197 }
198
199 ISaveable::Execute_SaveToSaveGame(Actor, SaveGame);
200 }
201}
