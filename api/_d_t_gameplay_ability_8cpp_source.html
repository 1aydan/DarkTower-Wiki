152{
153 if (!IsInstantiated() || !IsActive())
154 {
155 return false;
156 }
157
158 if (ActivationGroup == NewGroup)
159 {
160 return true;

162
163 UDTAbilitySystemComponent* Asc = GetDTAbilitySystemComponentFromActorInfo();
164 check(Asc);
165
166 if ((ActivationGroup != EAbilityActivationGroup::Exclusive_Blocking) && Asc->IsActivationGroupBlocked(NewGroup))
167 {
168 // This ability can't change groups if it's blocked (unless it is the one doing the blocking).
169 return false;

173 {

296 const bool bServerShouldActivate = ActorInfo->IsNetAuthority() && bIsServerExecution;

320{

322 {
323 return;
324 }
325
326 if (!bRechargeStacksOnCooldown)
327 {
328 return;
329 }
330
331 if (GetCurrentStacks() >= GetMaxStacks())
332 {
333 return;
334 }
335
336 UGameplayEffect* CooldownGE = GetCooldownGameplayEffect();
337 if (!CooldownGE)
338 {
339 UE_LOG(LogDTAbility, Warning, TEXT("%s: StartStackRegeneration - No CooldownGameplayEffectClass set, cannot regenerate stacks"), *GetName());
340 return;
341 }
342
343 UAbilitySystemComponent* ASC = GetAbilitySystemComponentFromActorInfo();
344 if (!ASC)
345 {
346 return;
347 }
348
349 if (ASC->GetOwnerRole() != ROLE_Authority)
350 {
351 return;
352 }
353
354 const FGameplayTagContainer* CooldownTags = GetCooldownTags();
355 if (CooldownTags && CooldownTags->Num() > 0)
356 {
357 if (ASC->HasAnyMatchingGameplayTags(*CooldownTags))
358 {
359 return;
360 }
361 }
362
363 // No cooldown tag is active — apply the GE to start (or restart) the regen loop.
364 FGameplayAbilitySpec* Spec = GetCurrentAbilitySpec();
365 if (Spec)
366 {
367 FGameplayEffectSpecHandle CooldownHandle = MakeOutgoingGameplayEffectSpec(CooldownGE->GetClass(), GetAbilityLevel());
368
369 if (!CooldownHandle.IsValid())
370 {
371 return;
372 }
373

375 if (Duration > 0.0f)
376 {
377 CooldownHandle.Data->SetSetByCallerMagnitude(DTGameplayTags::TAG_SETBYCALLER_DATA_COOLDOWN, Duration);
378 }
379
380 ApplyGameplayEffectSpecToOwner(Spec->Handle, GetCurrentActorInfo(), GetCurrentActivationInfo(), CooldownHandle);
