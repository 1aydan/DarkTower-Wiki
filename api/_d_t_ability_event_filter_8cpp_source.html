219bool UDTAbilityEventFilter_DirectionalAngle::Matches_Implementation(const FGameplayEventData& EventData) const
220{
221 const AActor* Target = EventData.Target.Get();
222 const AActor* Instigator = EventData.Instigator.Get();
223 if (!Target || !Instigator)
224 {
225 return false;
226 }
227
228 const AActor* RefActor = (ReferenceActor == EDTDirectionFilterActor::Target) ? Target : Instigator;
229 const AActor* SubjectActor = (ReferenceActor == EDTDirectionFilterActor::Target) ? Instigator : Target;
230
231 const FQuat YawRot = FQuat(RefActor->GetActorUpVector(), FMath::DegreesToRadians(DirectionDegrees));
232 const FVector ConeAxis = YawRot.RotateVector(RefActor->GetActorForwardVector());
233
234 const FVector ToSubject = (SubjectActor->GetActorLocation() - RefActor->GetActorLocation()).GetSafeNormal();
235 const float DotProduct = FVector::DotProduct(ConeAxis, ToSubject);
236 return DotProduct >= FMath::Cos(FMath::DegreesToRadians(ConeHalfAngleDegrees));

239bool UDTAbilityEventFilter_AttributeRatio::Matches_Implementation(const FGameplayEventData& EventData) const
240{
241 if (!CurrentAttribute.IsValid() || !MaxAttribute.IsValid())
242 {
243 return false;
244 }
245
246 const AActor* ActorToCheck = bCheckTarget
247 ? EventData.Target.Get()
248 : EventData.Instigator.Get();
249
250 // Fall back to context if actor not available from event data
251 if (!IsValid(ActorToCheck) && EventData.ContextHandle.IsValid())
252 {
253 const FGameplayEffectContext* Context = EventData.ContextHandle.Get();
254 if (Context)
255 {
256 ActorToCheck = bCheckTarget ? nullptr : Context->GetInstigator();
257 if (bCheckTarget && !ActorToCheck)
258 {
259 const TArray<TWeakObjectPtr<AActor>>& ContextActors = Context->GetActors();
260 if (ContextActors.Num() > 0 && ContextActors[0].IsValid())
261 {
262 ActorToCheck = ContextActors[0].Get();
263 }
264 }
265 }
266 }
267
268 if (!IsValid(ActorToCheck))
269 {
270 return false;
271 }
272
273 const UAbilitySystemComponent* ASC = UAbilitySystemBlueprintLibrary::GetAbilitySystemComponent(
274 const_cast<AActor*>(ActorToCheck));
275 if (!ASC)
276 {
277 return false;
278 }
279
280 bool bFoundCurrent = false;
281 bool bFoundMax = false;
282 const float Current = ASC->GetGameplayAttributeValue(CurrentAttribute, bFoundCurrent);
283 const float Max = ASC->GetGameplayAttributeValue(MaxAttribute, bFoundMax);
284
285 if (!bFoundCurrent || !bFoundMax || Max <= 0.0f)
286 {
287 return false;
288 }
289
290 const float Ratio = FMath::Clamp(Current / Max, 0.0f, 1.0f);
291 return Ratio >= MinRatio && Ratio <= MaxRatio;
