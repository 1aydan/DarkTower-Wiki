72{
73 if (!LV_SessionList)

77 }
78
79 ClearSessionList();
80
81 TArray<UDTOnlineSessionObject*> SessionObjects;
82 for (const FSSessionFindStruct& SessionData : SessionResults)
83 {
84 UDTOnlineSessionObject* SessionObject = NewObject<UDTOnlineSessionObject>(this);

88 SessionObjects.Add(SessionObject);

90 }
91
92 if (SessionObjects.Num() > 0)
93 {
94 SetStatusText(FText::FromString(FString::Printf(TEXT("Found %d server(s)"), SessionObjects.Num())));
95
96 LV_SessionList->SetListItems(SessionObjects);
97 }
98 else
99 {
100 SetStatusText(FText::FromString(TEXT("No servers found")));
