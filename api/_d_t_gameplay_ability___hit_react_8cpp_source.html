135FVector UDTGameplayAbility_HitReact::ComputeWorldHitDirection(const AActor* Avatar, const FGameplayEventData& EventData)
136{
137 if (!Avatar)
138 {
139 return FVector::ForwardVector;
140 }
141
142 FVector Direction = FVector::ZeroVector;
143 const AActor* InstigatorActor = EventData.Instigator.Get();
144
145 if (InstigatorActor)
146 {
147 Direction = (Avatar->GetActorLocation() - InstigatorActor->GetActorLocation()).GetSafeNormal2D();
148 }
149
150 if (Direction.IsNearlyZero())
151 {
152 const FHitResult* HitResult = EventData.ContextHandle.GetHitResult();
153 if (HitResult)
154 {
155 Direction = (-HitResult->ImpactNormal).GetSafeNormal2D();
156 }
157 }
158
159 if (Direction.IsNearlyZero())
160 {
161 Direction = (-Avatar->GetActorForwardVector()).GetSafeNormal2D();
162 }
163
164 return Direction;
165}

186void UDTGameplayAbility_HitReact::ApplyHitReact(const EDTHitReactTier Tier, const UDTCharacterData& CharacterData, const FVector& WorldHitDirection, const FGameplayEventData& EventData)
187{
188 const bool bInterrupting = (Tier == EDTHitReactTier::Medium || Tier == EDTHitReactTier::Heavy);
189 const bool bHeavy = (Tier == EDTHitReactTier::Heavy);
190 const FDTHitReactConfig& Config = CharacterData.HitReactConfig;
191 const EDTHitDirection Direction = UDTBlueprintLibrary::GetHitDirection(GetAvatarActorFromActorInfo(), WorldHitDirection);
192
193 if (bInterrupting)
194 {
195 ActiveInterruptTier = Tier;
196 if (!ChangeActivationGroup(EAbilityActivationGroup::Exclusive_Blocking))
197 {
198 UE_LOG(LogDTAbility, Warning, TEXT("UDTGameplayAbility_HitReact::ApplyHitReact: [%s] failed to change activation group to blocking."), *GetName());
199 }
200 }
201
202 UAnimMontage* Montage = nullptr;
203 if (bInterrupting)
204 {
205 const ADTCharacter* Avatar = Cast<ADTCharacter>(GetAvatarActorFromActorInfo());
206 Montage = Avatar ? Avatar->GetHitReactMontage(Tier, Direction) : CharacterData.GetHitReactMontage(Tier, Direction);
207 }
208
209 const float PlayRate = FMath::Max(Config.HeavyPlayRate, 0.01f);
210 float StateDuration = bHeavy ? Config.HeavyStateDuration : Config.MediumStateDuration;
211
212 if (Montage)
213 {
214 StateDuration = Montage->GetPlayLength() / PlayRate;
215 }
216
217 if (HasAuthority(&CurrentActivationInfo))
218 {
219 UAbilitySystemComponent* ASC = GetAbilitySystemComponentFromActorInfo();
220
221 FGameplayTag CueTag;
222 switch (Tier)
223 {
224 case EDTHitReactTier::None:
225 break;
226 case EDTHitReactTier::Light:
227 CueTag = Config.LightGameplayCueTag;
228 break;
229 case EDTHitReactTier::Medium:
230 CueTag = Config.MediumGameplayCueTag;
231 break;
232 case EDTHitReactTier::Heavy:
233 CueTag = Config.HeavyGameplayCueTag;
234 break;
235 }
236
237 if (ASC && CueTag.IsValid())
238 {
239 const FGameplayCueParameters CueParams = MakeHitReactCueParameters(GetAvatarActorFromActorInfo(), CharacterData, WorldHitDirection, EventData);
240 ASC->ExecuteGameplayCue(CueTag, CueParams);
241 }
242
243 if (bInterrupting)
244 {
245 const TSubclassOf<UGameplayEffect> StateEffect = bHeavy ? HeavyStateEffect : MediumStateEffect;
246 if (StateEffect)
247 {
248 const FGameplayEffectSpecHandle SpecHandle = MakeOutgoingGameplayEffectSpec(StateEffect, GetAbilityLevel());
249 if (SpecHandle.IsValid())
250 {
251 SpecHandle.Data->SetSetByCallerMagnitude(DTGameplayTags::TAG_SETBYCALLER_DATA_DURATION, StateDuration);
252 InterruptStateEffectHandle = ApplyGameplayEffectSpecToOwner(CurrentSpecHandle, CurrentActorInfo, CurrentActivationInfo, SpecHandle);
253 }
254 }
255 }
256 }
257
258 if (!bInterrupting)
259 {
260 return;
261 }
262
263 if (Montage)
264 {
265 UAbilityTask_PlayMontageAndWait* MontageTask = UAbilityTask_PlayMontageAndWait::CreatePlayMontageAndWaitProxy(this, NAME_None, Montage, PlayRate);
266 MontageTask->OnCompleted.AddDynamic(this, &UDTGameplayAbility_HitReact::OnReactFinished);
267 MontageTask->OnInterrupted.AddDynamic(this, &UDTGameplayAbility_HitReact::OnReactFinished);
268 MontageTask->OnCancelled.AddDynamic(this, &UDTGameplayAbility_HitReact::OnReactFinished);
269 MontageTask->ReadyForActivation();
270 }
271 else
272 {
273 UAbilityTask_WaitDelay* DelayTask = UAbilityTask_WaitDelay::WaitDelay(this, StateDuration);
274 DelayTask->OnFinish.AddDynamic(this, &UDTGameplayAbility_HitReact::OnReactFinished);
275 DelayTask->ReadyForActivation();
276 }
277
278 // Yield instantly if a separate fatal hit lands mid-react.
279 UAbilityTask_WaitGameplayTagAdded* DeathWatchTask = UAbilityTask_WaitGameplayTagAdded::WaitGameplayTagAdd(this, DTGameplayTags::TAG_STATUS_DEAD, nullptr, true);
280 DeathWatchTask->Added.AddDynamic(this, &UDTGameplayAbility_HitReact::OnDeathDuringReact);
281 DeathWatchTask->ReadyForActivation();
282}
