30void UDTGameplayAbility_Channel::BeginChannel()
31{
32 if (!CommitAbility(CurrentSpecHandle, CurrentActorInfo, CurrentActivationInfo))
33 {
34 EndAbility(CurrentSpecHandle, CurrentActorInfo, CurrentActivationInfo, true, true);
35 return;
36 }
37
38 bIsChanneling = true;
39 bChannelEnding = false;
40 bChannelEndBroadcast = false;
41 ChannelTickCount = 0;
42
43 // Periodic resource drain (e.g. mana over time). Handle is stored so it is removed when the channel ends.
44 if (ChannelCostEffect)
45 {
46 ChannelCostHandle = ApplyGameplayEffectToOwner(CurrentSpecHandle, CurrentActorInfo, CurrentActivationInfo, ChannelCostEffect.GetDefaultObject(), GetAbilityLevel());
47 }
48
49 // Optional looping cue on the owner (only if the ability isn't managing cues in Blueprint).
50 if (ChannelLoopCueTag.IsValid())
51 {
52 if (UAbilitySystemComponent* ASC = GetAbilitySystemComponentFromActorInfo())
53 {
54 ASC->AddGameplayCue(ChannelLoopCueTag);
55 bChannelCueAdded = true;
56 }
57 }
58
59 StartChannelMontage();
60 StartInputListener();
61
62 if (MaxChannelDuration > 0.0f)
63 {
64 if (UWorld* World = GetWorld())
65 {
66 World->GetTimerManager().SetTimer(ChannelDurationTimerHandle, this, &ThisClass::OnMaxChannelDurationReached, MaxChannelDuration, false);
67 }
68 }
69
70 NotifyChannelStarted();
71
72 StartChannelTick();
73}

90void UDTGameplayAbility_Channel::StartChannelMontage()
91{
92 UAnimMontage* MontageToPlay = GetAbilityMontageToPlay();
93 ActiveChannelMontage = MontageToPlay;
94 if (!MontageToPlay)
95 {
96 return;
97 }
98
99 ActiveMontageTask = UAbilityTask_PlayMontageAndWait::CreatePlayMontageAndWaitProxy(this, NAME_None, MontageToPlay, 1.0f, ChannelLoopSectionName, true, 1.0f);
100 if (ActiveMontageTask)
101 {
102 ActiveMontageTask->OnCompleted.AddDynamic(this, &ThisClass::OnChannelMontageCompleted);
103 ActiveMontageTask->OnBlendOut.AddDynamic(this, &ThisClass::OnChannelMontageBlendOut);
104 ActiveMontageTask->OnInterrupted.AddDynamic(this, &ThisClass::OnChannelMontageInterrupted);
105 ActiveMontageTask->OnCancelled.AddDynamic(this, &ThisClass::OnChannelMontageCancelled);
106
107 ActiveMontageTask->ReadyForActivation();
108 }

155void UDTGameplayAbility_Channel::HandleChannelTick()
156{
157 if (bChannelEnding || !bIsChanneling)
158 {
159 return;
160 }
161
162 // Resource depletion is only judged on the authority; the server's EndAbility replicates to clients.
163 if (ShouldEvaluateEndConditions() && !CanSustainChannel())
164 {
165 EndChannel(true, false);
166 return;
167 }
168
169 NotifyChannelTick(ChannelTickCount++);
170}

172void UDTGameplayAbility_Channel::EndChannel(bool bResourceDepleted, bool bWasCancelled)
173{
174 if (bChannelEnding)
175 {
176 return;
177 }
178
179 bChannelEnding = true;
180 bIsChanneling = false;
181
182 StopChannelUpkeep();
183
184 if (!bChannelEndBroadcast)
185 {
186 bChannelEndBroadcast = true;
187 NotifyChannelEnded(bResourceDepleted, bWasCancelled);
188 }
189
190 if (bWasCancelled)
191 {
192 FinishChannel(true);
193 return;
194 }
195
196 // Play the outro section and let the montage's completion finish the ability.
197 if (ChannelEndSectionName != NAME_None && IsChannelMontageActive())
198 {
199 MontageJumpToSection(ChannelEndSectionName);
200 return;
201 }
202
203 FinishChannel(false);
204}

206void UDTGameplayAbility_Channel::StopChannelUpkeep()
207{
208 if (ChannelCostHandle.IsValid())
209 {
210 BP_RemoveGameplayEffectFromOwnerWithHandle(ChannelCostHandle);
211 ChannelCostHandle.Invalidate();
212 }
213
214 if (bChannelCueAdded)
215 {
216 if (UAbilitySystemComponent* ASC = GetAbilitySystemComponentFromActorInfo())
217 {
218 ASC->RemoveGameplayCue(ChannelLoopCueTag);
219 }
220
221 bChannelCueAdded = false;
222 }
223
224 if (UWorld* World = GetWorld())
225 {
226 World->GetTimerManager().ClearTimer(ChannelTickTimerHandle);
227 World->GetTimerManager().ClearTimer(ChannelDurationTimerHandle);
228 }
229
230 if (ActiveInputReleaseTask)
231 {
232 ActiveInputReleaseTask->EndTask();
233 ActiveInputReleaseTask = nullptr;
234 }
235
236 if (ActiveInputPressTask)
237 {
238 ActiveInputPressTask->EndTask();
239 ActiveInputPressTask = nullptr;
240 }

264bool UDTGameplayAbility_Channel::CanSustainChannel() const
265{
266 // With no drain effect the channel never ends from resource depletion (input/duration still apply).
267 if (!ChannelCostEffect || !ChannelResourceAttribute.IsValid())
268 {
269 return true;
270 }
271
272 const UAbilitySystemComponent* ASC = GetAbilitySystemComponentFromActorInfo();
273 if (!ASC)
274 {
275 return true;
276 }
277
278 return ASC->GetNumericAttribute(ChannelResourceAttribute) > ChannelEndResourceThreshold;
279}

346void UDTGameplayAbility_Channel::EndAbility(const FGameplayAbilitySpecHandle Handle, const FGameplayAbilityActorInfo* ActorInfo, const FGameplayAbilityActivationInfo ActivationInfo, bool bReplicateEndAbility, bool bWasCancelled)
347{
348 // Safety net: if the ability is ended/cancelled externally (death, group replacement, etc.) without
349 // going through EndChannel, still tear down the drain, cue, timers and input tasks.
350 StopChannelUpkeep();
351
352 if (bIsChanneling && !bChannelEndBroadcast)
353 {
354 bChannelEndBroadcast = true;
355 NotifyChannelEnded(false, bWasCancelled);
356 }
357
358 ActiveMontageTask = nullptr;
359 ActiveChannelMontage = nullptr;
360 bIsChanneling = false;
361 bChannelEnding = false;
362 bChannelEndBroadcast = false;
363 ChannelTickCount = 0;
364
365 Super::EndAbility(Handle, ActorInfo, ActivationInfo, bReplicateEndAbility, bWasCancelled);
