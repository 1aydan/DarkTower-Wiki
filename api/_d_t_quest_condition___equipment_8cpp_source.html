47 FProgressRequirement UDTQuestCondition_Equipment::EvaluateItemRequirements(const TArray<FEquipmentSlotEntry>& EquippedSlots) const
48{
49 // Build per-definition required amounts, skipping null definitions.
50 TMap<const UInventoryItemDefinition*, int32> RequiredAmounts;
51 int32 TotalRequired = 0;
52 for (const FDTQuestEquipmentRequirement& Req : ItemRequirements)
53 {
54 if (!Req.ItemDefinition)
55 {
56 continue;
57 }
58
59 const int32 Amount = FMath::Max(1, Req.RequiredAmount);
60 RequiredAmounts.FindOrAdd(Req.ItemDefinition) += Amount;
61 TotalRequired += Amount;
62 }
63
64 if (RequiredAmounts.IsEmpty())
65 {
66 return {}; // No definition requirements
67 }
68
69 // Count equipped items that match any required definition.
70 TMap<const UInventoryItemDefinition*, int32> EquippedCounts;
71 int32 TotalMatching = 0;
72 for (const FEquipmentSlotEntry& SlotEntry : EquippedSlots)
73 {
74 // Default equipment don't count (Maybe TODO: if we actually want this behavior)
75 if (!SlotEntry.Slot.EquippedItem || SlotEntry.Slot.EquippedItem->IsDefaultEquipment())
76 {
77 continue;
78 }
79
80 const UInventoryItemDefinition* Item = SlotEntry.Slot.EquippedItem->GetItemDefinition();
81 if (Item && RequiredAmounts.Contains(Item))
82 {
83 ++EquippedCounts.FindOrAdd(Item);
84 ++TotalMatching;
85 }
86 }
87
88 FProgressRequirement Out;
89 Out.Target = static_cast<float>(TotalRequired);
90
91 if (bRequireAllDefinitions)
92 {
93 // Every definition must independently meet its required count.
94 for (const auto& [Def, Required] : RequiredAmounts)
95 {
96 const int32 Equipped = EquippedCounts.FindRef(Def);
97 Out.Progress += static_cast<float>(FMath::Min(Equipped, Required));
98 Out.bSatisfied &= Equipped >= Required;
99 }
100 }
101 else
102 {
103 // Any matching item counts toward the total.
104 Out.Progress = static_cast<float>(FMath::Min(TotalMatching, TotalRequired));
105 Out.bSatisfied = TotalMatching >= TotalRequired;
106 }
107
108 return Out;
109}

111FProgressRequirement UDTQuestCondition_Equipment::EvaluateSlotRequirements(const TArray<FEquipmentSlotEntry>& EquippedSlots) const
112{
113 FProgressRequirement Out;
114
115 for (const FDTQuestEquipmentSlotRequirement& SlotReq : SlotRequirements)
116 {
117 if (SlotReq.RequiredSlotTags.IsEmpty())
118 {
119 continue;
120 }
121
122 const int32 Required = FMath::Max(1, SlotReq.RequiredAmount);
123 Out.Target += static_cast<float>(Required);
124
125 int32 Matching = 0;
126 for (const FEquipmentSlotEntry& SlotEntry : EquippedSlots)
127 {
128 if (SlotEntry.Slot.EquippedItem && !SlotEntry.Slot.EquippedItem->IsDefaultEquipment() && IsSlotTagMatched(SlotEntry.SlotTag, SlotReq))
129 {
130 ++Matching;
131 }
132 }
133
134 Out.Progress += static_cast<float>(FMath::Min(Matching, Required));
135 Out.bSatisfied &= Matching >= Required;
136 }
137
138 return Out;

165{
