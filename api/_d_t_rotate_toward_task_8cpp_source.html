102 // Check if we've reached the desired orientation first.
103 if (CalculateAngleDifferenceDot(Pawn->GetActorForwardVector(), ToTarget) >= PrecisionDot)
104 {
105 return EStateTreeRunStatus::Succeeded;
106 }
107
108 const FRotator DesiredRotation = ToTarget.Rotation();
109 const FRotator CurrentRotation = Pawn->GetActorRotation();
110 const FRotator TargetYawRotation(CurrentRotation.Pitch, DesiredRotation.Yaw, CurrentRotation.Roll);
111
112 FRotator NewRotation;
113
114 float YawRate = 0.0f;
115 if (const UCharacterMovementComponent* CMC = InstanceData.MovementComponent.Get())
116 {
117 YawRate = CMC->RotationRate.Yaw;
118 }
119
120 if (YawRate <= 0.0f)
121 {
122 // Snap instantly.
123 NewRotation = TargetYawRotation;
124 }
125 else
126 {
127 NewRotation = FMath::RInterpConstantTo(CurrentRotation, TargetYawRotation, DeltaTime, YawRate);
128 }
129
130 Pawn->SetActorRotation(NewRotation);
131 return EStateTreeRunStatus::Running;
132 }
133
134 FVector FocalPoint = FAISystem::InvalidLocation;
135 if (InstanceData.bFocusActorSet)
136 {
137 if (const AActor* FocusActor = AIController->GetFocusActorForPriority(EAIFocusPriority::Gameplay))
138 {
139 FocalPoint = FocusActor->GetActorLocation();
140 }
141 }
142 else
143 {
144 FocalPoint = AIController->GetFocalPointForPriority(EAIFocusPriority::Gameplay);
145 }
146
147 if (FocalPoint == FAISystem::InvalidLocation)
148 {
149 return EStateTreeRunStatus::Failed;
150 }
151
152 const float AngleDifference = CalculateAngleDifferenceDot(
153 Pawn->GetActorForwardVector(),
154 FocalPoint - Pawn->GetActorLocation());
155
156 if (AngleDifference >= PrecisionDot)
157 {
158 return EStateTreeRunStatus::Succeeded;
159 }
