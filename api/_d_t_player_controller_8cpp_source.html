405void ADTPlayerController::LoadAndSendSaveData()
406{
407 if (!PlayerState)
408 {
409 UE_LOG(LogDT, Warning, TEXT("ADTPlayerController::LoadAndSendSaveData - No PlayerState"));
410 Server_SaveDataTransferComplete();
411 return;
412 }
413
414 UDTSaveSubsystem* SaveSubsystem = GetGameInstance()->GetSubsystem<UDTSaveSubsystem>();
415 if (!SaveSubsystem)
416 {
417 UE_LOG(LogDT, Warning, TEXT("ADTPlayerController::LoadAndSendSaveData - No SaveSubsystem"));
418 Server_SaveDataTransferComplete();
419 return;
420 }
421
422 const FString SlotName = SaveSubsystem->GetCurrentSlotName();
423 const int32 UserIndex = SaveSubsystem->GetCurrentUserIndex();
424
425 if (!SaveSubsystem->DoesSaveExist(SlotName, UserIndex))
426 {
427 UE_LOG(LogDT, Warning, TEXT("ADTPlayerController::LoadAndSendSaveData - No save file exists for slot '%s', sending empty data"), *SlotName);

430 }

432 USaveGame* LoadedSave = UGameplayStatics::LoadGameFromSlot(SlotName, UserIndex);
433 UDTSaveGame* DTSave = Cast<UDTSaveGame>(LoadedSave);
434
435 if (!DTSave)
436 {
437 UE_LOG(LogDT, Warning, TEXT("ADTPlayerController::LoadAndSendSaveData - Failed to load save from slot '%s'"), *SlotName);
438 Server_SaveDataTransferComplete();
439 return;

441
442 TArray<uint8> SaveGameBytes;
443 if (!UDTSaveGame::SerializeSaveGame(DTSave, SaveGameBytes))
444 {
445 UE_LOG(LogDT, Warning, TEXT("ADTPlayerController::LoadAndSendSaveData - Failed to serialize save game"));

447 return;

450 UE_LOG(LogDT, Verbose, TEXT("ADTPlayerController::LoadAndSendSaveData - Loaded and serialized save from slot '%s': %d bytes"),
451 *SlotName, SaveGameBytes.Num());

456void ADTPlayerController::SendSaveDataInChunks(const TArray<uint8>& SaveGameBytes)
457{
458 constexpr int32 ChunkSize = 32 * 1024;
459
460 PendingSaveGameBytes = SaveGameBytes;
461
462 // Calculate chunk count
463 TotalSaveChunks = FMath::Max(1, FMath::CeilToInt(static_cast<float>(SaveGameBytes.Num()) / ChunkSize));
464 CurrentChunkIndex = 0;
465
466 UE_LOG(LogDT, Verbose, TEXT("ADTPlayerController::SendSaveDataInChunks - Starting chunked transfer: %d total chunks (%d bytes)"),
467 TotalSaveChunks, SaveGameBytes.Num());
468

472void ADTPlayerController::SendNextSaveChunk()
473{
474 constexpr int32 ChunkSize = 32 * 1024;
475 constexpr int32 ChunksPerFrame = 2;
476
477 int32 ChunksSentThisFrame = 0;
478
479 while (CurrentChunkIndex < TotalSaveChunks && ChunksSentThisFrame < ChunksPerFrame)
480 {
481 const int32 StartIndex = CurrentChunkIndex * ChunkSize;
482 const int32 EndIndex = FMath::Min(StartIndex + ChunkSize, PendingSaveGameBytes.Num());
483 const int32 CurrentChunkSize = EndIndex - StartIndex;
484
485 TArray<uint8> Chunk;
486 Chunk.Reserve(CurrentChunkSize);
487 Chunk.Append(&PendingSaveGameBytes[StartIndex], CurrentChunkSize);
488
489 Server_ReceiveSaveDataChunk(Chunk, CurrentChunkIndex, TotalSaveChunks);
490
491 CurrentChunkIndex++;
492 ChunksSentThisFrame++;
493 }
494
495 if (CurrentChunkIndex >= TotalSaveChunks)
496 {
497 GetWorldTimerManager().ClearTimer(ChunkSendTimer);
498
499 Server_SaveDataTransferComplete();
500
501 PendingSaveGameBytes.Empty();
502
503 UE_LOG(LogDT, Verbose, TEXT("ADTPlayerController::SendNextChunk - All chunks sent successfully"));
504 }
505}

522{
523 if (!PlayerState)
524 {
525 UE_LOG(LogDT, Warning, TEXT("ADTPlayerController::Server_SaveDataTransferComplete - No PlayerState"));

527 }
528
529 ADTPlayerState* DTPS = Cast<ADTPlayerState>(PlayerState);

540 {
541 FullSaveData.Append(*Chunk);

546 }
547 }
548
549 UE_LOG(LogDT, Verbose, TEXT("ADTPlayerController::Server_SaveDataTransferComplete - Reassembled %d bytes"), FullSaveData.Num());
550

553 {
554 if (!UDTSaveGame::DeserializeSaveGame(FullSaveData, SaveGame, this))

556 UE_LOG(LogDT, Warning, TEXT("ADTPlayerController::Server_SaveDataTransferComplete - Failed to deserialize save game"));
557 }
558 }
559
560 UE_LOG(LogDT, Verbose, TEXT("ADTPlayerController::Server_SaveDataTransferComplete [%s] - Calling LoadFromSaveGame with %s save"),
561 *GetNameSafe(this), SaveGame ? TEXT("valid") : TEXT("null (new player)"));
562
563 DTPS->Execute_LoadFromSaveGame(DTPS, SaveGame);
564
565 if (!GetPawn() && CanRestartPlayer())
566 {
567 UE_LOG(LogDT, Verbose, TEXT("ADTPlayerController::Server_SaveDataTransferComplete [%s] - Save initialized, requesting restart"), *GetNameSafe(this));
568 GetWorldTimerManager().SetTimerForNextTick(this, &APlayerController::ServerRestartPlayer);
569 }
570
571 ReceivedChunks.Reset();
572 ExpectedTotalChunks = 0;
573
574 UE_LOG(LogDT, Verbose, TEXT("ADTPlayerController::Server_SaveDataTransferComplete - Save data applied to PlayerState"));
575}

997void ADTPlayerController::Server_CycleSpectatorTarget_Implementation(bool bForward)
998{
999 if (!HasAuthority())
1000 {
1001 return;
1002 }
1003
1004 // Only valid when this controller is in spectating state.
1005 if (!IsInState(NAME_Spectating))
1006 {
1007 return;
1008 }
1009
1010 const ADTGameState* DTGameState = GetWorld() ? GetWorld()->GetGameState<ADTGameState>() : nullptr;
1011 if (!DTGameState)
1012 {
1013 return;
1014 }
1015
1016 TArray<APlayerState*> AliveStates;
1017 for (APlayerState* PS : DTGameState->PlayerArray)
1018 {
1019 if (!PS || PS == PlayerState)
1020 {
1021 continue;
1022 }
1023
1024 const APawn* PlayerPawn = PS->GetPawn();
1025 if (!PlayerPawn || !PlayerPawn->Implements<UDTTargetableInterface>())
1026 {
1027 continue;
1028 }
1029
1030 UDTAbilityCoreComponent* AbilityCore = IDTTargetableInterface::Execute_GetAbilityCoreComponent(PlayerPawn);
1031 if (AbilityCore && AbilityCore->IsAlive())
1032 {
1033 AliveStates.Add(PS);
1034 }
1035 }
1036
1037 if (AliveStates.Num() == 0)
1038 {
1039 return;
1040 }
1041
1042 APlayerState* CurrentVT = nullptr;
1043 if (PlayerCameraManager && PlayerCameraManager->GetViewTarget())
1044 {
1045 if (APawn* VTPawn = Cast<APawn>(PlayerCameraManager->GetViewTarget()))
1046 {
1047 CurrentVT = VTPawn->GetPlayerState();
1048 }
1049 }
1050
1051 int32 CurrentIdx = AliveStates.IndexOfByKey(CurrentVT);
1052 int32 NextIdx;
1053 if (CurrentIdx == INDEX_NONE)
1054 {
1055 NextIdx = 0;
1056 }
1057 else
1058 {
1059 const int32 Delta = bForward ? 1 : -1;
1060 NextIdx = (CurrentIdx + Delta + AliveStates.Num()) % AliveStates.Num();
1061 }
1062
1063 APlayerState* NextTarget = AliveStates[NextIdx];
1064 if (NextTarget && NextTarget->GetPawn())
1065 {
1066 SetViewTargetWithBlend(NextTarget->GetPawn(), 0.25f);
1067 }
1068
1069 Client_UpdateSpectatorTarget(NextTarget);
1070}
