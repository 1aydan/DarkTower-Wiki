192FVector ADTCompanion::CalculateDesiredLocation() const
193{
194 FVector DesiredLocation;
195
196 if (bIsGoingToLocation)
197 {
198 DesiredLocation = TargetLocation;
199 DesiredLocation.Z += CalculateFloatOffset();
200
201 return DesiredLocation;
202 }
203
204 if (!FollowTarget)
205 {
206 return GetActorLocation();
207 }
208
209 if (FollowTarget->Implements<UDTCompanionTargetInterface>())
210 {
211 DesiredLocation = IDTCompanionTargetInterface::Execute_GetFollowLocation(FollowTarget);
212 }
213 else
214 {
215 DesiredLocation = FollowTarget->GetActorLocation();
216 }
217
218 if (ShouldOrbit())
219 {
220 const float RadianAngle = FMath::DegreesToRadians(CurrentOrbitAngle);
221 const FVector OrbitOffset = FVector(FMath::Cos(RadianAngle) * CurrentOrbitRadius,FMath::Sin(RadianAngle) * CurrentOrbitRadius,0.0f);
222
223 if (bOrbitAroundActorLocation)
224 {
225 DesiredLocation = FollowTarget->GetActorLocation() + OrbitOffset;
226 }
227 else
228 {
229 DesiredLocation += OrbitOffset;
230 }
231 }
232 else
233 {
234 if (!FollowOffset.IsNearlyZero())
235 {
236 DesiredLocation += FollowOffset;
237 }
238 else
239 {
240 DesiredLocation += -FollowTarget->GetActorForwardVector();
241 }
242 }
243
244 DesiredLocation.Z += CalculateFloatOffset();
245
246 return DesiredLocation;
247}

259void ADTCompanion::UpdateFollowMovement(float DeltaTime)
260{
261 if (bEnableFloating)
262 {
263 FloatTimeAccumulator += DeltaTime;
264 }
265
266 if (ShouldOrbit())
267 {
268 CurrentOrbitAngle += OrbitSpeed * DeltaTime;
269 if (CurrentOrbitAngle >= 360.0f)
270 {
271 CurrentOrbitAngle -= 360.0f;
272 }
273
274 CurrentOrbitRadius = FMath::FInterpTo(CurrentOrbitRadius, OrbitRadius, DeltaTime, MovementSpeed);
275 }
276 else if (bOrbitOnlyWhenTargetIdle && FollowTarget)
277 {
278 const FVector ToCompanion = GetActorLocation() - FollowTarget->GetActorLocation();
279 const FVector ToCompanionXY = FVector(ToCompanion.X, ToCompanion.Y, 0.0f);
280 if (!ToCompanionXY.IsNearlyZero())
281 {
282 CurrentOrbitAngle = FMath::RadiansToDegrees(FMath::Atan2(ToCompanionXY.Y, ToCompanionXY.X));
283 CurrentOrbitRadius = ToCompanionXY.Size();
284 }
285 }
286
287 const FVector DesiredLocation = CalculateDesiredLocation();
288 const FVector CurrentLocation = GetActorLocation();
289 FVector NewLocation;
290
291 if (bUseSpringInterpolation)
292 {
293 NewLocation = UKismetMathLibrary::VectorSpringInterp(
294 CurrentLocation,
295 DesiredLocation,
296 SpringVelocity,
297 SpringStiffness,
298 SpringDamping,
299 DeltaTime,
300 SpringMass
301 );
302 }
303 else
304 {
305 NewLocation = FMath::VInterpTo(CurrentLocation, DesiredLocation, DeltaTime, MovementSpeed);
306 }
307
308 if (bPreventWallClipping)
309 {
310 if (WallClipDisabledTimeRemaining > 0.0f)
311 {
312 WallClipDisabledTimeRemaining = FMath::Max(0.0f, WallClipDisabledTimeRemaining - DeltaTime);
313 }
314
315 bool bWasBlocked = false;
316 if (WallClipDisabledTimeRemaining <= 0.0f)
317 {
318 NewLocation = ApplyWallClipPrevention(CurrentLocation, NewLocation, bWasBlocked);
319 }
320
321 if (bEnableWallClipFallback && bWasBlocked)
322 {
323 const float DistanceToDesired = FVector::Dist(CurrentLocation, DesiredLocation);
324 if (DistanceToDesired > WallClipDisableDistance)
325 {
326 WallClipDisabledTimeRemaining = WallClipDisableDuration;
327 }
328 }
329 }
330
331 SetActorLocation(NewLocation);
332 UpdateRotation(DeltaTime, DesiredLocation);
333}

335FVector ADTCompanion::ApplyWallClipPrevention(const FVector& CurrentLocation, const FVector& DesiredTargetLocation, bool& bBlocked) const
336{
337 bBlocked = false;
338
339 if (!GetWorld() || !CollisionComponent || CurrentLocation.Equals(DesiredTargetLocation, KINDA_SMALL_NUMBER))
340 {
341 return DesiredTargetLocation;
342 }
343
344 FCollisionQueryParams Params(SCENE_QUERY_STAT(DTCompanionWallClip), false);
345 Params.AddIgnoredActor(this);
346 Params.AddIgnoredActor(FollowTarget);
347
348 FHitResult Hit;
349
350 if (!GetWorld()->SweepSingleByChannel(Hit,CurrentLocation,DesiredTargetLocation,FQuat::Identity,WallClipTraceChannel,FCollisionShape::MakeSphere(CollisionComponent->GetUnscaledSphereRadius()),Params))
351 {
352 return DesiredTargetLocation;
353 }
354
355 bBlocked = true;
356
357 const FVector HitSafeLocation = Hit.Location + (Hit.Normal * WallClipHitOffset);
358 if (!bSlideAlongWalls)
359 {
360 return HitSafeLocation;
361 }
362
363 const FVector MoveDelta = DesiredTargetLocation - CurrentLocation;
364 const FVector SlideDelta = FVector::VectorPlaneProject(MoveDelta, Hit.Normal);
365 if (SlideDelta.IsNearlyZero())
366 {
367 return HitSafeLocation;
368 }
369
370 const FVector SlideTarget = CurrentLocation + SlideDelta;
371 FHitResult SlideHit;
372
373 if (GetWorld()->SweepSingleByChannel(SlideHit,CurrentLocation,SlideTarget,FQuat::Identity,WallClipTraceChannel,FCollisionShape::MakeSphere(CollisionComponent->GetUnscaledSphereRadius()), Params))
374 {
375 return SlideHit.Location + (SlideHit.Normal * WallClipHitOffset);
376 }
377
378 return SlideTarget;
379}

396bool ADTCompanion::ShouldOrbit() const
397{
398 if (!bEnableOrbit || !FollowTarget || bIsGoingToLocation)
399 {
400 return false;
401 }
402
403 if (!bOrbitOnlyWhenTargetIdle)
404 {
405 return true;
406 }
407
408 const float TargetSpeedSq = FollowTarget->GetVelocity().SizeSquared();
409 return TargetSpeedSq <= FMath::Square(OrbitIdleSpeedThreshold);
