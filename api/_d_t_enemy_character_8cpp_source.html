93void ADTEnemyCharacter::ApplyScaledStats() const
94{
95 checkf(AbilitySystemComponent, TEXT("ADTEnemyCharacter::ApplyScaledStats - Enemy %s has no AbilitySystemComponent"), *GetName());
96
97 const UDTEnemyScalingData* ScalingData = GetScalingData();
98 if (!ScalingData)
99 {
100 return;
101 }
102
103 const UDTAttributeSet* AttributeSet = AbilitySystemComponent->GetSet<UDTAttributeSet>();
104 if (!AttributeSet)
105 {
106 UE_LOG(LogDT, Error, TEXT("ADTEnemyCharacter::ApplyScaledStats - Enemy %s has no DTAttributeSet, cannot apply scaling"), *GetName());
107 return;
108 }
109
110 const ADTGameState* GameState = GetWorld()->GetGameState<ADTGameState>();
111 if (!GameState)
112 {
113 UE_LOG(LogDT, Warning, TEXT("ADTEnemyCharacter::ApplyScaledStats - Enemy %s could not find DTGameState, defaulting to floor 1"), *GetName());
114 return;
115 }
116
117 const int32 CurrentFloor = GameState->GetCurrentDungeonFloor();
118 const float TierMultiplier = GetTierMultiplierFromTags(ScalingData);
119
120 FGameplayEffectSpecHandle SpecHandle = ScalingData->CreateEnemyScalingSpec(AbilitySystemComponent, AttributeSet, CurrentFloor, TierMultiplier);
121
122 if (!SpecHandle.IsValid())
123 {
124 UE_LOG(LogDT, Error, TEXT("Enemy %s failed to create scaling spec"), *GetName());
125 return;
126 }
127
128 AbilitySystemComponent->ApplyGameplayEffectSpecToSelf(*SpecHandle.Data.Get());
129}
