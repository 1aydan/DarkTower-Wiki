79static float CalculateDynamicValueInternal(const FDynamicValueDefinitionBase& Value, const FDTDynamicValueContext& Context, int32 Depth)
80{
81 switch (Value.ValueSource)
82 {
83 case EDynamicDescriptionValueSource::Attribute:
84 {
85 const UAbilitySystemComponent* ASC = (Value.CaptureSource == EDynamicValueAttributeCaptureSource::Target) ? Context.TargetASC.Get() : Context.SourceASC.Get();
86 if (!ASC || !Value.SourceAttribute.IsValid())
87 {
88 return 0.0f;
89 }
90
91 const float CoefficientKey = Context.EvaluationLevel > 0 ? static_cast<float>(Context.EvaluationLevel) : FMath::Clamp(Context.NormalizedValue, 0.0f, 1.0f);
92 const float Coefficient = Value.Coefficient.GetValueAtLevel(CoefficientKey);
93 return ASC->GetNumericAttribute(Value.SourceAttribute) * Coefficient;
94 }
95
96 case EDynamicDescriptionValueSource::AbilityValue:
97 {
98 if (Depth >= MaxDynamicValueReferenceDepth || !Context.OwningDefinition || !Value.SourceValueTag.IsValid())
99 {
100 return 0.0f;
101 }
102
103 if (const FDynamicAbilityValue* Referenced = Context.OwningDefinition->FindDynamicValue(Value.SourceValueTag))
104 {
105 return CalculateAbilityValueInternal(*Referenced, Context, Depth + 1);
106 }
107
108 return 0.0f;
109 }
110
111 case EDynamicDescriptionValueSource::Range:
112 case EDynamicDescriptionValueSource::ScalableFloat:
113 default:
114 return Value.EvaluateScalarAtLevel(Context.EvaluationLevel, Context.NormalizedValue);
115 }
116}

118// Combines an ability value's contributions into damage = (sum of Add) x (1 + sum of Increase) x (product of Multiply),
119// or an Override that replaces the result. The Add bucket is the additive base pool (base + Phys x coeff + flat); the
120// Increase bucket is additive-percentage ((1 + sum), 0.10 = +10%); the Multiply bucket is "more". No contributions => 0.
121static float CalculateAbilityValueInternal(const FDynamicAbilityValue& Value, const FDTDynamicValueContext& Context, int32 Depth)
122{
123 float Sum = 0.0f;
124 float IncreaseSum = 0.0f;
125 float Multiplier = 1.0f;
126 bool bHasOverride = false;
127 float OverrideValue = 0.0f;
128
129 for (const FDynamicValueDefinitionBase& Term : Value.ValueModifiers)
130 {
131 const float TermValue = CalculateDynamicValueInternal(Term, Context, Depth);
132 switch (Term.Operation)
133 {
134 case EStatModifierOp::Multiply:
135 Multiplier *= TermValue;
136 break;
137 case EStatModifierOp::Increase:
138 IncreaseSum += TermValue;
139 break;
140 case EStatModifierOp::Override:
141 OverrideValue = TermValue;
142 bHasOverride = true;
143 break;
144 case EStatModifierOp::Add:
145 default:
146 Sum += TermValue;
147 break;
148 }
149 }
150
151 return bHasOverride ? OverrideValue : Sum * (1.0f + IncreaseSum) * Multiplier;
152}

154float FDynamicValueDefinitionBase::EvaluateScalarAtLevel(int32 EvaluationLevel, float NormalizedValue) const
155{
156 // Range source interpolates between min/max by the normalized roll.
157 if (ValueSource == EDynamicDescriptionValueSource::Range)
158 {
159 return FMath::Lerp(MinValue, MaxValue, FMath::Clamp(NormalizedValue, 0.0f, 1.0f));
160 }
161
162 // ScalableFloat evaluates the scalable curve, either directly at the level or accumulated up to it.
163 if (EvaluationLevel <= 0)
164 {
165 return ScalableValue.GetValueAtLevel(FMath::Clamp(NormalizedValue, 0.0f, 1.0f));
166 }
167
168 if (ScalableEvaluationMode == EScalableValueEvaluationMode::Direct)
169 {
170 return ScalableValue.GetValueAtLevel(static_cast<float>(EvaluationLevel));
171 }
172
173 switch (Operation)
174 {
175 case EStatModifierOp::Add:
176 case EStatModifierOp::Increase:
177 {
178 float AccumulatedValue = 0.0f;
179 for (int32 Level = 1; Level <= EvaluationLevel; ++Level)
180 {
181 AccumulatedValue += ScalableValue.GetValueAtLevel(static_cast<float>(Level));
182 }
183 return AccumulatedValue;
184 }
185
186 case EStatModifierOp::Multiply:
187 {
188 float AccumulatedValue = 1.0f;
189 for (int32 Level = 1; Level <= EvaluationLevel; ++Level)
190 {
191 AccumulatedValue += ScalableValue.GetValueAtLevel(static_cast<float>(Level)) - 1.0f;
192 }
193 return AccumulatedValue;
194 }
195
196 case EStatModifierOp::Override:
197 default:
198 return ScalableValue.GetValueAtLevel(static_cast<float>(EvaluationLevel));
199 }
200}
