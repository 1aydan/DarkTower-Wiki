497void ADTGameplayGameMode::RespawnPlayer(ADTPlayerController* PlayerController)
498{
499 if (!HasAuthority() || !PlayerController)
500 {
501 return;
502 }
503
504 // Determine respawn location strategy from the active mode definition.
505 EDTRespawnLocationType LocationType = EDTRespawnLocationType::PlayerStart;
506 const UDTProgressionData* ProgressionData = UDTBlueprintLibrary::GetProgressionData();
507 if (ProgressionData)
508 {
509 const UDTGameModeDefinition* ModeDef = ProgressionData->FindGameModeDefinition(CurrentDungeonSettings.ModeDefinitionTag);
510 if (ModeDef)
511 {
512 LocationType = ModeDef->RespawnLocationType;
513 }
514 }
515
516 // Get rid of any leftover spectator pawn before respawning.
517 APawn* OldPawn = PlayerController->GetPawn();
518 if (OldPawn)
519 {
520 PlayerController->UnPossess();
521 OldPawn->Destroy();
522 }
523

527 {

537 }
538 }
539
540 if (!bSpawned)
541 {
542 // Fall back to standard PlayerStart-based restart.
543 RestartPlayer(PlayerController);
544 }

552bool ADTGameplayGameMode::FindRespawnTransformNearAlly(const ADTPlayerController* DeadController, FTransform& OutTransform) const
553{
554 if (!DeadController)
555 {
556 return false;
557 }
558
559 const ADTGameState* DTGameState = GetGameState<ADTGameState>();
560 if (!DTGameState)
561 {
562 return false;
563 }
564
565 APawn* DeadPawn = DeadController->GetPawn();
566 const FVector DeadLocation = DeadPawn ? DeadPawn->GetActorLocation() : FVector::ZeroVector;
567
568 APawn* ClosestAlly = nullptr;
569 float ClosestDistSq = TNumericLimits<float>::Max();
570
571 for (APlayerState* PS : DTGameState->PlayerArray)
572 {
573 if (!PS || PS == DeadController->PlayerState)
574 {
575 continue;
576 }
577
578 APawn* CandidatePawn = PS->GetPawn();
579 if (!CandidatePawn || !CandidatePawn->Implements<UDTTargetableInterface>())
580 {
581 continue;
582 }
583
584 if (IDTTargetableInterface::Execute_IsDead(CandidatePawn))
585 {
586 continue;
587 }
588
589 const float DistSq = FVector::DistSquared(CandidatePawn->GetActorLocation(), DeadLocation);
590 if (DistSq < ClosestDistSq)
591 {
592 ClosestDistSq = DistSq;
593 ClosestAlly = CandidatePawn;
594 }
595 }
596
597 if (!ClosestAlly)
598 {
599 return false;
600 }
601
602 // Spawn slightly behind the ally so they don't overlap.
603 const FVector AllyLoc = ClosestAlly->GetActorLocation();
604 const FRotator AllyRot = ClosestAlly->GetActorRotation();
605 const FVector Offset = -AllyRot.Vector() * 200.0f;
606 OutTransform = FTransform(AllyRot, AllyLoc + Offset);
607 return true;
