29void UDTCurrencyComponent::AddCurrency(FGameplayTag CurrencyType, int64 Amount)
30{
31 if (!CurrencyType.IsValid())
32 {
33 UE_LOG(LogDTLoot, Warning, TEXT("UDTCurrencyComponent::AddCurrency — invalid CurrencyType tag"));
34 return;
35 }
36
37 if (Amount <= 0)
38 {
39 UE_LOG(LogDTLoot, Warning, TEXT("UDTCurrencyComponent::AddCurrency — Amount must be > 0 (got %lld)"), Amount);
40 return;
41 }
42
43 const UDTGlobalData* GlobalData = UDTBlueprintLibrary::GetGlobalDataFast();
44 if (!GlobalData)
45 {
46 UE_LOG(LogDTLoot, Warning, TEXT("UDTCurrencyComponent::AddCurrency — could not get GlobalData"));
47 return;
48 }
49
50 const UDTCurrencyGlobalData* CurrencyGlobalData = GlobalData->CurrencyGlobalData;
51 if (!CurrencyGlobalData || !CurrencyGlobalData->CurrencyTypes.Contains(CurrencyType))
52 {
53 UE_LOG(LogDTLoot, Warning, TEXT("UDTCurrencyComponent::AddCurrency — CurrencyType '%s' is not registered in CurrencyGlobalData"), *CurrencyType.ToString());
54 return;
55 }
56
57 FCurrencyBalance& Entry = FindOrAddBalance(CurrencyType);
58 Entry.Amount += Amount;
59
60 BroadcastCurrencyChange(CurrencyType, Entry.Amount, Amount);
61}

83bool UDTCurrencyComponent::SpendCurrency(FGameplayTag CurrencyType, int64 Amount)
84{
85 if (!CurrencyType.IsValid() || Amount <= 0) { return false; }
86
87 FCurrencyBalance* Entry = FindBalance(CurrencyType);
88 if (!Entry || Entry->Amount < Amount) { return false; }
89
90 Entry->Amount -= Amount;
91 BroadcastCurrencyChange(CurrencyType, Entry->Amount, -Amount);
92 return true;
93}

106FCurrencySaveData UDTCurrencyComponent::GetSaveData() const
107{
108 const UDTCurrencyGlobalData* GlobalCurrencyData = nullptr;
109 if (const UDTGlobalData* GlobalData = GetDefault<UDTGlobalData>())
110 {
111 GlobalCurrencyData = GlobalData->CurrencyGlobalData;
112 }
113
114 FCurrencySaveData Data;
115 for (const FCurrencyBalance& Entry : Balances)
116 {
117 if (GlobalCurrencyData && GlobalCurrencyData->IsCurrencyTransient(Entry.CurrencyType))
118 {
119 continue;
120 }
121
122 Data.Balances.Add(Entry.CurrencyType, Entry.Amount);
123 }

170void UDTCurrencyComponent::BroadcastCurrencyChange(FGameplayTag CurrencyType, int64 NewAmount, int64 Delta) const
171{
172 OnCurrencyChanged.Broadcast(CurrencyType, NewAmount, Delta);
173
174 if (Delta > 0)
175 {
176 ADTPlayerState* PS = GetPlayerState<ADTPlayerState>();
177
178 if (UDTAbilitySystemComponent* ASC = Cast<UDTAbilitySystemComponent>(PS->GetAbilitySystemComponent()))
179 {
180 FGameplayEventData Payload;
181 Payload.EventTag = DTGameplayTags::TAG_EVENT_CURRENCY_AWARDED;
182 Payload.EventMagnitude = static_cast<float>(Delta);
183 Payload.Instigator = PS;
184 Payload.Target = PS;
185
186 ASC->SendAbilityEvent(DTGameplayTags::TAG_EVENT_CURRENCY_AWARDED, Payload);
187 }
188 }
