59 case MAT_Box:
60 const FVector BoxCenter = StartLocation + (OwnerActor->GetActorForwardVector() * MeleeAttackParams.HalfExtents.X);

62 break;

66 return TArray<FHitResult>();
67 }

70
71 #if ENABLE_DRAW_DEBUG
72 if (MeleeAttackParams.bDebug)
73 {
74 const FColor DebugColor = ValidHitActors.Num() > 0 ? FColor::Green : FColor::Red;
75
76 switch (MeleeAttackParams.AttackType)
77 {
78 case MAT_Cone:
79 // handled in the ConeOverlapActors function

86 case MAT_Box:

288void UDTGameplayAbility_Melee::ApplyMeleeDamage(const TArray<FHitResult>& Hits)
289{
290 if (Hits.Num() <= 0)

292 UE_LOG(LogDTAbility, VeryVerbose, TEXT("UDTGameplayAbility_Melee::ApplyMeleeDamage - No valid hit actors found."));

294 }

302
303 UDTAbilitySystemComponent* SourceASC = Cast<UDTAbilitySystemComponent>(GetAbilitySystemComponentFromActorInfo());
304 if (!SourceASC)
305 {
306 UE_LOG(LogDTAbility, Warning, TEXT("UDTGameplayAbility_Melee::ApplyMeleeDamage - Could not get SourceASC."));
307 return;
308 }
309
310 const int32 AbilityLevel = GetAbilityLevel();
311
312 for (const FHitResult& Hit : Hits)
313 {
314 AActor* HitActor = Hit.GetActor();
315 if (!HitActor)
316 {
317 continue;
318 }
319
320 if (!IsTargetValid(HitActor))
321 {
322 UE_LOG(LogDTAbility, Verbose, TEXT("UDTGameplayAbility_Melee::ApplyMeleeDamage - Rejected target %s (invalid or out of reach)."), *HitActor->GetName());
323 continue;
324 }
325
326 UAbilitySystemComponent* TargetASC = UAbilitySystemGlobals::GetAbilitySystemComponentFromActor(HitActor);
327 if (!TargetASC)
328 {
329 UE_LOG(LogDTAbility, Verbose, TEXT("UDTGameplayAbility_Melee::ApplyMeleeDamage - Hit actor %s has no ASC, skipping damage application."), *HitActor->GetName());
330 continue;
331 }
332
333 FActiveGameplayEffectHandle DamageEffectHandle;
334 FDTDamageResult DamageResult;
335
336 UDTCombatSubsystem::DealDamageToTarget(this, SourceActor, HitActor, GetMeleeDamage(), MeleeAttackParams.DamageTypeTag, DamageEffectHandle, DamageResult, SourceActor, FGameplayTagContainer(), Hit);
337
338 // Apply any additional on-hit effects (debuffs, knockback, etc.) layered on top of the core damage.
339 FGameplayEffectContextHandle EffectContextHandle = SourceASC->MakeEffectContext();
340 EffectContextHandle.AddHitResult(Hit);
341 EffectContextHandle.SetAbility(this);
342
343 for (const TSubclassOf<UGameplayEffect>& Effect : MeleeAttackParams.MeleeDamageEffects)
344 {
345 if (Effect)
346 {
347 SourceASC->ApplyGameplayEffectToTarget(Effect->GetDefaultObject<UGameplayEffect>(), TargetASC, AbilityLevel, EffectContextHandle);
348 }
349 }
350
351 if (MeleeAttackParams.HitGameplayCueTag.IsValid())
352 {
353 FGameplayEffectContextHandle CueContext = TargetASC->MakeEffectContext();
354
355 CueContext.AddHitResult(Hit);
356 TargetASC->ExecuteGameplayCue(MeleeAttackParams.HitGameplayCueTag, CueContext);
357 }
358
359 OnMeleeHitActor(HitActor, Hit);
360 }
