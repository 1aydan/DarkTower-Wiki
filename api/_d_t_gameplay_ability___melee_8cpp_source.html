44 AActor* OwnerActor = GetAvatarActorFromActorInfo();

46 {

48 return TArray<FHitResult>();
49 }
50
51 FVector StartLocation = OwnerActor->GetActorLocation();
52
53 if (HitboxParams.SocketName != NAME_None)
54 {
55 const ACharacter* OwnerCharacter = Cast<ACharacter>(OwnerActor);
56 USkeletalMeshComponent* Mesh = OwnerCharacter ? OwnerCharacter->GetMesh() : OwnerActor->FindComponentByClass<USkeletalMeshComponent>();
57
58 if (Mesh && Mesh->DoesSocketExist(HitboxParams.SocketName))
59 {
60 StartLocation = Mesh->GetSocketLocation(HitboxParams.SocketName);
61 }
62 else
63 {
64 UE_LOG(LogDTAbility, Warning, TEXT("UDTGameplayAbility_Melee::GetMeleeHitActors - Socket %s not found on %s, using actor location."), *HitboxParams.SocketName.ToString(), *OwnerActor->GetName());
65 }
66 }
67
68 StartLocation += OwnerActor->GetActorRotation().RotateVector(HitboxParams.Offset);
69
70 TArray<AActor*> ActorsToIgnore;
71 ActorsToIgnore.Add(OwnerActor);
72
73 TArray<AActor*> HitActors;
74
75 const bool bDrawDebug = HitboxParams.bDebug || CVarMeleeDebugDraw.GetValueOnGameThread();
76
77 switch (HitboxParams.AttackType)
78 {
79 case MAT_Cone:
80 UDTBlueprintLibrary::ConeOverlapActors(this, StartLocation, OwnerActor->GetActorForwardVector(), HitboxParams.ConeRange, HitboxParams.ConeAngle, MeleeAttackParams.ObjectTypesToHit, AActor::StaticClass(), ActorsToIgnore, HitActors, bDrawDebug);
81 break;
82
83 case MAT_Sphere:
84 UKismetSystemLibrary::SphereOverlapActors(this, StartLocation, HitboxParams.Radius, MeleeAttackParams.ObjectTypesToHit, AActor::StaticClass(), ActorsToIgnore, HitActors);
85 break;
86
87 case MAT_Box:
88 const FVector BoxCenter = StartLocation + (OwnerActor->GetActorForwardVector() * HitboxParams.HalfExtents.X);
89 UKismetSystemLibrary::BoxOverlapActorsWithOrientation(this, BoxCenter, HitboxParams.HalfExtents,OwnerActor->GetActorRotation(), MeleeAttackParams.ObjectTypesToHit, AActor::StaticClass(), ActorsToIgnore, HitActors);
90 break;
91
92 default:
93 UE_LOG(LogDTAbility, Warning, TEXT("UDTGameplayAbility_Melee::GetMeleeHitActors - Invalid MeleeAttackType."));
94 return TArray<FHitResult>();
95 }
96
97 TArray<AActor*> ValidHitActors = HitActors.FilterByPredicate([this](AActor* Actor) { return IsTargetValid(Actor); });
98
99 #if ENABLE_DRAW_DEBUG
100 // Cone draws its own richer debug (shape + per-hit lines) inside ConeOverlapActors above.
101 if (bDrawDebug && HitboxParams.AttackType != MAT_Cone)
102 {
103 const FColor DebugColor = ValidHitActors.Num() > 0 ? FColor::Green : FColor::Red;
104 HitboxParams.DrawDebugShape(GetWorld(), StartLocation, OwnerActor->GetActorRotation(), DebugColor, 2.0f);
105 }
106 #endif
107
108 TArray<FHitResult> HitResults;
109 HitResults.Reserve(ValidHitActors.Num());
110
111 for (AActor* ValidActor : ValidHitActors)
112 {
113 FHitResult Hit;
114
115 Hit.HitObjectHandle = FActorInstanceHandle(ValidActor);
116 Hit.bBlockingHit = true;
117
118 FVector ContactPoint = ValidActor->GetActorLocation();
119 UPrimitiveComponent* HitComponent = Cast<UPrimitiveComponent>(ValidActor->GetRootComponent());
120 if (!HitComponent)
121 {
122 HitComponent = ValidActor->FindComponentByClass<UPrimitiveComponent>();
123 }
124

127 Hit.Component = HitComponent;

129

132 ContactPoint = ClosestPoint;
133 }
134 else
135 {
136 ContactPoint = HitComponent->GetComponentLocation();
137 }
138 }
139
140 Hit.ImpactPoint = ContactPoint;
141 Hit.Location = ContactPoint;
142 Hit.ImpactNormal = (StartLocation - ContactPoint).GetSafeNormal();
143 Hit.Normal = Hit.ImpactNormal;
144 Hit.Distance = FVector::Dist(StartLocation, ContactPoint);
145
146 HitResults.Add(Hit);
147 }

281void UDTGameplayAbility_Melee::ApplyMeleeDamage(const TArray<FHitResult>& Hits)
282{
283 if (Hits.Num() <= 0)
284 {
285 UE_LOG(LogDTAbility, VeryVerbose, TEXT("UDTGameplayAbility_Melee::ApplyMeleeDamage - No valid hit actors found."));
286 return;
287 }

289 AActor* SourceActor = GetAvatarActorFromActorInfo();

293 return;
294 }

296 UDTAbilitySystemComponent* SourceASC = Cast<UDTAbilitySystemComponent>(GetAbilitySystemComponentFromActorInfo());
297 if (!SourceASC)
298 {
299 UE_LOG(LogDTAbility, Warning, TEXT("UDTGameplayAbility_Melee::ApplyMeleeDamage - Could not get SourceASC."));
300 return;
301 }
302
303 const int32 AbilityLevel = GetAbilityLevel();
304
305 for (const FHitResult& Hit : Hits)
306 {
307 AActor* HitActor = Hit.GetActor();
308 if (!HitActor)
309 {
310 continue;
311 }
312
313 if (!IsTargetValid(HitActor))
314 {
315 UE_LOG(LogDTAbility, Verbose, TEXT("UDTGameplayAbility_Melee::ApplyMeleeDamage - Rejected target %s (invalid or out of reach)."), *HitActor->GetName());
316 continue;
317 }
318
319 UAbilitySystemComponent* TargetASC = UAbilitySystemGlobals::GetAbilitySystemComponentFromActor(HitActor);
320 if (!TargetASC)
321 {
322 UE_LOG(LogDTAbility, Verbose, TEXT("UDTGameplayAbility_Melee::ApplyMeleeDamage - Hit actor %s has no ASC, skipping damage application."), *HitActor->GetName());
323 continue;
324 }
325
326 FActiveGameplayEffectHandle DamageEffectHandle;
327 FDTDamageResult DamageResult;
328
329 UDTCombatSubsystem::DealDamageToTarget(this, SourceActor, HitActor, GetMeleeDamage(), MeleeAttackParams.DamageTypeTag, DamageEffectHandle, DamageResult, SourceActor, FGameplayTagContainer(), Hit, nullptr, FGameplayEffectContextHandle(), MeleeAttackParams.HitReactMagnitude, MeleeAttackParams.KnockbackMagnitude);
330
331 FGameplayEffectContextHandle EffectContextHandle = SourceASC->MakeEffectContext();
332 EffectContextHandle.AddHitResult(Hit);
333 EffectContextHandle.SetAbility(this);
334
335 for (const TSubclassOf<UGameplayEffect>& Effect : MeleeAttackParams.MeleeDamageEffects)
336 {
337 if (Effect)
338 {
339 SourceASC->ApplyGameplayEffectToTarget(Effect->GetDefaultObject<UGameplayEffect>(), TargetASC, AbilityLevel, EffectContextHandle);
340 }
341 }
342
343 if (MeleeAttackParams.HitGameplayCueTag.IsValid())
344 {
345 FGameplayEffectContextHandle CueContext = TargetASC->MakeEffectContext();
346
347 CueContext.AddHitResult(Hit);
348 TargetASC->ExecuteGameplayCue(MeleeAttackParams.HitGameplayCueTag, CueContext);
349 }
350
351 OnMeleeHitActor(HitActor, Hit);
352 }
